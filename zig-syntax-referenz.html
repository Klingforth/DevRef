// Das erste Zig-Programm
const std = @import("std");

pub fn main() !void {
    const stdout = std.io.getStdOut().writer();
    try stdout.print("Hallo, Welt!\n", .{});
}

// Kompilieren und ausführen:
// zig build-exe hallo.zig && ./hallo
// zig run hallo.zig

// Zig-Philosophie:
// 1. Keine versteckten Speicherallokationen (kein malloc hinter deinem Rücken!)
// 2. Kein undefiniertes Verhalten still ignoriert
// 3. comptime statt Makros/Templates (zur Compile-Zeit auswerten!)
// 4. Fehlerbehandlung via Fehler-Unions (T!E) – kein try/catch-Stack
// 5. Optionals via ?T – kein Null-Pointer ohne Kennzeichnung
// 6. C-Interop direkt eingebaut (kein FFI nötig)
// 7. Cross-Compilation eingebaut (zig cc als C-Compiler nutzbar!)

// Zig cross-compiliert zu allem:
// zig build-exe src.zig -target x86_64-windows-gnu
// zig build-exe src.zig -target aarch64-linux-musl
// zig build-exe src.zig -target wasm32-freestanding

// Zig als C/C++ Compiler (Drop-in Ersatz!):
// zig cc main.c -o main
// zig c++ main.cpp -o main

const std = @import("std");

pub fn main() void {
    // const – unveränderlich (Stack oder comptime)
    const x: i32 = 42;
    const pi: f64 = 3.14159265358979;

    // var – veränderlich
    var zaehler: u32 = 0;
    zaehler += 1;

    // Typ-Inferenz mit @TypeOf oder einfach weglassen wenn klar
    const y = 100;          // comptime_int
    const f = 3.14;         // comptime_float
    const s = "Hallo";      // *const [5:0]u8

    // undefined – explizit nicht initialisiert (bewusste Entscheidung!)
    var puffer: [1024]u8 = undefined;  // Kein Nulling! Enthält Garbage

    // Blockausdrücke – Blöcke sind Ausdrücke!
    const wert = blk: {
        const a = 10;
        const b = 20;
        break :blk a + b;      // 30
    };

    // Unterstriche in Zahlen (Lesbarkeit)
    const million: u64 = 1_000_000;
    const hex = 0xFF_FF;
    const bin = 0b1111_0000;
    const oct = 0o777;

    // Comptime-Konstanten – zur Compile-Zeit bekannt
    comptime {
        const prüfe = @sizeOf(u64);
        if (prüfe != 8) @compileError("Erwartet 8 Bytes!");
    }

    // Globale Variablen
    // (außerhalb von Funktionen)
    // var global: u32 = 0;           // Mutable
    // const MAX: u32 = 1024;         // Const

    // threadlocal
    // threadlocal var tls_var: u32 = 0;

    _ = x; _ = pi; _ = y; _ = f; _ = s; _ = puffer; _ = wert; _ = million;
}

// Zig ist sehr explizit bei Arithmetik!
// Kein stilles Integer-Overflow (in Debug-Mode: panic!)

// Normale Operatoren (können paniken bei Overflow!)
const a: i32 = 100;
const b: i32 = 200;
const c = a + b;       // 300  (panic in Debug wenn Overflow!)
const d = a - b;       // -100
const e = a * b;       // 20000
const f = @divTrunc(a, 3); // 33  (Ganzzahl-Division Richtung 0)
const g = @rem(a, 3);      // 1   (Rest, Vorzeichen von Dividend)
const h = @mod(a, 3);      // 1   (Rest, immer >= 0)

// Wrapping-Arithmetik (kein Panic, wraps around)
const x: u8 = 255;
const y = x +% 1;      // 0   (wraps! % = wrapping)
const z = x *% 2;      // 254 (wraps)
const w = x -% 10;     // 245 (wraps)

// Sättigungs-Arithmetik (bleibt bei Max/Min)
const s1 = x +| 10;    // 255 (saturates! | = saturating)
const s2 = x *| 2;     // 255 (saturates)

// Bit-Operatoren
a & b      // AND
a | b      // OR
a ^ b      // XOR
~a         // NOT
a << 2     // Shift left
a >> 1     // Shift right (logisch für unsigned, arithmetisch für signed)

// Vergleiche
a == b   a != b
a < b    a <= b   a > b   a >= b

// Logisch
a and b  // Kurzschluss
a or  b  // Kurzschluss
!a       // NOT

const std = @import("std");

// ── STRUCT ────────────────────────────────────────────────────────
const Punkt = struct {
    x: f32,
    y: f32,

    // Methoden – self ist der erste Parameter
    pub fn distanz(self: Punkt, andere: Punkt) f32 {
        const dx = self.x - andere.x;
        const dy = self.y - andere.y;
        return @sqrt(dx * dx + dy * dy);
    }

    // Statische Funktion (kein self)
    pub fn ursprung() Punkt {
        return .{ .x = 0, .y = 0 };
    }
};

// Struct-Literal: .{ .feld = wert }
const p1 = Punkt{ .x = 3.0, .y = 4.0 };
const p2 = Punkt.ursprung();
const abstand = p1.distanz(p2);  // 5.0

// ── ENUM ──────────────────────────────────────────────────────────
const Richtung = enum {
    Nord, Sued, Ost, West,

    pub fn gegenteil(self: Richtung) Richtung {
        return switch (self) {
            .Nord => .Sued,
            .Sued => .Nord,
            .Ost  => .West,
            .West => .Ost,
        };
    }
};

const r = Richtung.Nord;
const g = r.gegenteil();     // .Sued

// Enum mit expliziten Werten
const StatusCode = enum(u16) {
    Ok         = 200,
    NotFound   = 404,
    ServerError = 500,
    _,   // Erlaubt beliebige u16-Werte (non-exhaustive)
};

const code = StatusCode.Ok;
const num = @intFromEnum(code);   // 200
const back = @enumFromInt(StatusCode, 404);  // .NotFound

// ── TAGGED UNION ──────────────────────────────────────────────────
const Form = union(enum) {
    Kreis:     struct { radius: f32 },
    Rechteck:  struct { breite: f32, hoehe: f32 },
    Dreieck:   struct { a: f32, b: f32, c: f32 },

    pub fn flaeche(self: Form) f32 {
        return switch (self) {
            .Kreis     => |k| std.math.pi * k.radius * k.radius,
            .Rechteck  => |r| r.breite * r.hoehe,
            .Dreieck   => |t| blk: {
                const s = (t.a + t.b + t.c) / 2.0;
                break :blk @sqrt(s*(s-t.a)*(s-t.b)*(s-t.c));
            },
        };
    }
};

const kreis = Form{ .Kreis = .{ .radius = 5.0 } };
std.debug.print("Fläche: {d:.2}\n", .{kreis.flaeche()});
Optionals & Fehler-Unions 
const std = @import("std");

// ── OPTIONALS (?T) ────────────────────────────────────────────────
// ?T = T oder null – kein Null-Pointer ohne Kennzeichnung!

fn findeInListe(liste: []const i32, wert: i32) ?usize {
    for (liste, 0..) |item, idx| {
        if (item == wert) return idx;
    }
    return null;
}

const zahlen = [_]i32{3, 1, 4, 1, 5, 9};
const pos = findeInListe(&zahlen, 5);

// if mit Optional-Auspacken
if (pos) |idx| {
    std.debug.print("Gefunden bei {d}\n", .{idx});
} else {
    std.debug.print("Nicht gefunden\n", .{});
}

// orelse – Standardwert wenn null
const idx = pos orelse 0;

// Optionaler Zeiger
var wert: ?i32 = 42;
wert = null;                   // OK

// .? – Optional auspacken (panic wenn null!)
const sicher = wert.?;        // Panic wenn null! Nur wenn sicher

// ── FEHLER-UNIONS (E!T) ───────────────────────────────────────────
// In Zig sind Fehler WERTE, keine Exceptions!
// T!E oder anyerror!T

const DateiError = error{
    DateiNichtGefunden,
    KeineBerechtigungen,
    IO,
};

fn leseDatei(pfad: []const u8) DateiError![]u8 {
    _ = pfad;
    return DateiError.DateiNichtGefunden;
}

// try – propagiert Fehler nach oben (wie ? in Rust)
pub fn main() !void {
    const inhalt = try leseDatei("test.txt");
    // Falls Fehler: main gibt denselben Fehler zurück
    _ = inhalt;
}

// catch – Fehler abfangen
const ergebnis = leseDatei("test.txt") catch |err| blk: {
    std.debug.print("Fehler: {s}\n", .{@errorName(err)});
    break :blk "";
};

// catch mit Standardwert
const daten = leseDatei("cfg.txt") catch "";

// switch auf Fehler
leseDatei("x.txt") catch |err| switch (err) {
    DateiError.DateiNichtGefunden => std.debug.print("404\n", .{}),
    DateiError.KeineBerechtigungen => std.debug.print("403\n", .{}),
    else => return err,          // Andere Fehler weitergeben
};

// errdefer – Cleanup nur bei Fehler
fn alloziere(allocator: std.mem.Allocator) ![]u8 {
    const buf = try allocator.alloc(u8, 1024);
    errdefer allocator.free(buf);   // Nur bei Fehler freigeben!
    try risikoReiche();              // Wenn das fehlschlägt...
    return buf;                     // buf wurde NICHT freigegeben
}

const std = @import("std");

// ── ZEIGER ────────────────────────────────────────────────────────
// Zig-Zeiger sind NIEMALS null (außer ?*T optional pointer)!

var wert: i32 = 42;
const ptr: *i32 = &wert;    // Zeiger auf wert
ptr.* = 99;                  // Dereferenzieren und setzen
std.debug.print("{d}\n", .{wert});    // 99

// Const-Zeiger: Zeiger unveränderlich, Wert unveränderlich
const cptr: *const i32 = &wert;
// cptr.* = 5;  // Compile-Fehler! Wert ist const

// Optionaler Zeiger (nullable)
var opt_ptr: ?*i32 = null;
opt_ptr = &wert;
if (opt_ptr) |p| { p.* = 7; }

// Many-item pointer (für C-Interop, kein Length!)
const mptr: [*]i32 = @ptrCast(&wert);
_ = mptr;

// ── ALLOCATORS ────────────────────────────────────────────────────
// Zig: Kein globaler malloc! Allocator wird explizit übergeben.
// Das ermöglicht: Testing, Memory-Tracking, Custom Allocators!

pub fn beispiel() !void {
    // General Purpose Allocator (für Entwicklung, tracked leaks)
    var gpa = std.heap.GeneralPurposeAllocator(.{}){};
    defer _ = gpa.deinit();          // Prüft auf Memory Leaks!
    const allocator = gpa.allocator();

    // Heap-Allokation
    const buf = try allocator.alloc(u8, 1024);
    defer allocator.free(buf);       // IMMER defer+free!

    // Einzelnes Objekt
    const ptr2 = try allocator.create(i32);
    defer allocator.destroy(ptr2);
    ptr2.* = 42;

    // Arena Allocator (alles auf einmal freigeben!)
    var arena = std.heap.ArenaAllocator.init(allocator);
    defer arena.deinit();            // Alles auf einmal frei!
    const arena_alloc = arena.allocator();

    const arr = try arena_alloc.alloc(i32, 100);
    const str = try arena_alloc.dupe(u8, "kopie");
    _ = arr; _ = str;
    // arena.deinit() gibt alles frei – kein einzelnes free nötig!

    // Stack Allocator (feste Größe, kein Heap!)
    var stack_buf: [4096]u8 = undefined;
    var fba = std.heap.FixedBufferAllocator.init(&stack_buf);
    const stack_alloc = fba.allocator();
    const stack_arr = try stack_alloc.alloc(u8, 100);
    _ = stack_arr;
}

Comptime – Compile-Zeit-Programmierung 
const std = @import("std");

// comptime = zur Compile-Zeit auswerten
// Das ist Zigs Antwort auf Makros, Templates und Generics!

// ── COMPTIME-TYPEN ────────────────────────────────────────────────
// Typen sind comptime-Werte in Zig!
const MeinTyp = i32;           // "Variable" die einen Typ hält
const arr: [5]MeinTyp = undefined;

// Generische Datenstruktur via comptime
fn Stack(comptime T: type) type {
    return struct {
        daten:  []T,
        anzahl: usize,

        const Self = @This();

        pub fn push(self: *Self, item: T) void {
            self.daten[self.anzahl] = item;
            self.anzahl += 1;
        }

        pub fn pop(self: *Self) ?T {
            if (self.anzahl == 0) return null;
            self.anzahl -= 1;
            return self.daten[self.anzahl];
        }
    };
}

const IntStack = Stack(i32);
const StrStack = Stack([]const u8);

// ── COMPTIME-AUSWERTUNG ───────────────────────────────────────────
// Alles was zur Compile-Zeit bekannt ist, kann comptime sein

// Comptime-Blöcke
const TABELLE: [10]i32 = comptime blk: {
    var t: [10]i32 = undefined;
    for (&t, 0..) |*v, i| {
        v.* = @as(i32, @intCast(i * i));  // Quadrate
    }
    break :blk t;
};
// TABELLE = [0,1,4,9,16,25,36,49,64,81] – zur Compile-Zeit!

// ── @typeInfo – Typ-Introspection ────────────────────────────────
fn zeigeFelder(comptime T: type) void {
    const info = @typeInfo(T);
    switch (info) {
        .Struct => |s| {
            std.debug.print("{s} hat {d} Felder:\n",
                .{@typeName(T), s.fields.len});
            inline for (s.fields) |feld| {
                std.debug.print("  {s}: {s}\n",
                    .{feld.name, @typeName(feld.type)});
            }
        },
        else => std.debug.print("Kein Struct\n", .{}),

// Generische Linked List mit comptime T
fn SinglyLinkedList(comptime T: type) type {
    return struct {
        const Self = @This();

        const Node = struct {
            wert:     T,
            naechster: ?*Node,
        };

        kopf:  ?*Node,
        laenge: usize,

        allocator: std.mem.Allocator,

        pub fn init(allocator: std.mem.Allocator) Self {
            return .{ .kopf = null, .laenge = 0, .allocator = allocator };
        }

        pub fn deinit(self: *Self) void {
            var aktuell = self.kopf;
            while (aktuell) |node| {
                const naechster = node.naechster;
                self.allocator.destroy(node);
                aktuell = naechster;
            }
        }

        pub fn pushFront(self: *Self, wert: T) !void {
            const node = try self.allocator.create(Node);
            node.* = .{ .wert = wert, .naechster = self.kopf };
            self.kopf = node;
            self.laenge += 1;
        }

        pub fn popFront(self: *Self) ?T {
            const node = self.kopf orelse return null;
            self.kopf = node.naechster;
            const wert = node.wert;
            self.allocator.destroy(node);
            self.laenge -= 1;
            return wert;
        }

        pub fn contains(self: Self, wert: T) bool {
            var aktuell = self.kopf;
            while (aktuell) |node| {
                if (node.wert == wert) return true;
                aktuell = node.naechster;
            }
            return false;
        }

        pub fn drucke(self: Self) void {
            var aktuell = self.kopf;
            while (aktuell) |node| {
                std.debug.print("{any} -> ", .{node.wert});
                aktuell = node.naechster;
            }
            std.debug.print("null\n", .{});
        }
    };
}

pub fn main() !void {
    var gpa = std.heap.GeneralPurposeAllocator(.{}){};
    defer _ = gpa.deinit();
    const allocator = gpa.allocator();

    // Für jeden Typ eine eigene Liste!
    var int_liste = SinglyLinkedList(i32).init(allocator);
    defer int_liste.deinit();

    try int_liste.pushFront(3);
    try int_liste.pushFront(2);

    // Gleiche Liste für Strings!

const KonfigFehler = error{
    DateiNichtGefunden,
    UngueltigesFormat,
    Fehlendesfeld,
};

const Konfiguration = struct {
    host:     []const u8,
    port:     u16,
    debug:    bool,
    max_verb: u32,
};

// Echte Fehler-Propagation via try
fn parsePort(s: []const u8) KonfigFehler!u16 {
    const n = std.fmt.parseInt(u16, s, 10) catch
        return KonfigFehler.UngueltigesFormat;
    return n;
}

fn parseBool(s: []const u8) KonfigFehler!bool {
    if (std.mem.eql(u8, s, "true"))  return true;
    if (std.mem.eql(u8, s, "false")) return false;
    return KonfigFehler.UngueltigesFormat;
}

fn leseKonfiguration(
    allocator: std.mem.Allocator,
    pfad:      []const u8,
) KonfigFehler!Konfiguration {
    const inhalt = std.fs.cwd().readFileAlloc(
        allocator, pfad, 4096
    ) catch return KonfigFehler.DateiNichtGefunden;
    defer allocator.free(inhalt);

    var host:     ?[]const u8 = null;
    var port:     ?u16        = null;
    var debug:    bool        = false;
    var max_verb: u32         = 100;

    var zeilen = std.mem.tokenizeScalar(u8, inhalt, '\n');
    while (zeilen.next()) |zeile| {
        const trim = std.mem.trim(u8, zeile, " \r");
        if (trim.len == 0 or trim[0] == '#') continue;

        const eq = std.mem.indexOfScalar(u8, trim, '=')
            orelse return KonfigFehler.UngueltigesFormat;

        const schluessel = std.mem.trim(u8, trim[0..eq],   " ");
        const wert       = std.mem.trim(u8, trim[eq+1..],  " ");

        if (std.mem.eql(u8, schluessel, "host")) {
            host = wert;
        } else if (std.mem.eql(u8, schluessel, "port")) {
            port = try parsePort(wert);     // try propagiert Fehler!
        } else if (std.mem.eql(u8, schluessel, "debug")) {
            debug = try parseBool(wert);
        } else if (std.mem.eql(u8, schluessel, "max_verbindungen")) {
            max_verb = std.fmt.parseInt(u32, wert, 10)
                catch return KonfigFehler.UngueltigesFormat;
        }
    }

    return Konfiguration{
        .host     = host     orelse return KonfigFehler.Fehlendesfeld,
        .port     = port     orelse return KonfigFehler.Fehlendesfeld,
        .debug    = debug,
        .max_verb = max_verb,
    };
}

pub fn main() void {
    var gpa = std.heap.GeneralPurposeAllocator(.{}){};
    defer _ = gpa.deinit();

    const cfg = leseKonfiguration(gpa.allocator(), "config.ini") catch |err| {
        std.debug.print("Konfigurationsfehler: {s}\n", .{@errorName(err)});
        std.process.exit(1);
    };

    std.debug.print("Host: {s}:{d} (debug={})\n",
        .{cfg.host, cfg.port, cfg.debug});
}

                  .z = self.z - other.z };
    }

    pub fn scale(self: Vec3, s: f64) Vec3 {
        return .{ .x = self.x * s, .y = self.y * s, .z = self.z * s };
    }

    pub fn dot(self: Vec3, other: Vec3) f64 {
        return self.x*other.x + self.y*other.y + self.z*other.z;
    }

    pub fn normSq(self: Vec3) f64 { return self.dot(self); }
    pub fn norm(self: Vec3) f64   { return @sqrt(self.normSq()); }
};

const Koerper = struct {
    pos:   Vec3,
    vel:   Vec3,
    masse: f64,
};

const G: f64 = 6.674e-11;

fn zeitschritt(koerper: []Koerper, dt: f64) void {
    // Kräfte berechnen
    var kraefte = [_]Vec3{.{.x=0,.y=0,.z=0}} ** 4;

    for (koerper, 0..) |ki, i| {
        for (koerper[i+1..], i+1..) |kj, j| {
            const r   = kj.pos.sub(ki.pos);
            const dist = r.norm() + 1e-10;   // Softening

            const f_mag = G * ki.masse * kj.masse / (dist * dist);
            const f_dir = r.scale(1.0 / dist);

            kraefte[i] = kraefte[i].add(f_dir.scale(f_mag));
            kraefte[j] = kraefte[j].sub(f_dir.scale(f_mag));
        }
    }

    // Leapfrog-Integration
    for (koerper, 0..) |*k, i| {
        k.vel = k.vel.add(kraefte[i].scale(dt / k.masse));
        k.pos = k.pos.add(k.vel.scale(dt));
    }
}

pub fn main() !void {
    const stdout = std.io.getStdOut().writer();

    // Sonnensystem (vereinfacht, in SI-Einheiten)
    const AU = 1.496e11;
    var koerper = [_]Koerper{
        // Sonne
        .{ .pos = .{.x=0,    .y=0, .z=0},
           .vel = .{.x=0,    .y=0, .z=0},
           .masse = 1.989e30 },
        // Erde
        .{ .pos = .{.x=AU,   .y=0, .z=0},
           .vel = .{.x=0,    .y=29784, .z=0},
           .masse = 5.972e24 },
        // Mars
        .{ .pos = .{.x=1.524*AU, .y=0, .z=0},
           .vel = .{.x=0,    .y=24070, .z=0},
           .masse = 6.39e23 },
    };

    // 365 Tage simulieren
    const dt: f64 = 3600.0;   // 1 Stunde
    const schritte: usize = 365 * 24;

    var t: f64 = 0;
    var s: usize = 0;
    while (s < schritte) : (s += 1) {
        zeitschritt(&koerper, dt);
        t += dt;

        if (s % (24 * 30) == 0) {  // Monatlich ausgeben
            const erde = koerper[1];
            const d_von_sonne = erde.pos.norm() / AU;
            try stdout.print("Tag {d:>4}: Erde {d:.3} AU\n",
                .{@as(u32, @intFromFloat(t / 86400.0)), d_von_sonne});
        }
    }
}

// Zeigt Zigs mächtigste Fähigkeit: Typen zur Compile-Zeit erstellen

const std = @import("std");

// Erstelle einen typsicheren Deserializer für jeden Struct-Typ
fn JsonDeserializer(comptime T: type) type {
    return struct {
        pub fn deserialize(allocator: std.mem.Allocator, json: []const u8) !T {
            const parsed = try std.json.parseFromSlice(
                std.json.Value, allocator, json, .{});
            defer parsed.deinit();

            return try fromValue(T, parsed.value, allocator);
        }

        fn fromValue(comptime Typ: type, val: std.json.Value,
                     allocator: std.mem.Allocator) !Typ {
            const info = @typeInfo(Typ);

            return switch (info) {
                .Int => switch (val) {
                    .integer => |n| @intCast(n),
                    else => error.TypMismatch,
                },
                .Float => switch (val) {
                    .float   => |f| @floatCast(f),
                    .integer => |n| @floatFromInt(n),
                    else => error.TypMismatch,
                },
                .Bool => switch (val) {
                    .bool => |b| b,
                    else => error.TypMismatch,
                },
                .Pointer => |ptr| switch (ptr.child) {
                    u8 => switch (val) {
                        .string => |s| try allocator.dupe(u8, s),
                        else => error.TypMismatch,
                    },
                    else => error.NichtUnterstuetzt,
                },
                .Optional => |opt| switch (val) {
                    .null => null,
                    else  => try fromValue(opt.child, val, allocator),
                },
                .Struct => |s| blk: {
                    const obj = switch (val) {
                        .object => |o| o,
                        else => return error.ErwarteObjekt,
                    };
                    var result: Typ = undefined;
                    inline for (s.fields) |feld| {
                        const json_val = obj.get(feld.name) orelse {
                            if (@typeInfo(feld.type) == .Optional) {
                                @field(result, feld.name) = null;
                                continue;
                            }
                            return error.FehlendesField;
                        };
                        @field(result, feld.name) =
                            try fromValue(feld.type, json_val, allocator);
                    }
                    break :blk result;
                },
                else => error.NichtUnterstuetzt,
            };
        }
    };
}

// Verwendung
const User = struct {
    id:    i64,
    name:  []const u8,
    alter: ?i64,
    aktiv: bool,
};

pub fn main() !void {
    var gpa = std.heap.GeneralPurposeAllocator(.{}){};
    defer _ = gpa.deinit();
    const allocator = gpa.allocator();

    const json =
        \{"id": 1, "name": "Alice", "alter": 30, "aktiv": true}
    ;

    const Deser = JsonDeserializer(User);
    const user = try Deser.deserialize(allocator, json);
    defer allocator.free(user.name);

    std.debug.print("User: id={d} name={s} alter={?d} aktiv={}\n",
        .{user.id, user.name, user.alter, user.aktiv});
    // User: id=1 name=Alice alter=30 aktiv=true
}
