/* Definition – meist im :root */
:root{
  --farbe-primär:    #3b82f6;
  --farbe-text:      #1e293b;
  --abstand-base:    1rem;
  --radius-card:     12px;
  --schatten:        0 4px 24px rgba(0, 0, 0, .08);

/* Verwendung */
.button {
  background:    var(--farbe-primär);
  border-radius: var(--radius-card);
  padding:       var(--abstand-base) calc(var(--abstand-base) * 2);
}

/* Fallback-Wert */
color:var(--farbe-text, #333);

/* Überschreiben im Dark Mode */
@media (prefers-color-scheme:dark) {
:root {
    --farbe-text:   #f1f5f9;
    --farbe-primär: #60a5fa;
}
}

/* Lokale Überschreibung */
.card-warning {
  --farbe-primär: #f59e0b;   /* nur innerhalb dieser Komponente */

padding:  clamp(1rem, 5vw, 3rem);

/* min() und max() */

    @layer – Cascade Layers
  

  
/* Schichten definieren (Reihenfolge = Priorität) */
@layer reset, base, components, utilities;

@layer reset {
  *, *::before, *::after { box-sizing: border-box; margin: 0;}
}

@layer base {
  body { font-family: system-ui; line-height: 1.5;}
}

@layer components {
  .button { padding: .5rem 1rem; border-radius: 6px;}
}

@layer utilities {
  .mt-4 { margin-top: 1rem;}
  .hidden { display: none;}
}

/* Höhere Schicht gewinnt – kein !important nötig */
/* Styles außerhalb von @layer gewinnen immer      */

/* Einfache Übergänge */
.button {
  background: #3b82f6;
  transform:  scale(1);
  transition: background 200ms ease,
              transform  150ms ease;

.button:hover{
  background: #1d4ed8;
  transform:  scale(1.05);
}

/* transition Kurzform: property duration timing-function delay */
transition:all 300ms ease-in-out;
transition:color 200ms ease, opacity 300ms ease 100ms;

/* Timing Functions */
ease          /* langsam → schnell → langsam (Standard) */
ease-in       /* langsam → schnell                      */
ease-out      /* schnell → langsam                      */
linear        /* konstant                               */
cubic-bezier(0.34, 1.56, 0.64, 1)  /* Federwirkung    */



  






  
  

    


  

    @keyframes & animation
  

  
/* Keyframes definieren */
@keyframes einblenden {
  from { opacity: 0; transform: translateY(-10px);}
  to   { opacity: 1; transform: translateY(0);}
}

@keyframes pulsieren {
  0%, 100% { transform: scale(1);}
  50%       { transform: scale(1.05);}
}

/* Animation anwenden */
.modal {
  animation: einblenden 300ms ease-out;
}

.badge {
  animation: pulsieren 2s ease-in-out infinite;
}

/* animation Kurzform: name duration timing iteration delay direction */
animation:einblenden 300ms ease-out 1 forwards;
animation-play-state:paused;   /* via JS pausierbar */

/* Performance-Hinweis: Nur transform und opacity animieren!
   Sie laufen auf der GPU und erzeugen kein Layout-Reflow. */



  





  




  
  

    


  

    Media Queries
  

  
/* Mobile First – kleinstes zuerst, dann erweitern */

/* Basis-Styles (mobile) */
.container { padding: 1rem;}
.grid { grid-template-columns: 1fr;}

/* Tablet */
@media (min-width:640px) {
  .container { padding: 2rem;}
  .grid { grid-template-columns: 1fr 1fr;}
}

/* Desktop */
@media (min-width:1024px) {
  .container { max-width: 1200px; margin: 0 auto;}
  .grid { grid-template-columns: repeat(3, 1fr);}
}

/* Feature Queries */
@media (hover:hover) {
  .button:hover { background: var(--farbe-hover);}
}

/* Dunkel-Modus */
@media (prefers-color-scheme:dark) { ...}

/* Reduzierte Bewegung */
@media (prefers-reduced-motion:reduce) {
  * { animation: none !important; transition: none !important;}
}

/* Drucken */
@media print {
  nav, footer { display: none;}
}
