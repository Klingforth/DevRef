        k.id,

CREATE TABLE – vollständiges Beispiel 
CREATE TABLE kunden (
    -- Primärschlüssel (PostgreSQL: SERIAL oder GENERATED)
    id          SERIAL          PRIMARY KEY,

    -- Pflichtfelder (NOT NULL)
    name        VARCHAR(100)    NOT NULL,
    email       VARCHAR(200)    NOT NULL    UNIQUE,
    erstellt_am TIMESTAMP       NOT NULL    DEFAULT NOW(),

    -- Optionale Felder
    stadt       VARCHAR(50),
    alter       INTEGER         CHECK (alter >= 0 AND alter <= 150),
    kontostand  DECIMAL(12, 2)  DEFAULT 0.00,

    -- Foreign Key
    kategorie_id INTEGER REFERENCES kategorien(id)
                         ON DELETE SET NULL
                         ON UPDATE CASCADE
);

-- Index manuell erstellen (beschleunigt Suche nach stadt)
CREATE INDEX idx_kunden_stadt ON kunden(stadt);

-- Unique Composite Index
CREATE UNIQUE INDEX idx_name_email
    ON kunden(name, email);

UPDATE – Daten aktualisieren 
-- Einzelnes Feld aktualisieren
UPDATE kunden
SET kontostand = kontostand + 100
WHERE id = 1;

-- Mehrere Felder gleichzeitig
UPDATE kunden
SET
    stadt      = 'Frankfurt',
    kontostand = 0
WHERE name = 'Dave Fischer';

-- Update mit JOIN (Daten aus anderer Tabelle)
UPDATE kunden k
SET kontostand = kontostand + b.gesamt
FROM (
    SELECT kunden_id, SUM(betrag) AS gesamt
    FROM bestellungen
    GROUP BY kunden_id
) b
WHERE k.id = b.kunden_id;

-- RETURNING – zeigt was geändert wurde (PostgreSQL)
UPDATE kunden
SET kontostand = kontostand * 1.05
WHERE stadt = 'Berlin'
RETURNING id, name, kontostand;

-- Die 2 Kunden mit höchstem Kontostand pro Stadt
WITH rang AS (
    SELECT
        name,
        stadt,
        kontostand,
        ROW_NUMBER() OVER (
            PARTITION BY stadt
            ORDER BY kontostand DESC
        ) AS rn
    FROM kunden
)
SELECT name, stadt, kontostand
FROM rang
WHERE rn <= 2
ORDER BY stadt, kontostand DESC;

-- Kategorie-Tabelle mit Eltern-Kind-Beziehung
-- CREATE TABLE kategorien (id INT, name TEXT, eltern_id INT);

WITH RECURSIVE kategorie_pfad AS (
    -- Basisfall: Startkategorie
    SELECT
        id,
        name,
        eltern_id,
        name::TEXT AS pfad,
        0 AS tiefe
    FROM kategorien
    WHERE eltern_id IS NULL   -- Wurzelkategorien

    UNION ALL

    -- Rekursiver Fall: Kinder der bisherigen
    SELECT
        k.id,
        k.name,
        k.eltern_id,
        kp.pfad || ' > ' || k.name,
        kp.tiefe + 1
    FROM kategorien k
    JOIN kategorie_pfad kp ON k.eltern_id = kp.id
)
SELECT tiefe, name, pfad
FROM kategorie_pfad
ORDER BY pfad;

-- Ergebnis:
-- 0 | Elektronik     | Elektronik
-- 1 | Computer       | Elektronik > Computer
-- 2 | Laptops        | Elektronik > Computer > Laptops
-- 1 | Smartphones    | Elektronik > Smartphones
-- Umsatz pro Monat und Stadt als Kreuztabelle
SELECT
    TO_CHAR(datum, 'YYYY-MM') AS monat,
    SUM(CASE WHEN k.stadt = 'Berlin'  THEN b.betrag ELSE 0 END) AS berlin,
    SUM(CASE WHEN k.stadt = 'Hamburg' THEN b.betrag ELSE 0 END) AS hamburg,
    SUM(CASE WHEN k.stadt = 'München' THEN b.betrag ELSE 0 END) AS münchen,
    SUM(b.betrag)                                                  AS gesamt
FROM bestellungen b
JOIN kunden k ON b.kunden_id = k.id
GROUP BY TO_CHAR(datum, 'YYYY-MM')
ORDER BY monat;

-- PostgreSQL hat auch crosstab() via tablefunc-Extension
-- für dynamisches Pivoting
-- Zusammenhängende Bereiche in Sequenzen finden
-- Typisch: lückenlose Zeiträume, aufeinanderfolgende IDs

-- Beispiel: Benutzer war wie lange aktiv? (online_log Tabelle)
-- CREATE TABLE online_log (user_id INT, tag DATE, online BOOL);

WITH nummeriert AS (
    SELECT
        user_id,
        tag,
        tag - (ROW_NUMBER() OVER (
            PARTITION BY user_id
            ORDER BY tag
        ) * INTERVAL '1 day')::DATE AS gruppe
    FROM online_log
    WHERE online = TRUE
),
islands AS (
    SELECT
        user_id,
        MIN(tag) AS von,
        MAX(tag) AS bis,
        COUNT(*) AS anzahl_tage
    FROM nummeriert
    GROUP BY user_id, gruppe
)
SELECT
    user_id,
    von,
    bis,
    anzahl_tage
FROM islands
ORDER BY user_id, von;

-- Ergebnis:
-- user_id | von        | bis        | anzahl_tage
