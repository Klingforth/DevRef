-- Haskell ist ANDERS als alle anderen Sprachen im DevRef:
--
-- 1. PURE: Funktionen haben keine Seiteneffekte
--    f(x) gibt IMMER denselben Wert zurück – garantiert
--
-- 2. LAZY: Werte werden erst berechnet wenn nötig
--    Unendliche Listen sind möglich!
--
-- 3. STARK TYPISIERT: Der Compiler kennt jeden Typ
--    Wenn es kompiliert, hat es oft keine Laufzeitfehler
--
-- 4. FUNKTIONEN SIND WERTE
--    Funktionen können übergeben, zurückgegeben, kombiniert werden
--
-- 5. KEINE STATEMENTS – nur Ausdrücke
--    Kein "x = 5; x = 6;" – Variablen sind unveränderlich!

Maybe-Monad – sicheres Null-Handling 
-- Maybe als Monad: Fehler propagieren ohne if/null-Checks

-- Das Problem ohne Monad:
lookupUser1 :: Int -> Maybe String
lookupUser1 1 = Just "Alice"
lookupUser1 _ = Nothing

lookupEmail1 :: String -> Maybe String
lookupEmail1 "Alice" = Just "alice@example.com"
lookupEmail1 _       = Nothing

-- Ohne Monad: nervige Verschachtelung
getUserEmail_manuell :: Int -> Maybe String
getUserEmail_manuell uid =
    case lookupUser1 uid of
        Nothing   -> Nothing
        Just name -> case lookupEmail1 name of
            Nothing    -> Nothing
            Just email -> Just email

-- Mit Monad (>>= = bind):
getUserEmail_bind :: Int -> Maybe String
getUserEmail_bind uid = lookupUser1 uid >>= lookupEmail1

-- Mit do-Notation (sieht aus wie imperativer Code!):
getUserEmail :: Int -> Maybe String
getUserEmail uid = do
    name  <- lookupUser1 uid    -- wenn Nothing: ganzer Ausdruck Nothing
    email <- lookupEmail1 name  -- wenn Nothing: ganzer Ausdruck Nothing
    return email                 -- = Just email

getUserEmail 1    -- Just "alice@example.com"
getUserEmail 99   -- Nothing

-- do-Notation für Listen (= Listenverständnis!)
pythagoreisch :: [(Int, Int, Int)]
pythagoreisch = do
    c <- [1..20]
    b <- [1..c]
    a <- [1..b]
    if a^2 + b^2 == c^2     -- guard
        then return (a,b,c)
        else []

-- Identisch mit List Comprehension:
pythagoreisch' =
    [ (a,b,c)
    | c <- [1..20]
    , b <- [1..c]
    , a <- [1..b]
    , a^2 + b^2 == c^2
    ]

-- guard aus Control.Monad
import Control.Monad (guard)
pythagoreisch'' = do
    c <- [1..20]; b <- [1..c]; a <- [1..b]
    guard (a^2 + b^2 == c^2)
    return (a,b,c)

module Parser where

import Data.Char (isDigit, isAlpha, isSpace)

-- Ein Parser ist eine Funktion die einen String nimmt
-- und entweder (Wert, Rest) oder Fehler zurückgibt
newtype Parser a = Parser { runParser :: String -> Maybe (a, String) }

-- Functor: Ergebnis transformieren
instance Functor Parser where
    fmap f (Parser p) = Parser $ input ->
        case p input of
            Nothing        -> Nothing
            Just (x, rest) -> Just (f x, rest)

-- Applicative: Parser sequenzieren
instance Applicative Parser where
    pure x = Parser $ input -> Just (x, input)
    (Parser pf) <*> (Parser pa) = Parser $ input ->
        case pf input of
            Nothing       -> Nothing
            Just (f, rest) -> case pa rest of
                Nothing        -> Nothing
                Just (x, rest') -> Just (f x, rest')

-- Monad: Parser abhängig voneinander
instance Monad Parser where
    return = pure
    (Parser p) >>= f = Parser $ input ->
        case p input of
            Nothing       -> Nothing
            Just (x, rest) -> runParser (f x) rest

-- Alternative: erstes oder zweites versuchen
instance Alternative Parser where
    empty = Parser $ const Nothing
    (Parser p1) <|> (Parser p2) = Parser $ input ->
        p1 input <|> p2 input

-- Primitiver Parser: ein Zeichen
item :: Parser Char
item = Parser $ case
    []     -> Nothing
    (c:cs) -> Just (c, cs)

-- Prüfe Bedingung
satisfy :: (Char -> Bool) -> Parser Char
satisfy pred = do
    c <- item
    if pred c then return c else empty

-- Konkrete Parser
digit :: Parser Char
digit = satisfy isDigit

letter :: Parser Char
letter = satisfy isAlpha

space :: Parser Char
space = satisfy isSpace

-- Kombinatoren
many :: Parser a -> Parser [a]
many p = some p <|> pure []

some :: Parser a -> Parser [a]
some p = (:) <$> p <*> many p

-- Zahl parsen
natural :: Parser Int
natural = read <$> some digit

-- Verwendung
runParser natural "123abc"    -- Just (123, "abc")
runParser natural "abc"       -- Nothing
runParser (some letter) "hello world"  -- Just ("hello", " world")

    | gerade    = (sorted !! halbe + sorted !! (halbe-1)) / 2

    gerade = even n

module Sortieren where

-- QuickSort – der klassische Haskell-Einzeiler
quicksort :: Ord a => [a] -> [a]
quicksort []     = []
quicksort (x:xs) = quicksort kleiner ++ [x] ++ quicksort grösser
  where
    kleiner = filter (<  x) xs
    grösser = filter (>= x) xs

-- Merge Sort
mergesort :: Ord a => [a] -> [a]
mergesort []  = []
mergesort [x] = [x]
mergesort xs  = merge (mergesort links) (mergesort rechts)
  where
    (links, rechts) = splitAt (length xs `div` 2) xs
    merge [] ys = ys
    merge xs [] = xs
    merge (x:xs) (y:ys)
        | x <= y    = x : merge xs (y:ys)
        | otherwise = y : merge (x:xs) ys

-- Insertion Sort
insertionsort :: Ord a => [a] -> [a]
insertionsort = foldr einfügen []
  where
    einfügen x [] = [x]
    einfügen x (y:ys)
        | x <= y    = x : y : ys
        | otherwise = y : einfügen x ys

-- Sortieren mit Schlüssel
sortierNach :: Ord b => (a -> b) -> [a] -> [a]
sortierNach f []     = []
sortierNach f (x:xs) = sortierNach f kleiner ++ [x] ++ sortierNach f grösser
  where
    kleiner = filter (y -> f y <  f x) xs
    grösser = filter (y -> f y >= f x) xs

main :: IO ()
main = do
    let zahlen = [5, 2, 8, 1, 9, 3, 7, 4, 6]
    putStrLn $ "Original:    " ++ show zahlen
    putStrLn $ "Quicksort:   " ++ show (quicksort zahlen)
    putStrLn $ "Mergesort:   " ++ show (mergesort zahlen)

    let wörter = ["Haskell", "ist", "wunderschön", "und", "mächtig"]
    putStrLn $ "Nach Länge:  " ++ show (sortierNach length wörter)
