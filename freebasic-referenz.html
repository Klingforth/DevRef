Wie FreeBasic funktioniert
' FREEBASIC - KEIN Interpreter, sondern Compiler!
' Quellcode (.bas) → fbc → Maschinencode (.exe / ELF)
' Keine Zeilennummern nötig
' Case-insensitiv: DIM = Dim = dim

' Kompilieren (Terminal):
'   fbc meinprogramm.bas          → meinprogramm (Linux) / meinprogramm.exe (Win)
'   fbc -exx meinprogramm.bas     → mit Fehlerprüfung (Bounds, Null-Ptr)
'   fbc -O 2 meinprogramm.bas     → mit Optimierung
'   fbc -lang qb meinprogramm.bas → QB/QBASIC-Kompatibilitätsmodus

' Typisches Programm:
Print "Hallo, Welt!"
Sleep   ' Warte auf Tastendruck (Console bleibt offen)
End 0

' Kommentare:
' Das ist ein Kommentar (einfaches Anführungszeichen)

x = 5 : y = 10 : Print x + y

' Zeilenfortsetzung mit Underscore:
Dim ergebnis As Integer _
    = 100 + 200

' FUNCTION gibt einen Wert zurück
Function Addiere(ByVal a As Integer, ByVal b As Integer) As Integer
  Return a + b           ' Return verlässt die Funktion sofort
End Function

' Alternativ: Funktionsname als Rückgabevariable (QB-Stil)
Function Fakultaet(ByVal n As Integer) As LongInt
  If n <= 1 Then
    Fakultaet = 1           ' Rückgabe via Funktionsname
  Else
    Fakultaet = n * Fakultaet(n - 1)  ' Rekursion
  End If
End Function

' STRING zurückgeben
Function WiederholenStr(s As String, n As Integer) As String
  Dim r As String
  For i As Integer = 1 To n : r &= s : Next
  Return r
End Function

' TYPE MIT METHODEN (OOP-Erweiterung)
Type Vektor2D
  x As Double
  y As Double
  Declare Function Laenge() As Double
  Declare Sub Normalisiere()
End Type

Function Vektor2D.Laenge() As Double
  Return Sqr(This.x^2 + This.y^2)
End Function

Sub Vektor2D.Normalisiere()
  Dim l As Double = This.Laenge()
  If l <> 0 Then This.x /= l : This.y /= l
End Sub

Dim v As Vektor2D
v.x = 3.0 : v.y = 4.0
Print v.Laenge()              ' => 5.0
v.Normalisiere()

' *** BUBBLESORT – FreeBasic ***

Sub BubbleSort(arr() As Integer)
  Dim n As Integer = UBound(arr)
  Dim getauscht As Boolean
  Dim tmp As Integer

  Do
    getauscht = False
    For i As Integer = 0 To n - 1
      If arr(i) > arr(i + 1) Then
        tmp          = arr(i)
        arr(i)       = arr(i + 1)
        arr(i + 1) = tmp
        getauscht    = True
      End If
    Next
  Loop While getauscht
End Sub

BubbleSort(zahlen())

Print "Nachher: ";
For i As Integer = 0 To 7 : Print zahlen(i); " "; : Next : Print

' *** LINKED LIST mit Type + Ptr – FreeBasic ***

Type Node
  wert    As Integer
  naechst As Node Ptr
End Type

Function NeuerKnoten(ByVal v As Integer) As Node Ptr
  Dim n As Node Ptr = Allocate(SizeOf(Node))
  n->wert    = v
  n->naechst = 0
  Return n
End Function

Sub ListeAusgeben(kopf As Node Ptr)
  Dim cur As Node Ptr = kopf
  While cur <> 0
    Print cur->wert; " -> ";
    cur = cur->naechst
  Wend
  Print "NIL"
End Sub

Dim a As Node Ptr = NeuerKnoten(10)
Dim b As Node Ptr = NeuerKnoten(20)
Dim c As Node Ptr = NeuerKnoten(30)
a->naechst = b
b->naechst = c

ListeAusgeben(a)   ' => 10 -> 20 -> 30 -> NIL

DeAllocate(a) : DeAllocate(b) : DeAllocate(c)
Sleep
' *** DATEI-LOGGER – FreeBasic ***

Type Logger
  dateiname As String
  fn        As Integer
  Declare Sub Oeffnen(pfad As String)
  Declare Sub Log(nachricht As String)
  Declare Sub Schliessen()
End Type

Sub Logger.Oeffnen(pfad As String)
  This.dateiname = pfad
  This.fn = FreeFile()
  Open pfad For Append As #This.fn
End Sub

Sub Logger.Log(nachricht As String)
  Dim zeitstempel As String = Format(Now(), "yyyy-mm-dd hh:mm:ss")
  Dim zeile       As String = "[" & zeitstempel & "] " & nachricht
  Print #This.fn, zeile
  Print zeile               ' Auch auf Konsole
End Sub

Sub Logger.Schliessen()
  Close #This.fn
End Sub

Dim L As Logger
L.Oeffnen("app.log")
L.Log("Programm gestartet")
L.Log("Berechnung läuft...")
L.Log("Fertig.")
L.Schliessen()
Sleep
