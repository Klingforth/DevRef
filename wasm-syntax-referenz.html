Was WebAssembly ist 
;; WebAssembly (Wasm) – W3C Standard 2019
;; Binäres Laufzeitformat + portables Kompilierungsziel
;; WAT = WebAssembly Text Format (menschenlesbar)

;; STACK-MASCHINE: Instruktionen schieben Werte auf einen Stack
;; Kein Register-Konzept wie x86 – alles geht über den Stack

;; Das einfachste Wasm-Modul:
(module
  (func $addiere (param $a i32) (param $b i32) (result i32)
    local.get $a     ;; Stack: [a]
    local.get $b     ;; Stack: [a, b]
    i32.add          ;; Stack: [a+b]
  )
  (export "addiere" (func $addiere))
)

;; WARUM WASM?
;; 1. Near-native Performance im Browser (besser als JS für CPU-intensive Tasks)
;; 2. Sicher: sandboxed, kein Zugriff auf Speicher außerhalb
;; 3. Portabel: läuft überall (Browser, Server, Edge, IoT)
;; 4. Kompaktes Binärformat (kleiner als JS, schnelles Parsen)
;; 5. Sprachunabhängig: Kompilierungsziel für C, C++, Rust, Go, Zig...

;; TYPEN – nur 4 Werttypen (MVP):
;; i32  – 32-Bit Integer (Standard für Boolesche Werte!)
;; i64  – 64-Bit Integer
;; f32  – 32-Bit Float
;; f64  – 64-Bit Float (Standard-Float)
;; + v128 – 128-Bit SIMD Vektor (neuere Erweiterung)
;; + funcref, externref – Referenztypen

;; KEIN:
;; - Heap (nur lineares Speicher-Array: memory)
;; - Strings (sind Byte-Arrays im Speicher)
;; - Exceptions (wurden später hinzugefügt)
;; - Threads (optional, Atomics)
;; - GC (optional, GC-Proposal)

;; Konvertieren & ausführen:
;; wat2wasm datei.wat -o datei.wasm    (wabt-tools)
;; node -e "WebAssembly.instantiateStreaming(fetch('f.wasm'))"
;; wasmtime datei.wasm                 (standalone runtime)
Modulstruktur & Sektionen 
;; Ein vollständiges Wasm-Modul zeigt alle Sektionen

(module
  ;; ── TYP-SEKTION: Funktionssignaturen ─────────────────────────
  (type $add_type (func (param i32 i32) (result i32)))

  ;; ── IMPORT-SEKTION: Aus Host-Umgebung importieren ─────────────
  ;; Format: (import "modul" "name" ...)
  (import "env" "log_int" (func $log_int (param i32)))
  (import "env" "speicher" (memory 1))   ;; Mind. 1 Page (64 KB)

  ;; ── SPEICHER: 1 Page = 64 KB ──────────────────────────────────
  ;; (memory 1)          ;; Min 1 Page
  ;; (memory 1 10)       ;; Min 1, Max 10 Pages
  ;; (memory (export "mem") 1)  ;; Direkt exportieren

  ;; ── GLOBALE VARIABLEN ─────────────────────────────────────────
  (global $zaehler (mut i32) (i32.const 0))
  (global $PI f64 (f64.const 3.14159265358979))

  ;; ── TABELLE (für indirekten Aufruf / funcref) ─────────────────
  (table 4 funcref)

  ;; ── DATENSEKTION: Speicher initialisieren ─────────────────────
  (data (i32.const 0) "Hallo, Wasm!")   ;; Null-terminierter String

  ;; ── FUNKTIONEN ────────────────────────────────────────────────
  (func $addiere (param $a i32) (param $b i32) (result i32)
    local.get $a
    local.get $b
    i32.add
  )

  (func $start
    i32.const 42
    call $log_int
  )

  ;; ── START: automatisch beim Laden aufrufen ────────────────────
  (start $start)

  ;; ── EXPORT-SEKTION ────────────────────────────────────────────
  (export "addiere"  (func $addiere))
  (export "zaehler"  (global $zaehler))
  ;; (export "mem"   (memory 0))       ;; Speicher exportieren
)
Stack-Maschine verstehen 
;; WASM IST EINE STACK-MASCHINE
;; Instruktionen konsumieren Argumente vom Stack und pushen Ergebnisse

;; Beispiel: (a + b) * (c - d)
(func $komplex
  (param $a i32) (param $b i32)
  (param $c i32) (param $d i32)
  (result i32)

  local.get $a    ;; Stack: [a]
  local.get $b    ;; Stack: [a, b]
  i32.add         ;; Stack: [a+b]         -- konsumiert a,b; pusht Ergebnis

  local.get $c    ;; Stack: [a+b, c]
  local.get $d    ;; Stack: [a+b, c, d]
  i32.sub         ;; Stack: [a+b, c-d]

  i32.mul         ;; Stack: [(a+b)*(c-d)] -- fertig!
)

;; 1. Flach (Stack-Stil) – wie echtes WAT
(func $add_flach (param i32 i32) (result i32)
  local.get 0     ;; Index statt Name!
  local.get 1
  i32.add
)

;; 2. Verschachtelt (S-Expression-Stil) – lesbarer
(func $add_nested (param i32 i32) (result i32)
  (i32.add
    (local.get 0)
    (local.get 1)
  )
)
;; Beide erzeugen IDENTISCHES Binärformat!

;; LOKALE VARIABLEN
(func $mit_locals (param $x i32) (result i32)
  (local $temp i32)    ;; Lokale Variable deklarieren
  (local $erg  i32)

  local.get $x
  i32.const 2
  i32.mul
  local.set $temp       ;; temp = x * 2

  local.get $temp
  i32.const 1
  i32.add
  local.tee $erg        ;; erg = temp+1, LÄSST Wert auf Stack!

  ;; Stack enthält noch erg → ist der Rückgabewert
)

;; ── IF / ELSE ────────────────────────────────────────────────────
;; Bedingung ist i32: 0 = false, alles andere = true

(func $ist_positiv (param $n i32) (result i32)
  local.get $n
  i32.const 0
  i32.gt_s        ;; n > 0 → 1 oder 0
  if (result i32) ;; if konsumiert i32 vom Stack
    i32.const 1   ;; dann-Zweig
  else
    i32.const 0   ;; sonst-Zweig
  end
)

;; if ohne Ergebnis (void)
(func $drucke_wenn_positiv (param $n i32)
  local.get $n
  i32.eqz
  if              ;; if n == 0: überspringe
  else
    local.get $n
    call $log_int
  end
)

;; ── BLOCK: strukturierter Sprung ──────────────────────────────────
;; br $label = break/springe zum Ende von block
;; (aus block heraus = nach block)

(func $suche (param $arr i32) (param $len i32) (param $val i32)
  (result i32)
  (local $i i32)
  (local $found i32)

  block $gefunden
    loop $schleife
      local.get $i
      local.get $len
      i32.ge_u
      br_if $gefunden     ;; i >= len → Schleife verlassen

      ;; arr[i] laden
      local.get $arr      ;; Basis-Adresse
      local.get $i
      i32.const 4
      i32.mul
      i32.add             ;; Adresse = arr + i*4
      i32.load            ;; Wert laden

      local.get $val
      i32.eq
      if
        local.get $i
        local.set $found
        br $gefunden      ;; Gefunden! Raus aus block
      end

      local.get $i
      i32.const 1
      i32.add
      local.set $i        ;; i++

      br $schleife        ;; Zurück zum Schleifen-Anfang
    end                   ;; end loop
  end                     ;; end block

  local.get $found
)

;; ── LOOP ─────────────────────────────────────────────────────────
;; ACHTUNG: br in loop = zurück zum ANFANG (nicht Ende!)
;; br in block = zum ENDE (heraus)

(func $summe (param $n i32) (result i32)
  (local $i i32)
  (local $acc i32)

  loop $wieder
    local.get $i
    local.get $n
    i32.lt_s          ;; i < n?
    if
      local.get $acc
      local.get $i
      i32.add
      local.set $acc  ;; acc += i

      local.get $i
      i32.const 1
      i32.add
      local.set $i    ;; i++

      br $wieder      ;; Weiter
    end
  end

  local.get $acc      ;; Ergebnis zurück
)
Funktionen & Aufrufe 
;; ── DIREKTE AUFRUFE ──────────────────────────────────────────────
(module
  (func $quadrat (param $x i32) (result i32)
    local.get $x
    local.get $x
    i32.mul
  )

  (func $summe_quadrate (param $a i32) (param $b i32) (result i32)
    local.get $a
    call $quadrat       ;; a²  – Stack: [a²]

    local.get $b
    call $quadrat       ;; b²  – Stack: [a², b²]

    i32.add             ;; a² + b²
  )

  ;; ── MEHRERE RÜCKGABEWERTE (MVP 2.0) ──────────────────────────
  (func $min_max (param $a i32) (param $b i32)
    (result i32 i32)    ;; Zwei Rückgabewerte!
    local.get $a
    local.get $b
    i32.lt_s
    if (result i32 i32)
      local.get $a      ;; min, max
      local.get $b
    else
      local.get $b      ;; min, max
      local.get $a
    end
  )

  ;; Destrukturierung: beide Werte landen auf Stack
  (func $test
    i32.const 5
    i32.const 3
    call $min_max       ;; Stack: [3, 5]  (min unten, max oben)
    drop                ;; max verwerfen
    drop                ;; min verwerfen
  )

  ;; ── INDIREKTER AUFRUF (funcref / Tabelle) ────────────────────
  (table 2 funcref)
  (elem (i32.const 0) $quadrat $summe_quadrate)

  (func $dispatch (param $idx i32) (param $x i32) (result i32)
    local.get $x        ;; Argument für Funktion
    local.get $idx      ;; Tabellen-Index
    call_indirect (type 0) ;; Aufruf via Tabelle – wie Funktionszeiger!
  )

  (export "summe_quadrate" (func $summe_quadrate))
  (export "dispatch"       (func $dispatch))
)
Speicher & Strings 
;; Wasm-Speicher = lineares Byte-Array
;; Kein Heap-Management – du machst alles selbst (oder via Compiler)
;; 1 Memory Page = 65.536 Bytes (64 KB)

(module
  (memory (export "mem") 1)   ;; 1 Page = 64KB, auch exportieren

  ;; Konstante Strings in Datasektion
  (data (i32.const 100) "Hallo, Wasm!")  ;; bei Offset 100

  ;; Zahlen schreiben und lesen
  (func $set_int (param $addr i32) (param $val i32)
    local.get $addr
    local.get $val
    i32.store           ;; 4 Bytes schreiben
  )

  (func $get_int (param $addr i32) (result i32)
    local.get $addr
    i32.load            ;; 4 Bytes lesen
  )

  ;; Array-Zugriff (4-Byte-Integers)
  (func $arr_get (param $base i32) (param $idx i32) (result i32)
    local.get $base
    local.get $idx
    i32.const 4
    i32.mul
    i32.add             ;; Adresse = base + idx * 4
    i32.load
  )

  (func $arr_set (param $base i32) (param $idx i32) (param $val i32)
    local.get $base
    local.get $idx
    i32.const 4
    i32.mul
    i32.add             ;; Adresse berechnen
    local.get $val
    i32.store           ;; Schreiben
  )

  ;; String-Länge berechnen (null-terminiert)
  (func $strlen (param $ptr i32) (result i32)
    (local $len i32)
    block $fertig
      loop $weiter
        local.get $ptr
        local.get $len
        i32.add
        i32.load8_u     ;; Ein Byte lesen
        i32.eqz         ;; Null-Terminator?
        br_if $fertig   ;; Ja → fertig

        local.get $len
        i32.const 1
        i32.add
        local.set $len  ;; len++

        br $weiter
      end
    end
    local.get $len
  )

  ;; memcpy – n Bytes von src nach dst kopieren
  (func $memcpy
    (param $dst i32) (param $src i32) (param $n i32)
    local.get $dst
    local.get $src
    local.get $n
    memory.copy         ;; Eingebaute Bulk-Memory Instruktion!
  )

  (export "set_int" (func $set_int))
  (export "get_int" (func $get_int))
  (export "strlen"  (func $strlen))
)

  ;; ── FIBONACCI (iterativ) ──────────────────────────────────────
  (func $fib (param $n i32) (result i32)
    (local $a i32)
    (local $b i32)
    (local $t i32)
    (local $i i32)

    i32.const 0
    local.set $a       ;; a = 0
    i32.const 1
    local.set $b       ;; b = 1

    local.get $n
    i32.eqz
    if (result i32)
      i32.const 0      ;; fib(0) = 0
    else
      block $fertig (result i32)
        loop $naechste
          local.get $b
          local.tee $t    ;; t = b (auch auf Stack für $fertig falls n==1)
          drop

          local.get $a
          local.get $b
          i32.add
          local.set $b    ;; b = a + b

          local.get $t
          local.set $a    ;; a = t (alter Wert von b)

          local.get $i
          i32.const 1
          i32.add
          local.tee $i    ;; i++

          local.get $n
          i32.const 1
          i32.sub
          i32.lt_s        ;; i < n-1?
          br_if $naechste
        end
        local.get $b      ;; Rückgabe
      end
    end
  )

  ;; ── ARRAY-SUMME ──────────────────────────────────────────────
  (func $array_sum (param $ptr i32) (param $len i32) (result i32)
    (local $i   i32)
    (local $sum i32)

    block $done
      loop $again
        local.get $i
        local.get $len
        i32.ge_u
        br_if $done             ;; i >= len: fertig

        local.get $sum
        local.get $ptr          ;; Basisadresse
        local.get $i
        i32.const 4
        i32.mul
        i32.add                 ;; Offset: i * 4
        i32.load                ;; arr[i]
        i32.add
        local.set $sum          ;; sum += arr[i]

        local.get $i
        i32.const 1
        i32.add
        local.set $i            ;; i++

        br $again
      end
    end
    local.get $sum
  )

  ;; ── MAXIMUM ──────────────────────────────────────────────────
  (func $array_max (param $ptr i32) (param $len i32) (result i32)
    (local $i   i32)
    (local $max i32)

    ;; max = arr[0]
    local.get $ptr
    i32.load
    local.set $max

    i32.const 1
    local.set $i

    block $done
      loop $again
        local.get $i
        local.get $len
        i32.ge_u
        br_if $done

        local.get $ptr
        local.get $i
        i32.const 4
        i32.mul
        i32.add
        i32.load                ;; arr[i]
        local.tee $i            ;; Zwischenspeichern... warte, $i ist schon belegt!
        drop
        ;; Korrekter Code:
        local.get $max
        ;; (verwende select für max)
        local.get $ptr
        local.get $i
        i32.const 4
        i32.mul
        i32.add
        i32.load                ;; arr[i]
        local.get $max
        local.get $ptr
        local.get $i
        i32.const 4
        i32.mul
        i32.add
        i32.load
        i32.gt_s                ;; arr[i] > max?
        select                  ;; arr[i] > max ? arr[i] : max
        local.set $max

        local.get $i
        i32.const 1
        i32.add
        local.set $i
        br $again
      end
    end
    local.get $max
  )

  (export "fib"        (func $fib))
  (export "array_sum"  (func $array_sum))
  (export "array_max"  (func $array_max))
  (export "mem"        (memory 0))
)

    local.get $a
    local.get $b
    f64.add
  )

  (func $wurzel (param $x f64) (result f64)
    local.get $x
    f64.sqrt
  )

  ;; Pythagorischer Abstand
  (func $pythagoras (param $a f64) (param $b f64) (result f64)
    local.get $a
    local.get $a
    f64.mul           ;; a²

    local.get $b
    local.get $b
    f64.mul           ;; b²

    f64.add           ;; a² + b²
    f64.sqrt          ;; sqrt(a² + b²)
  )

  ;; String in Speicher schreiben (für JS)
  (func $schreibe_string (param $ptr i32) (param $wert i32)
    ;; Konvertiere Zahl zu ASCII-Ziffern
    ;; Vereinfacht: schreibe 1-stellige Zahl
    local.get $ptr
    local.get $wert
    i32.const 48      ;; '0'
    i32.add           ;; ASCII-Zeichen
    i32.store8        ;; 1 Byte schreiben
  )

  (export "addiere"    (func $addiere))
  (export "wurzel"     (func $wurzel))
  (export "pythagoras" (func $pythagoras))

  (memory 1)

  (func $simd_add_f32x4
    (param $a_ptr i32)   ;; Zeiger auf 4 f32-Werte
    (param $b_ptr i32)   ;; Zeiger auf 4 f32-Werte
    (param $c_ptr i32)   ;; Ausgabe-Zeiger
    local.get $a_ptr
    v128.load           ;; Lädt 4 f32 auf einmal in v128-Register

    local.get $b_ptr
    v128.load

    f32x4.add           ;; Addiert alle 4 Paare gleichzeitig!

    local.get $c_ptr
    v128.store          ;; Schreibt 4 Ergebnisse auf einmal
  )

  ;; Dot Product (Skalarprodukt) von zwei 4D-Vektoren
  (func $dot_product_4d
    (param $a_ptr i32)
    (param $b_ptr i32)
    (result f32)
    (local $result v128)

    local.get $a_ptr
    v128.load

    local.get $b_ptr
    v128.load

    f32x4.mul           ;; a[i] * b[i] für alle 4

    ;; Summe der 4 Produkte

    local.tee $result
    drop

    local.get $result
    f32x4.extract_lane 1
    f32x4.add

  )
