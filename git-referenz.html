Staging Area – der Index 
# Die drei Bereiche in Git:
# Arbeitsverzeichnis  →  Stage (Index)  →  Repository

# Dateien zum Commit vormerken
git add datei.txt             # eine Datei
git add ordner/               # ganzes Verzeichnis
git add .                     # alles im aktuellen Verzeichnis
git add -A                    # alles (auch gelöschte)
git add -p                    # interaktiv: Hunks auswählen
git add -i                    # interaktiver Modus

# Status prüfen
git status
git diff --staged             # was ist gestaged?

# Staging rückgängig machen (Datei aus Stage entfernen)
git restore --staged datei.txt   # Modern (Git 2.23+)
git reset HEAD datei.txt         # Klassisch, gleiches Ergebnis

# Datei aus Arbeitsverzeichnis wiederherstellen (Änderungen verwerfen!)
git restore datei.txt             # Änderungen verwerfen – nicht rückgängig machbar!
git checkout -- datei.txt         # Klassisch

# .gitignore – Dateien ignorieren
echo "node_modules/" >> .gitignore
echo "*.log" >> .gitignore
echo ".env" >> .gitignore
git check-ignore -v datei.txt     # Warum wird Datei ignoriert?

# Bereits getrackte Datei jetzt ignorieren
git rm --cached datei.txt         # Aus Tracking entfernen
echo "datei.txt" >> .gitignore    # Dann ignorieren

# Gute Commit-Nachrichten (Conventional Commits):
# feat:     Neues Feature
# fix:      Bugfix
# docs:     Dokumentation
# style:    Formatierung (kein Code-Änderung)
# refactor: Refactoring ohne Feature/Fix
# test:     Tests hinzufügen/ändern
# chore:    Build, Config, Abhängigkeiten
# perf:     Performance-Verbesserung
# ci:       CI/CD-Änderungen
#
# Format: <typ>(<scope>): <kurze Beschreibung>
# Beispiel: feat(auth): OAuth2-Login implementiert

Mergen & Rebasen 
# ── MERGE ────────────────────────────────────────────────────
# Feature-Branch in main mergen
git switch main
git merge feature/login

# Merge immer als Merge-Commit (kein Fast-Forward)
git merge --no-ff feature/login

# Nur Fast-Forward erlauben (schlägt fehl wenn nicht möglich)
git merge --ff-only feature/login

# Merge abbrechen bei Konflikten
git merge --abort

# ── REBASE ───────────────────────────────────────────────────
# feature-Branch auf main rebasen
git switch feature/login
git rebase main

# Interaktiver Rebase – Commits umschreiben
git rebase -i HEAD~3            # Letzte 3 Commits bearbeiten
git rebase -i main              # Alle Commits seit main

# Im interaktiven Editor:
# pick   = Commit behalten
# reword = Commit-Nachricht ändern
# edit   = Commit bearbeiten
# squash = Mit vorherigem Commit zusammenführen
# fixup  = Wie squash aber Nachricht verwerfen
# drop   = Commit entfernen

# Rebase abbrechen
git rebase --abort

# Rebase nach Konflikt-Lösung fortsetzen
git add .
git rebase --continue

# Goldene Rebase-Regel:
# NIEMALS bereits gepushte Commits rebasen!
# (außer du weißt genau was du tust)

# ── CHERRY-PICK ───────────────────────────────────────────────
# Einzelnen Commit aus anderem Branch übernehmen
git cherry-pick abc1234

# Mehrere Commits
git cherry-pick abc1234 def5678

# Cherry-pick ohne Commit (nur Stage)
git cherry-pick -n abc1234
Merge-Konflikte lösen 
# Wenn git merge oder rebase einen Konflikt meldet:

# 1. Konflikt-Dateien anzeigen
git status   # zeigt "both modified: datei.txt"

# 2. Datei öffnen und Konflikt-Marker auflösen:
# <<<<<<< HEAD
# Deine Version (aktueller Branch)
# =======
# Eingehende Version (zu mergender Branch)
# >>>>>>> feature/login

# 3. Datei nach Auflösung stagen
git add datei.txt

# 4. Merge abschließen
git commit   # Merge-Commit Nachricht bestätigen

# Tools für Konflikt-Lösung
git mergetool               # Konfigurierbares Tool öffnen
git mergetool --tool=vimdiff

# Eigene Version behalten (für eine Datei)
git checkout --ours datei.txt
git add datei.txt

# Eingehende Version nehmen
git checkout --theirs datei.txt
git add datei.txt

Änderungen rückgängig machen 
# ── LOKAL – noch nicht gepusht ────────────────────────────────

# Einzelne Datei wiederherstellen (Änderungen im Arbeitsverz. verwerfen)
git restore datei.txt                  # Modern (Git 2.23+)
git checkout -- datei.txt              # Klassisch

# Datei aus Stage entfernen (unstagen)
git restore --staged datei.txt
git reset HEAD datei.txt               # Klassisch

# Letzten Commit rückgängig (Änderungen bleiben im Arbeitsverz.)
git reset HEAD~1                       # = git reset --mixed HEAD~1

# Letzten Commit rückgängig + Stage beibehalten
git reset --soft HEAD~1

# Letzten Commit rückgängig + alles verwerfen (DESTRUKTIV!)
git reset --hard HEAD~1

# ── GEPUSHT – bereits im Remote ──────────────────────────────

# Neuen Commit erstellen der den alten rückgängig macht (sicher!)
git revert HEAD                        # Letzten Commit revertieren
git revert abc1234                     # Bestimmten Commit
git revert HEAD~3..HEAD                # Mehrere Commits
git revert --no-commit HEAD~3..HEAD    # Ohne sofortigen Commit

# ── DATEI-VERSIONEN ───────────────────────────────────────────

# Datei aus einem bestimmten Commit wiederherstellen
git restore --source abc1234 datei.txt
git checkout abc1234 -- datei.txt      # Klassisch

# Datei aus einem anderen Branch holen
git restore --source feature/login -- src/auth.ts

# ── WAS WURDE GERADE GELÖSCHT? ───────────────────────────────

# Commit-History für eine Datei
git log --all --full-history -- datei.txt

# Reflog – ALLES was passiert ist (auch nach reset --hard!)
git reflog
git checkout HEAD@{3}                  # Zu früherem Zustand
git reset --hard HEAD@{5}              # Zustand wiederherstellen
reset vs revert vs restore 
# ── ÜBERSICHT: Die drei R's ───────────────────────────────────

# git restore – Dateien wiederherstellen (Git 2.23+)
git restore datei.txt              # Arbeitsverzeichnis auf HEAD setzen
git restore --staged datei.txt     # Stage auf HEAD setzen
git restore --source HEAD~2 datei  # Datei aus älterem Commit

# git reset – Commit-Pointer verschieben
git reset --soft HEAD~1   # Nur Pointer zurück, Stage + Dateien bleiben
git reset --mixed HEAD~1  # Pointer + Stage zurück, Dateien bleiben (Standard)
git reset --hard HEAD~1   # Alles zurück (DESTRUKTIV!)

# Merkhilfe:
# --soft  = commit rückgängig    → Stage hat Änderungen
# --mixed = commit + add rückgängig → Dateien haben Änderungen
# --hard  = alles rückgängig     → Nichts mehr übrig

# git revert – Sicheres Rückgängigmachen
git revert HEAD            # Erstellt neuen Commit der HEAD umkehrt
# Ändert die History NICHT – sicher für geteilte Branches

# Wann was nutzen?
# restore:  Datei-Änderungen verwerfen oder Datei aus Commit holen
# reset:    Lokale Commits entfernen (vor dem Push!)
# revert:   Commits die bereits gepusht wurden rückgängig machen
Stash – Änderungen zwischenspeichern 
# Änderungen temporär speichern (z.B. für Branch-Wechsel)
git stash                            # Alles wegspeichern
git stash push -m "WIP: Login-Form"  # Mit Beschreibung
git stash push -u                    # Auch ungetrackte Dateien
git stash push -p                    # Interaktiv (Hunks wählen)

# Stash-Liste
git stash list
# stash@{0}: On main: WIP: Login-Form
# stash@{1}: On feature: ...

# Stash anwenden
git stash pop                        # Anwenden + aus Liste entfernen
git stash apply                      # Anwenden, in Liste lassen
git stash apply stash@{2}            # Bestimmten Stash

# Stash inspizieren
git stash show stash@{0}             # Geänderte Dateien
git stash show -p stash@{0}          # Mit Diff

# Stash auf neuem Branch anwenden
git stash branch feature/neues stash@{0}

# Stash entfernen
git stash drop stash@{0}
git stash clear                      # Alle löschen

# Praktisches Muster:
git stash && git pull && git stash pop   # Pull + eigene Änderungen drüber

bisect – Fehler per Binärsuche finden 
# git bisect findet automatisch welcher Commit einen Bug eingeführt hat

# Suche starten
git bisect start

# Aktuellen Zustand als "schlecht" markieren (Bug vorhanden)
git bisect bad

# Bekannten guten Commit markieren (kein Bug)
git bisect good v1.0.0    # Tag, Hash oder HEAD~50

# Git checkt automatisch den mittleren Commit aus
# Testen: hat dieser Commit den Bug?

git bisect good   # Nein, dieser Commit ist ok
git bisect bad    # Ja, dieser Commit hat den Bug

# Git wiederholt bis der fehlerhafte Commit gefunden ist:
# "abc1234 is the first bad commit"

# Automatisiert mit Test-Skript
git bisect run ./test.sh   # 0 = good, 1-125 = bad

# Suche beenden
git bisect reset           # Zurück zum ursprünglichen HEAD
