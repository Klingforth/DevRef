/* Zeiger als Parameter – Änderungen wirken auf Original! */
void swap(int *a, int *b) {
    int tmp = *a;   /* dereferenzieren */
    *a = *b;
    *b = tmp;
}

int x = 5, y = 10;
swap(&x, &y);   /* x=10, y=5 */

/* Mehrere Rückgabewerte via Zeiger */
void min_max(const int *arr, int len, int *out_min, int *out_max) {
    *out_min = *out_max = arr[0];
    for (int i = 1; i < len; i++) {
        if (arr[i] < *out_min) *out_min = arr[i];
        if (arr[i] > *out_max) *out_max = arr[i];
    }
}

int zahlen[] = {3, 1, 4, 1, 5, 9, 2, 6};
int lo, hi;
min_max(zahlen, 8, &lo, &hi);
printf("min=%d max=%d\n", lo, hi);

/* Zeiger auf Zeiger – für dynamische Arrays als Parameter */
void vergroessern(int **arr, int *len) {
    int new_len = *len * 2;
    *arr = realloc(*arr, new_len * sizeof(int));
    *len = new_len;
}

/* ── 1. DANGLING POINTER ─────────────────────────────── */
int *schlecht(void) {
    int lokal = 42;
    return &lokal;   /* FEHLER: lokale Variable wird zerstört! */
}

/* Richtig: Heap oder static */
int *gut(void) {
    int *p = malloc(sizeof(int));
    *p = 42;
    return p;   /* Aufrufer muss free() aufrufen! */
}

/* ── 2. BUFFER OVERFLOW ──────────────────────────────── */
char buf[10];
strcpy(buf, "Dieser Text ist zu lang!");  /* FEHLER! */
/* Richtig: */
strncpy(buf, text, sizeof(buf) - 1);
buf[sizeof(buf) - 1] = '\0';
/* Oder: snprintf(buf, sizeof(buf), "%s", text); */

/* ── 3. USE AFTER FREE ────────────────────────────────── */
int *p = malloc(sizeof(int));
free(p);
*p = 42;   /* FEHLER: Undefined Behavior! */
/* Schutz: nach free immer NULL setzen */
free(p); p = NULL;
if (p) *p = 42;   /* sicher – macht nichts */

/* ── 4. DOPPELTES FREE ────────────────────────────────── */
free(p);
free(p);   /* FEHLER! */
/* Schutz: p = NULL nach free */

/* ── 5. NULL NICHT GEPRÜFT ───────────────────────────── */
int *p = malloc(sizeof(int));
/* Ohne Prüfung: */
*p = 42;   /* Crash wenn malloc fehlschlug! */
/* Richtig: immer prüfen */
if (!p) { perror("malloc"); exit(1); }

/* ── 6. OFF-BY-ONE ────────────────────────────────────── */
int arr[5];
for (int i = 0; i <= 5; i++) {   /* FEHLER: i <= 5 */
    arr[i] = i;                    /* arr[5] ist außerhalb! */
}

#include <stdlib.h>
#include <string.h>

typedef struct Knoten {
    int            wert;
    struct Knoten *naechster;
} Knoten;

typedef struct {
    Knoten *kopf;
    size_t  laenge;
} Liste;

/* Vorne einfügen */
void liste_push(Liste *l, int wert) {
    Knoten *neu = malloc(sizeof(Knoten));
    if (!neu) { perror("malloc"); exit(1); }
    neu->wert       = wert;
    neu->naechster  = l->kopf;
    l->kopf         = neu;
    l->laenge++;
}

/* Vorne entfernen */
int liste_pop(Liste *l) {
    if (!l->kopf) return 0;   /* Fehler: leere Liste */
    Knoten *alt = l->kopf;
    int wert    = alt->wert;
    l->kopf     = alt->naechster;
    free(alt);
    l->laenge--;
    return wert;
}

/* Ausgeben */
void liste_print(const Liste *l) {
    printf("[");
    for (Knoten *n = l->kopf; n; n = n->naechster) {
        printf("%d%s", n->wert, n->naechster ? " -> " : "");
    }
    printf("]\n");
}

/* Alle freigeben */
void liste_frei(Liste *l) {
    while (l->kopf) liste_pop(l);
}

int main(void) {
    Liste l = {NULL, 0};

    for (int i = 1; i <= 5; i++) liste_push(&l, i * 10);
    liste_print(&l);          /* [50 -> 40 -> 30 -> 20 -> 10] */
    printf("Länge: %zu\n", l.laenge);

    printf("Pop: %d\n", liste_pop(&l));   /* 50 */
    liste_print(&l);          /* [40 -> 30 -> 20 -> 10] */

    liste_frei(&l);
    return 0;
}

/* Generischer Vektor – speichert beliebige Typen */
typedef struct {
    void   *daten;        /* Roher Speicher */
    size_t  len;          /* Anzahl Elemente */
    size_t  cap;          /* Kapazität */
    size_t  element_size; /* Bytes pro Element */
} Vektor;

Vektor vektor_neu(size_t element_size) {
    return (Vektor){
        .daten        = NULL,
        .len          = 0,
        .cap          = 0,
        .element_size = element_size,
    };
}

void vektor_push(Vektor *v, const void *element) {
    if (v->len == v->cap) {
        v->cap  = v->cap == 0 ? 8 : v->cap * 2;
        v->daten = realloc(v->daten, v->cap * v->element_size);
        if (!v->daten) { perror("realloc"); exit(1); }
    }
    /* Byte-Arithmetik über char* */
    char *ziel = (char *)v->daten + v->len * v->element_size;
    memcpy(ziel, element, v->element_size);
    v->len++;
}

void *vektor_get(const Vektor *v, size_t idx) {
    if (idx >= v->len) return NULL;
    return (char *)v->daten + idx * v->element_size;
}

void vektor_frei(Vektor *v) {
    free(v->daten);
    *v = vektor_neu(v->element_size);
}

/* Makros für typsichere Nutzung */
#define VEC_PUSH(v, val)  do { typeof(val) _v = (val); vektor_push((v), &_v); } while(0)
#define VEC_GET(v, T, i)  (*(T *)vektor_get((v), (i)))

int main(void) {
    Vektor v = vektor_neu(sizeof(int));

    for (int i = 0; i < 10; i++) VEC_PUSH(&v, i * i);

    for (size_t i = 0; i < v.len; i++) {
        printf("%d ", VEC_GET(&v, int, i));
    }
    printf("\n");   /* 0 1 4 9 16 25 36 49 64 81 */

    vektor_frei(&v);
    return 0;
}
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define TABELLE_GROESSE 64

typedef struct Eintrag {
    char          *key;
    int            wert;
    struct Eintrag *naechster;   /* Für Kollisions-Chaining */
} Eintrag;

typedef struct {
    Eintrag *buckets[TABELLE_GROESSE];
} HashMap;

/* DJB2 Hash-Funktion */
static unsigned int hash(const char *key) {
    unsigned int h = 5381;
    while (*key) h = ((h << 5) + h) + (unsigned char)*key++;
    return h % TABELLE_GROESSE;
}

void map_set(HashMap *m, const char *key, int wert) {
    unsigned int idx = hash(key);
    Eintrag *e = m->buckets[idx];

    /* Vorhandenen Key aktualisieren */
    while (e) {
        if (strcmp(e->key, key) == 0) { e->wert = wert; return; }
        e = e->naechster;
    }

    /* Neuen Eintrag vorne einfügen */
    Eintrag *neu    = malloc(sizeof(Eintrag));
    neu->key        = strdup(key);
    neu->wert       = wert;
    neu->naechster  = m->buckets[idx];
    m->buckets[idx] = neu;
}

int map_get(const HashMap *m, const char *key, int *out) {
    unsigned int idx = hash(key);
    for (Eintrag *e = m->buckets[idx]; e; e = e->naechster) {
        if (strcmp(e->key, key) == 0) { *out = e->wert; return 1; }
    }
    return 0;   /* nicht gefunden */
}

void map_frei(HashMap *m) {
    for (int i = 0; i < TABELLE_GROESSE; i++) {
        Eintrag *e = m->buckets[i];
        while (e) {
            Eintrag *naechster = e->naechster;
            free(e->key);
            free(e);
            e = naechster;
        }
    }
}

int main(void) {
    HashMap m = {0};

    map_set(&m, "Alice",  30);
    map_set(&m, "Bob",    25);
    map_set(&m, "Carol",  35);
    map_set(&m, "Alice",  31);   /* Aktualisieren */

    int alter;
    if (map_get(&m, "Alice", &alter)) printf("Alice: %d\n", alter);   /* 31 */
    if (!map_get(&m, "Dave", &alter)) printf("Dave: nicht da\n");

    map_frei(&m);
    return 0;
}
/* Einfacher CSV-Parser mit Fehlerbehandlung */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>

#define MAX_FELDER  16
#define MAX_ZEILE   1024

typedef struct {
    char   felder[MAX_FELDER][256];
    int    anzahl;
} CSV_Zeile;

/* Zeile in Felder aufteilen */
int csv_parse_zeile(char *zeile, char trenner, CSV_Zeile *out) {
    out->anzahl = 0;
    char *ptr   = zeile;

    while (*ptr && out->anzahl < MAX_FELDER) {
        char *feld = out->felder[out->anzahl];
        int   len  = 0;

        /* Feld lesen bis Trenner oder Ende */
        while (*ptr && *ptr != trenner && *ptr != '\n' && *ptr != '\r') {
            if (len < 255) feld[len++] = *ptr;
            ptr++;
        }
        feld[len] = '\0';
        out->anzahl++;

        if (*ptr == trenner) ptr++;   /* Trenner überspringen */
    }
    return out->anzahl;
}

int main(int argc, char *argv[]) {
    if (argc < 2) {
        fprintf(stderr, "Verwendung: %s <csv-datei>\n", argv[0]);
        return EXIT_FAILURE;
    }

    FILE *f = fopen(argv[1], "r");
    if (!f) {
        fprintf(stderr, "Fehler: %s: %s\n", argv[1], strerror(errno));
        return EXIT_FAILURE;
    }

    char zeile[MAX_ZEILE];
    CSV_Zeile parsed;
    int zeilen_num = 0;

    /* Erste Zeile als Header */
    if (fgets(zeile, sizeof(zeile), f)) {
        csv_parse_zeile(zeile, ',', &parsed);
        printf("Spalten (%d): ", parsed.anzahl);
        for (int i = 0; i < parsed.anzahl; i++) {
            printf("[%s] ", parsed.felder[i]);
        }
        printf("\n\n");
    }

    /* Datenzeilen */
    while (fgets(zeile, sizeof(zeile), f)) {
        zeilen_num++;
        int felder = csv_parse_zeile(zeile, ',', &parsed);
        printf("Zeile %d: %d Felder\n", zeilen_num, felder);
        for (int i = 0; i < felder; i++) {
            printf("  [%d] = %s\n", i, parsed.felder[i]);
        }
    }

    printf("\nGesamt: %d Zeilen\n", zeilen_num);
    fclose(f);
    return EXIT_SUCCESS;
}
