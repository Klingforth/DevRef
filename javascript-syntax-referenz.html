function macheZaehler(start = 0) {
    let n = start;              // Upvalue – lebt in der Closure
    return {
        inc:   () => ++n,
        dec:   () => --n,
        wert:  () => n,
        reset: () => { n = start; },
    };
}

// Memoization
function memoize(fn) {
    const cache = new Map();
    return function(...args) {
        const key = JSON.stringify(args);
        if (cache.has(key)) return cache.get(key);
        const ergebnis = fn.apply(this, args);
        cache.set(key, ergebnis);
        return ergebnis;
    };
}

const fibMemo = memoize(function fib(n) {
    if (n < 2) return n;
    return fibMemo(n - 1) + fibMemo(n - 2);
});

// Unendlicher Generator
function* fibonacci() {
    let [a, b] = [0, 1];
    while (true) {
        yield a;
        [a, b] = [b, a + b];
    }
}

// yield* delegiert an anderen Iterator
function* flach(arr) {

class Tier {
    // Klassen-Feld (ES2022)
    #name;                        // privates Feld – nur innerhalb zugänglich
    static anzahl = 0;           // statisches Feld

    constructor(name, sound) {
        this.#name = name;
        this.sound = sound;
        Tier.anzahl++;
    }

    // Getter / Setter
    get name() { return this.#name; }
    set name(val) {
        if (typeof val !== "string") throw new TypeError("Name muss String sein");
        this.#name = val;
    }

    // Methode
    spreche() {
        return `${this.#name} sagt: ${this.sound}`;
    }

    // Statische Methode
    static erstelleAnonym() {
        return new Tier("Unbekannt", "...");
    }

    // toString
    toString() {
        return `Tier(${this.#name})`;
    }
}

// Vererbung
class Hund extends Tier {
    #tricks = [];                 // privates Feld mit Default

    constructor(name) {
        super(name, "Wuff");     // super() muss zuerst aufgerufen werden
    }

    lerne(trick) {
        this.#tricks.push(trick);
        return this;             // Method Chaining ermöglichen
    }

    get tricks() { return [...this.#tricks]; }   // Kopie zurückgeben

    // Methode überschreiben
    spreche() {
        const basis = super.spreche();   // Eltern-Methode aufrufen
        return `${basis} (${this.#tricks.length} Tricks gelernt)`;
    }
}

const rex = new Hund("Rex");
rex.lerne("Sitz").lerne("Platz");
console.log(rex.spreche());
console.log(rex instanceof Hund);  // true
console.log(rex instanceof Tier);  // true

class BankKonto {
    // Private Felder
    #kontostand = 0;
    #transaktionen = [];

    // Privates statisches Feld
    static #nextId = 1;

    constructor(inhaber, startGuthaben = 0) {
        this.inhaber = inhaber;
        this.id = BankKonto.#nextId++;
        this.#kontostand = startGuthaben;
    }

    einzahlen(betrag) {
        if (betrag <= 0) throw new RangeError("Betrag muss positiv sein");
        this.#kontostand += betrag;
        this.#transaktionen.push({ typ: "ein", betrag, datum: new Date() });
        return this;
    }

    abheben(betrag) {
        if (betrag > this.#kontostand) throw new Error("Nicht genug Guthaben");
        this.#kontostand -= betrag;
        this.#transaktionen.push({ typ: "ab", betrag, datum: new Date() });
        return this;
    }

    get kontostand() { return this.#kontostand; }

    // Private Methode
    #formatiereWaehrung(betrag) {
        return betrag.toLocaleString("de-DE", { style: "currency", currency: "EUR" });
    }

    kontoauszug() {
        return this.#transaktionen
            .map(t => `${t.typ === "ein" ? "+" : "-"} ${this.#formatiereWaehrung(t.betrag)}`)
            .join("\n");
    }
}

// Mixins – Mehrfach-Vererbung via Komposition
const Serializable = (Base) => class extends Base {
    toJSON() {
        return JSON.stringify(this);
    }
    static fromJSON(json) {
        return Object.assign(new this(), JSON.parse(json));
    }
};

const Validatable = (Base) => class extends Base {
    validate() {
        for (const [key, rules] of Object.entries(this.constructor.validations ?? {})) {
            for (const rule of rules) {
                if (!rule(this[key])) {
                    throw new Error(`Validierung fehlgeschlagen: ${key}`);
                }
            }
        }
        return true;
    }
};

// Mehrere Mixins anwenden
class User extends Serializable(Validatable(class {})) {
    static validations = {
        name:  [(v) => typeof v === "string" && v.length >= 2],
        email: [(v) => v?.includes("@")],
    };

    constructor(name, email) {
        super();
        this.name  = name;
        this.email = email;
    }
}

const alice = new User("Alice", "alice@example.com");
alice.validate();        // kein Fehler
alice.toJSON();          // {"name":"Alice","email":"alice@example.com"}

// Modernes fetch – gibt Promise<Response> zurück
async function apiRequest(url, optionen = {}) {
    const antwort = await fetch(url, {
        method:  "GET",
        headers: { "Content-Type": "application/json", ...optionen.headers },
        ...optionen,
    });

    // fetch wirft KEINEN Error bei 4xx/5xx – muss manuell geprüft werden!
    if (!antwort.ok) {
        throw new Error(`HTTP ${antwort.status}: ${antwort.statusText}`);
    }

    return antwort.json();
}

// POST mit Body
async function erstelleUser(daten) {
    return apiRequest("/api/users", {
        method: "POST",
        body:   JSON.stringify(daten),
    });
}

// Timeout via AbortController
async function mitTimeout(url, ms = 5000) {
    const controller = new AbortController();
    const timeoutId  = setTimeout(() => controller.abort(), ms);

    try {
        const antwort = await fetch(url, { signal: controller.signal });
        return await antwort.json();
    } catch (err) {
        if (err.name === "AbortError") throw new Error("Timeout!");
        throw err;
    } finally {
        clearTimeout(timeoutId);
    }
}

class MeineKlasse {
    [Symbol.iterator]() {                // for...of fähig machen
        let i = 0;
        const daten = [1, 2, 3];
        return { next: () => ({ value: daten[i++], done: i > daten.length }) };
    }

    [Symbol.toPrimitive](hint) {         // Typumwandlung kontrollieren
        if (hint === "number") return 42;
        return "MeineKlasse";
    }
}

// Robuster API-Client mit Retry und Timeout
class ApiClient {
    #baseUrl;
    #defaultHeaders;

    constructor(baseUrl, headers = {}) {
        this.#baseUrl = baseUrl;
        this.#defaultHeaders = {
            "Content-Type": "application/json",
            ...headers,
        };
    }

    async #request(method, pfad, optionen = {}) {
        const { body, headers = {}, timeout = 5000, versuche = 3 } = optionen;

        for (let versuch = 1; versuch <= versuche; versuch++) {
            const controller = new AbortController();
            const timer = setTimeout(() => controller.abort(), timeout);

            try {
                const antwort = await fetch(`${this.#baseUrl}${pfad}`, {
                    method,
                    headers: { ...this.#defaultHeaders, ...headers },
                    body:    body ? JSON.stringify(body) : undefined,
                    signal:  controller.signal,
                });

                if (!antwort.ok) throw new Error(`HTTP ${antwort.status}`);
                return await antwort.json();

            } catch (err) {
                if (versuch === versuche || err.name === "AbortError") throw err;
                await new Promise(r => setTimeout(r, 200 * versuch));  // Backoff
            } finally {
                clearTimeout(timer);
            }
        }
    }

    get(pfad, opts)         { return this.#request("GET",    pfad, opts); }
    post(pfad, body, opts)  { return this.#request("POST",   pfad, { ...opts, body }); }
    put(pfad, body, opts)   { return this.#request("PUT",    pfad, { ...opts, body }); }
    delete(pfad, opts)      { return this.#request("DELETE", pfad, opts); }
}

// Observer Pattern
class EventEmitter {
    #listeners = new Map();

    on(event, fn) {
        if (!this.#listeners.has(event)) this.#listeners.set(event, new Set());
        this.#listeners.get(event).add(fn);
        return () => this.off(event, fn);    // Rückgabe: Unsubscribe-Funktion
    }

    off(event, fn) {
        this.#listeners.get(event)?.delete(fn);
    }

    emit(event, ...args) {
        this.#listeners.get(event)?.forEach(fn => fn(...args));
    }

    once(event, fn) {
        const wrapper = (...args) => { fn(...args); this.off(event, wrapper); };
        return this.on(event, wrapper);
    }
}

class Store extends EventEmitter {
    #state;

    constructor(initialState) {
        super();
        this.#state = initialState;
    }

    get state() { return structuredClone(this.#state); }   // Kopie

    setState(updates) {
        const vorher = this.#state;
        this.#state  = { ...this.#state, ...updates };
        this.emit("change", this.#state, vorher);
    }
}

const pipe    = (...fns) => (x) => fns.reduce((v, f) => f(v), x);

// Curry
const curry = (fn) => {
    const arity = fn.length;
    return function curried(...args) {
        if (args.length >= arity) return fn(...args);
        return (...mehr) => curried(...args, ...mehr);
    };
};

// Lazy Pipeline – Daten werden erst bei Bedarf berechnet
function* map(iterable, fn) {
    for (const item of iterable) yield fn(item);
}

function* filter(iterable, pred) {
    for (const item of iterable) if (pred(item)) yield item;
}

function* take(iterable, n) {
    let count = 0;
    for (const item of iterable) {
        if (count++ >= n) return;
        yield item;
    }
}

function* naturliche() {
    let n = 1;
    while (true) yield n++;
}

// Erste 5 geraden Quadratzahlen – keine riesigen Arrays!

    ...take(
        filter(
            map(naturliche(), x => x * x),
            x => x % 2 === 0
        ),
        5
    )
];
console.log(ergebnis);   // [4, 16, 36, 64, 100]

// Als Klasse mit Methoden-Chaining
class LazySeq {
    constructor(iterable) { this._it = iterable; }
    map(fn)    { return new LazySeq(map(this._it, fn)); }
    filter(fn) { return new LazySeq(filter(this._it, fn)); }
    take(n)    { return [...take(this._it, n)]; }
}

const resultat = new LazySeq(naturliche())
    .map(x => x * x)
    .filter(x => x % 2 === 0)
    .take(5);
console.log(resultat);   // [4, 16, 36, 64, 100]
