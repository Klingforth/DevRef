Decorators
import functools, time

def timer(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        print(f"{func.__name__}: {time.perf_counter()-start:.4f}s")
        return result
    return wrapper

def retry(n=3, exc=(Exception,)):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for i in range(1, n+1):
                try: return func(*args, **kwargs)
                except exc as e:
                    if i == n: raise
        return wrapper
    return decorator

class Vektor:
    def __init__(self, x, y): self.x, self.y = x, y
    def __repr__(self): return f"V({self.x},{self.y})"
    def __add__(self, a):  return Vektor(self.x+a.x, self.y+a.y)
    def __mul__(self, s):  return Vektor(self.x*s, self.y*s)
    def __rmul__(self, s): return self.__mul__(s)
    def __neg__(self):     return Vektor(-self.x, -self.y)
    def __abs__(self):     return (self.x**2+self.y**2)**.5
    def __len__(self):     return 2
    def __getitem__(self,i): return (self.x, self.y)[i]
    def __iter__(self):    return iter((self.x, self.y))
    def __bool__(self):    return self.x!=0 or self.y!=0
    def __call__(self, s): return self * s
