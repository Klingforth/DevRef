TypeScript = JavaScript + statische Typen. Strukturelles Typsystem – kompatibel wenn die Struktur passt, nicht der Name.
Wichtigste tsconfig-Flags: "strict": true  ·  "noUncheckedIndexedAccess": true  ·  Vermeide any – nutze unknown

void function ohneReturn(): void {} Fuer Funktionen ohne Rueckgabewert.
object {} | [] | function(){} Alles ausser Primitiven. Selten direkt nutzen.

  switch (f.art) {            // art ist der Diskriminator

      return Math.PI * f.radius ** 2   // f ist Kreis
    case "rechteck":
      return f.breite * f.hoehe        // f ist Rechteck
  }
}

// Mapped Types – alle Properties transformieren
type Optional<T> = {
  [K in keyof T]?: T[K]           // wie Partial
}

type Nullable<T> = {

Klassen in TypeScript 
class Person {
  // Properties mit Sichtbarkeit (TS 2.0+)
  readonly id: number
  public   name: string
  protected email: string
  private  _passwort: string

  // Statische Property
  static anzahl = 0

  // Constructor Shorthand (sehr praktisch!)
  constructor(
    public readonly vorname: string,   // public + readonly automatisch!
    private nachname: string,
    public alter: number = 0          // Defaultwert
  ) {
    this.id = ++Person.anzahl
    this.name = `${vorname} ${nachname}`
    this.email = ""
    this._passwort = ""
  }

  // Getter / Setter
  get vollname(): string { return this.name }
  set vollname(v: string) {
    const teile = v.split(" ")
    this.vorname = teile[0]  // Fehler! readonly
  }

  // Statische Fabrik
  static erstelle(name: string): Person {
    const [v, n] = name.split(" ")
    return new Person(v ?? name, n ?? "")
  }

  // Methode
  vorstellen(): string {
    return `Ich bin ${this.vorname}, ${this.alter} Jahre alt.`
  }

  toString(): string {
    return `Person(${this.name})`
  }
}

console.log(alice.vorstellen())
console.log(Person.anzahl)     // 1

// Abstrakte Basisklasse
abstract class Form {
  abstract readonly art: string
  abstract flaeche(): number
  abstract umfang(): number

  // Konkrete Methode
  drucke(): void {
    console.log(`${this.art}: Fläche=${this.flaeche().toFixed(2)}`)
  }
}

class Kreis extends Form {
  readonly art = "Kreis"
  constructor(private radius: number) { super() }

  flaeche(): number { return Math.PI * this.radius ** 2 }
  umfang(): number  { return 2 * Math.PI * this.radius }
}

class Rechteck extends Form {
  readonly art = "Rechteck"
  constructor(private b: number, private h: number) { super() }

  flaeche(): number { return this.b * this.h }
  umfang(): number  { return 2 * (this.b + this.h) }
}

// Decorators transformieren Klassen/Methoden/Properties
// tsconfig: "experimentalDecorators": true  (legacy)
// Oder: Native Stage-3 Decorators (TS 5.0+)

// Klassen-Decorator
function singleton<T extends { new(...args: any[]): {} }>(Base: T) {
  let instanz: InstanceType<T> | null = null
  return class extends Base {
    constructor(...args: any[]) {
      super(...args)
      if (instanz) return instanz
      instanz = this as any
    }
  }
}

@singleton
class Config {
  wert = Math.random()
}

console.log(new Config().wert === new Config().wert)  // true!

// Methoden-Decorator
function logge(target: any, key: string, desc: PropertyDescriptor) {
  const original = desc.value
  desc.value = function(...args: any[]) {
    console.log(`${key} aufgerufen mit`, args)
    const ergebnis = original.apply(this, args)
    console.log(`${key} gibt zurück:`, ergebnis)
    return ergebnis
  }
}

class Rechner {
  @logge
  addiere(a: number, b: number): number {
    return a + b
  }
}

// Property Decorator
function pflicht(target: any, key: string) {
  let wert: any
  Object.defineProperty(target, key, {
    get: () => wert,
    set: (v) => {
      if (v === null || v === undefined)
        throw new Error(`${key} darf nicht null sein`)
      wert = v
    }
  })
}

// Parameter Decorator
function validiere(
  target: any, key: string, index: number
) {
  console.log(`Parameter ${index} von ${key} wird validiert`)
}

// async/await – sauberere Syntax
async function holeDaten(url: string): Promise<string> {
  const response = await fetch(url)
  if (!response.ok) {
    throw new Error(`HTTP ${response.status}`)
  }
  return response.text()
}

// Promise.all – parallel warten
async function ladeMehrere(urls: string[]): Promise<string[]> {
  return Promise.all(urls.map(url => holeDaten(url)))
}

// Promise.allSettled – alle, auch wenn manche scheitern
const ergebnisse = await Promise.allSettled([
  holeDaten("https://api1.example.com"),
  holeDaten("https://api2.example.com"),
])

for (const res of ergebnisse) {
  if (res.status === "fulfilled") console.log(res.value)
  else console.error(res.reason)
}

// Promise.race – erstes Ergebnis
const schnellste = await Promise.race([
  holeDaten("https://api1.example.com"),
  holeDaten("https://api2.example.com"),

// Fehlerbehandlung
async function sicher<T>(fn: () => Promise<T>): Promise<[T, null] | [null, Error]> {
  try {
    return [await fn(), null]
  } catch (err) {
    return [null, err instanceof Error ? err : new Error(String(err))]
  }
}

const [daten, fehler] = await sicher(() => holeDaten("https://api.example.com"))
if (fehler) { console.error(fehler.message); return }
console.log(daten)

// Async Generator
async function* seitenweise(url: string) {
  let seite = 1
  while (true) {
    const daten = await holeDaten(`${url}?page=${seite}`)
    const parsed = JSON.parse(daten)
    if (!parsed.items.length) break
    yield parsed.items
    seite++
  }
}

for await (const items of seitenweise("https://api.example.com/items")) {
  console.log("Seite:", items.length, "Items")
}

class QueryBuilder<T extends Record<string, unknown>> {
  private bedingungen: string[] = []
  private _limit?: number
  private _offset?: number

  where(feld: keyof T, wert: T[keyof T]): this {
    this.bedingungen.push(`${String(feld)} = '${wert}'`)
    return this  // Gibt this zurück fuer Chaining!
  }

  limit(n: number): this    { this._limit = n; return this }
  offset(n: number): this   { this._offset = n; return this }

  build(): string {
    let sql = "SELECT * FROM tabelle"
    if (this.bedingungen.length)
      sql += ` WHERE ${this.bedingungen.join(" AND ")}`
    if (this._limit)  sql += ` LIMIT ${this._limit}`
    if (this._offset) sql += ` OFFSET ${this._offset}`
    return sql
  }
}

  .where("name", "Alice")
  .where("alter", 30)
  .limit(10)
  .build()

// Vollstaendig typsicherer Event Emitter
type EventMap = Record<string, unknown>

class TypedEmitter<Events extends EventMap> {
  private listeners = new Map<
    keyof Events,
    Array<(data: Events[keyof Events]) => void>
  >()

  on<K extends keyof Events>(
    event: K,
    listener: (data: Events[K]) => void
  ): this {
    const existing = this.listeners.get(event) ?? []
    this.listeners.set(event, [...existing, listener as any])
    return this
  }

  off<K extends keyof Events>(
    event: K,
    listener: (data: Events[K]) => void
  ): this {
    const existing = this.listeners.get(event) ?? []
    this.listeners.set(event, existing.filter(l => l !== listener))
    return this
  }

  emit<K extends keyof Events>(event: K, data: Events[K]): this {
    const listeners = this.listeners.get(event) ?? []
    listeners.forEach(l => l(data as any))
    return this
  }

  once<K extends keyof Events>(
    event: K,
    listener: (data: Events[K]) => void
  ): this {
    const wrapper = (data: Events[K]) => {
      listener(data)
      this.off(event, wrapper)
    }
    return this.on(event, wrapper)
  }
}

interface ApiError {
  message: string
  code: number
  details?: unknown
}

type ApiResult<T> = Promise<
  | { ok: true;  data: T }
  | { ok: false; error: ApiError }
>

class ApiClient {
  constructor(
    private baseUrl: string,
    private defaultHeaders: Record<string, string> = {}
  ) {}

  private async request<T>(
    path: string,
    options: RequestInit = {}
  ): ApiResult<T> {
    try {
      const response = await fetch(`${this.baseUrl}${path}`, {
        ...options,
        headers: {
          "Content-Type": "application/json",
          ...this.defaultHeaders,
          ...options.headers,
        },
      })

      const data = await response.json()

      if (!response.ok) {
        return { ok: false, error: { message: data.message, code: response.status } }
      }

      return { ok: true, data: data as T }
    } catch (err) {
      return {
        ok: false,
        error: { message: err instanceof Error ? err.message : "Netzwerkfehler", code: 0 }
      }
    }
  }

  get<T>(path: string, headers?: Record<string, string>): ApiResult<T> {
    return this.request<T>(path, { method: "GET", headers })
  }

  post<T>(path: string, body: unknown): ApiResult<T> {
    return this.request<T>(path, {
      method: "POST",
      body: JSON.stringify(body),
    })
  }

  put<T>(path: string, body: unknown): ApiResult<T> {
    return this.request<T>(path, { method: "PUT", body: JSON.stringify(body) })
  }

  delete<T>(path: string): ApiResult<T> {
    return this.request<T>(path, { method: "DELETE" })
  }
}

type LoadingState = { status: "idle" }
type FetchingState = { status: "loading" }
type SuccessState<T> = { status: "success"; data: T; timestamp: Date }
type ErrorState = { status: "error"; message: string; retries: number }

type AsyncState<T> =
  | LoadingState
  | FetchingState
  | SuccessState<T>
  | ErrorState

// Aktionen
type Action<T> =
  | { type: "FETCH" }
  | { type: "SUCCESS"; data: T }
  | { type: "ERROR"; message: string }
  | { type: "RESET" }
  | { type: "RETRY" }

// Reducer – typsicher und exhaustiv
function reducer<T>(state: AsyncState<T>, action: Action<T>): AsyncState<T> {
  switch (action.type) {
    case "FETCH":
      return { status: "loading" }

    case "SUCCESS":
      return { status: "success", data: action.data, timestamp: new Date() }

    case "ERROR":
      return {
        status: "error",
        message: action.message,
        retries: state.status === "error" ? state.retries + 1 : 0
      }

    case "RESET":
      return { status: "idle" }

    case "RETRY":
      if (state.status !== "error") return state
      return { status: "loading" }

    default:
      const _exhaustive: never = action   // Compiler-Garantie
      return state
  }
}

// Render-Funktion mit Pattern Matching
function render<T>(state: AsyncState<T>): string {
  switch (state.status) {
    case "idle":    return "Bereit"
    case "loading": return "Laedt..."
    case "success": return `Daten geladen: ${JSON.stringify(state.data)}`
    case "error":   return `Fehler: ${state.message} (Versuche: ${state.retries})`
  }
}

interface UseQuery<T> {
  data:    T | undefined
  error:   Error | null
  loading: boolean

}

// Fetch Hook Implementation
function createUseQuery<T>(fetcher: () => Promise<T>): UseQuery<T> {
  let data:    T | undefined = undefined
  let error:   Error | null  = null
  let loading: boolean       = false
  const listeners = new Set<() => void>()

  async function refetch() {
    loading = true
    listeners.forEach(l => l())
    try {
      data    = await fetcher()
      error   = null
    } catch (e) {
      error   = e instanceof Error ? e : new Error(String(e))
    } finally {
      loading = false
      listeners.forEach(l => l())
    }
  }

  refetch()

  return {
    get data()    { return data },
    get error()   { return error },
    get loading() { return loading },
    refetch,
    subscribe(listener: () => void) {
      listeners.add(listener)
      return () => listeners.delete(listener)
    }
  } as UseQuery<T> & { subscribe: (l: () => void) => () => void }
}
