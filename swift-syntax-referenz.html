Structs & Classes – Value vs. Reference 
// ── STRUCT (Value Type) – bevorzugen! ────────────────────────────
// Kopiert beim Zuweisen/Übergeben (Copy-on-Write bei Collections)
struct Punkt {
    var x: Double
    var y: Double

    // Computed Property
    var länge: Double {
        (x*x + y*y).squareRoot()
    }

    // Mutating method (nötig bei structs!)
    mutating func verschiebe(dx: Double, dy: Double) {
        x += dx
        y += dy
    }

    // Static factory
    static let ursprung = Punkt(x: 0, y: 0)

    // Custom init (automatisch generiert wenn nicht definiert)
    init(x: Double = 0, y: Double = 0) {
        self.x = x
        self.y = y
    }
}

var p = Punkt(x: 3, y: 4)
print(p.länge)   // 5.0
p.verschiebe(dx: 1, dy: 0)

let p2 = p      // KOPIE! p2 unabhängig von p
// p2.x = 0    // Fehler wenn p2 als let!

// ── CLASS (Reference Type) – wenn nötig ───────────────────────────
// Teilt Referenz beim Zuweisen/Übergeben
class Tier {
    var name: String
    weak var besitzer: Person?   // weak: verhindert Retain Cycle!

    init(name: String) {
        self.name = name
    }

    // deinit: wird aufgerufen wenn Objekt freigegeben wird (ARC)
    deinit {
        print("\(name) wird freigegeben")
    }

    func laut() -> String { "..." }
}

class Hund: Tier {
    let rasse: String

    init(name: String, rasse: String) {
        self.rasse = rasse
        super.init(name: name)
    }

    override func laut() -> String { "Wuff!" }
}

let gleiches_rex = rex  // GLEICHE Referenz!
gleiches_rex.name = "Max"

Protocols – Protocol-Oriented Programming 
// Protocols in Swift = Interfaces + Default-Implementierungen
// POP (Protocol-Oriented Programming) ist Swifts Paradigma!

protocol Formbar {
    var fläche: Double { get }
    var umfang: Double { get }
    func beschreibe() -> String   // Keine Implementierung
}

// Default-Implementierung via Extension!
extension Formbar {
    func beschreibe() -> String {
        "\(type(of: self)): Fläche=\(String(format: "%.2f", fläche))"
    }
}

// Struct konformiert zu Protocol
struct Kreis: Formbar {
    let radius: Double
    var fläche: Double { Double.pi * radius * radius }
    var umfang: Double { 2 * Double.pi * radius }
}

struct Rechteck: Formbar {
    let breite, höhe: Double
    var fläche: Double { breite * höhe }
    var umfang: Double { 2 * (breite + höhe) }
}

// Protocol Composition – mehrere Protocols gleichzeitig
protocol Benennbar { var name: String { get } }

func verarbeite(form: Formbar & Benennbar) {
    print("\(form.name): \(form.fläche)")
}

// Associated Types (wie Generics in Protocols)
protocol Behälter {
    associatedtype Element
    mutating func hinzufügen(_ item: Element)
    var anzahl: Int { get }
    subscript(i: Int) -> Element { get }
}

// some (opaque type) – Compiler kennt genauen Typ
func macheForm() -> some Formbar {
    Kreis(radius: 5)
}

// any (existential) – Laufzeit-Dispatch
func zeige(form: any Formbar) {
    print(form.fläche)
}

Extensions & Generics 
// Extensions: Bestehende Typen OHNE Subtyp erweitern!

extension String {
    var istEmail: Bool {
        contains("@") && contains(".")
    }

    func wiederholt(_ n: Int) -> String {
        String(repeating: self, count: n)
    }

    var ersterBuchstabeGroß: String {
        guard let first = first else { return self }
        return first.uppercased() + dropFirst()
    }
}

"alice@example.com".istEmail  // true
"ab".wiederholt(3)            // "ababab"
"hallo".ersterBuchstabeGroß  // "Hallo"

// Extension auf eigene Typen
extension Kreis {
    static func einheit() -> Kreis {
        Kreis(radius: 1)
    }
}

// Extension mit Bedingung (Conditional Conformance)
extension Array: Formbar where Element == Kreis {
    var fläche: Double { map(\.fläche).reduce(0, +) }
    var umfang: Double { map(\.umfang).reduce(0, +) }
}

// ── GENERICS ─────────────────────────────────────────────────────
func tausche<T>(_ a: inout T, _ b: inout T) {
    let temp = a; a = b; b = temp
}

func maximum<T: Comparable>(_ a: T, _ b: T) -> T {
    a > b ? a : b
}

// Generische Struktur
struct Stack<Element> {
    private var items: [Element] = []
    mutating func push(_ item: Element) { items.append(item) }
    mutating func pop() -> Element? { items.popLast() }
    var top: Element? { items.last }
    var isEmpty: Bool { items.isEmpty }
}

var stack = Stack<Int>()
stack.push(1); stack.push(2); stack.push(3)
print(stack.pop()!)  // 3

// Where-Klausel (Constraints)
func enthält<T: Collection>(_ c: T, _ el: T.Element) -> Bool
    where T.Element: Equatable {
    c.contains(el)
}

// Primary Associated Types (Swift 5.7+)
func summe(_ seq: some Sequence<Int>) -> Int {
    seq.reduce(0, +)
}

// ── CLOSURES ─────────────────────────────────────────────────────
// Vollständige Schreibweise
let multipliziere = { (a: Int, b: Int) -> Int in
    a * b
}

// Kurzschreibweisen (Swift macht vieles inferierbar)

[1, 2, 3].map { $0 * $0 }           // [1, 4, 9]
[1, 2, 3].reduce(0) { $0 + $1 }     // 6
[1, 2, 3].reduce(0, +)              // 6 (Operator als Closure!)

// Optionals – kein nil ohne ? im Typ!
var name: String? = nil
name = "Alice"

// ── AUSPACKEN ────────────────────────────────────────────────────
// 1. Optional Binding (if let / guard let)
if let n = name {
    print("Name: \(n)")   // n ist hier String, nicht String?
}

// Neuer Name nicht nötig (Shadowing, Swift 5.7+)
if let name {
    print("Name: \(name)")
}

// 2. guard let – früher Abbruch
func verarbeite(_ name: String?) {
    guard let name else { return }   // name ist jetzt String
    print(name.uppercased())
}

// 3. Nil-Coalescing ??
let anzeige = name ?? "Anonym"

// 4. Optional Chaining ?.
let länge = name?.count   // nil wenn name nil, sonst Int

// 5. Forced Unwrap ! (Vorsicht! Crash wenn nil)
let sicher = name!   // nur wenn SICHER nicht nil

// ── PATTERN MATCHING ─────────────────────────────────────────────
// switch ist SEHR mächtig in Swift!
let wert: Any = 42

switch wert {
case let n as Int where n > 0:
    print("Positive Int: \(n)")
case let s as String:
    print("String: \(s)")
case is Double:
    print("Eine Double")
default:
    print("Unbekannt")
}

case 0..<13:    print("Kind")

}

Fehlerbehandlung – do/try/catch 
// Swift Fehlerbehandlung ist typsicher und explizit

enum ValidationError: Error {
    case leer
    case zuKurz(mindestens: Int)
    case ungültigesFormat
    case ungültigesAlter(alter: Int)
}

// throws – Funktion kann werfen
func validiereAlter(_ s: String) throws -> Int {
    guard !s.isEmpty else { throw ValidationError.leer }

    guard let n = Int(s) else { throw ValidationError.ungültigesFormat }

    guard n >= 0 && n <= 150 else {
        throw ValidationError.ungültigesAlter(alter: n)
    }

    return n
}

// Aufrufen mit do/try/catch
do {
    let alter = try validiereAlter("250")
    print("Alter: \(alter)")
} catch ValidationError.leer {
    print("Eingabe war leer")
} catch ValidationError.ungültigesAlter(let n) {
    print("Ungültiges Alter: \(n)")  // Associated Value!
} catch {
    print("Fehler: \(error)")       // error = alle anderen
}

// try? – nil wenn Fehler (kein throw nach oben)
let alter = try? validiereAlter("30")  // Optional<Int>

// try! – Crash wenn Fehler (nur wenn sicher!)
let alter2 = try! validiereAlter("25")  // Int (kein Optional)

}

Property Observers & Computed Properties 
struct Temperatur {
    // Stored Property mit Observers
    var celsius: Double {
        willSet { print("Wird gesetzt: \(newValue)") }
        didSet  { print("War: \(oldValue), Neu: \(celsius)") }
    }

    // Computed Properties (kein Speicher!)
    var fahrenheit: Double {
        get { celsius * 9/5 + 32 }
        set { celsius = (newValue - 32) * 5/9 }
    }

    var kelvin: Double {
        get { celsius + 273.15 }
        set { celsius = newValue - 273.15 }
    }

    // Read-only Computed Property (nur get)
    var beschreibung: String {
        celsius < 0 ? "Gefroren" : celsius < 20 ? "Kalt" : "Warm"
    }

    init(celsius: Double = 0) {
        self.celsius = celsius
    }
}

var t = Temperatur(celsius: 100)
print(t.fahrenheit)    // 212.0
t.kelvin = 0           // Setzt celsius auf -273.15

// ── PROPERTY WRAPPERS ─────────────────────────────────────────────
@propertyWrapper
struct Begrenzt<T: Comparable> {
    private var wert: T
    let min: T, max: T

    var wrappedValue: T {
        get { wert }
        set { wert = Swift.min(max, Swift.max(min, newValue)) }
    }

    init(wrappedValue: T, min: T, max: T) {
        self.min = min; self.max = max
        self.wert = Swift.min(max, Swift.max(min, wrappedValue))
    }
}

struct Einstellungen {
    @Begrenzt(min: 0, max: 100) var lautstärke: Int = 50
    @Begrenzt(min: 1, max: 10)  var zoom: Double = 1.0
}

var e = Einstellungen()
e.lautstärke = 150   // Automatisch begrenzt auf 100
print(e.lautstärke)  // 100

import Foundation

// ── ASYNC/AWAIT ───────────────────────────────────────────────────
// async markiert eine Funktion die await verwenden kann
func holeDaten(von url: URL) async throws -> Data {
    let (data, _) = try await URLSession.shared.data(from: url)
    return data
}

// Aufruf
func ladeAlles() async throws {
    let url = URL(string: "https://api.example.com")!
    let daten = try await holeDaten(von: url)
    print("Erhalten: \(daten.count) Bytes")
}

// ── STRUCTURED CONCURRENCY ───────────────────────────────────────
func parallel() async throws {
    // async let – startet sofort, await holt Ergebnis
    async let bild   = ladeBild(url: bildURL)
    async let profil = ladeProfil(id: 42)
    async let stats  = ladeStatistik()

    // Alle drei laufen PARALLEL!
    let (b, p, s) = try await (bild, profil, stats)
    print(b, p, s)
}

// TaskGroup – dynamische Anzahl paralleler Tasks
func ladeAlleNutzer() async throws -> [User] {
    let ids = [1, 2, 3, 4, 5]

    return try await withThrowingTaskGroup(of: User.self) { group in
        for id in ids {
            group.addTask { try await ladeNutzer(id: id) }
        }

        var users: [User] = []
        for try await user in group {
            users.append(user)
        }
        return users
    }
}

// Task { } – nicht-async Kontext → async
Task {
    let result = try await ladeAlles()
}

// Task.sleep – Pause ohne Thread zu blockieren!
try await Task.sleep(for: .seconds(1))

// ── CANCELLATION ─────────────────────────────────────────────────
func abbruchbar() async throws {
    for i in 0..<100 {
        try Task.checkCancellation()   // Wirft wenn abgebrochen
        await verarbeite(i)
    }
}

let task = Task { try await abbruchbar() }
task.cancel()   // Abbrechbar
Actors – sichere Concurrency 
// Actors schützen Mutablen State vor Data Races!
// Swift 6: Strikte Concurrency-Prüfung

actor BankKonto {
    private var kontostand: Double

    init(anfangsBetrag: Double) {
        self.kontostand = anfangsBetrag
    }

    // Alle Methoden im Actor sind automatisch isoliert!
    func einzahlen(_ betrag: Double) {
        kontostand += betrag
    }

    func abheben(_ betrag: Double) throws {
        guard kontostand >= betrag else {
            throw KontoError.nichtGenugGeld
        }
        kontostand -= betrag
    }

    // nonisolated – kann ohne await aufgerufen werden
    nonisolated var beschreibung: String {
        "BankKonto"
    }

    var aktuell: Double { kontostand }
}

// Verwendung: IMMER await für Actor-Methoden!
let konto = BankKonto(anfangsBetrag: 1000)
await konto.einzahlen(500)
let stand = await konto.aktuell   // 1500

// Generic API Client
class APIClient {
    let baseURL: URL
    let session = URLSession.shared
    let decoder = JSONDecoder()

    init(baseURL: URL) {
        self.baseURL = baseURL
    }

    func lade<T: Decodable>(_ pfad: String) async throws -> T {
        let url = baseURL.appendingPathComponent(pfad)
        let (data, response) = try await session.data(from: url)

        guard let http = response as? HTTPURLResponse,
              (200...299).contains(http.statusCode) else {
            throw URLError(.badServerResponse)
        }

        return try decoder.decode(T.self, from: data)
    }
}

struct HTMLBuilder {

    }

    }

    }

    }

func tag(_ name: String, attrs: String = "",
         @HTMLBuilder content: () -> String) -> String {
    let a = attrs.isEmpty ? "" : " \(attrs)"
    return "<\(name)\(a)>\n\(content())\n</\(name)>"
}

    "<!DOCTYPE html>
" + tag("html") { content() }
}

// Einkaufsliste mit Actor (Thread-safe)
actor Einkaufsliste {
    private var artikel: [String: Int] = [:]   // Artikel: Menge

    func hinzufügen(_ artikel: String, menge: Int = 1) {
        self.artikel[artikel, default: 0] += menge
    }

    func entfernen(_ artikel: String) {
        self.artikel.removeValue(forKey: artikel)
    }

    func menge(von artikel: String) -> Int {
        self.artikel[artikel] ?? 0
    }

    var alle: [(artikel: String, menge: Int)] {
        self.artikel.map { ($0.key, $0.value) }
            .sorted { $0.artikel < $1.artikel }
    }

    var gesamtArtikel: Int {
        self.artikel.values.reduce(0, +)
    }
}

// Parallel von mehreren Tasks befüllen (sicher durch Actor!)
func befülle(_ liste: Einkaufsliste) async {
    await withTaskGroup(of: Void.self) { gruppe in
        let einkäufe = [
            ("Äpfel", 5), ("Bananen", 3), ("Milch", 2),
            ("Brot", 1), ("Äpfel", 3),   // Äpfel nochmal!
        ]

        for (artikel, menge) in einkäufe {
            gruppe.addTask {
                await liste.hinzufügen(artikel, menge: menge)
            }
        }
    }
}

// AsyncSequence – eigene async Sequenz
struct ZählerSequenz: AsyncSequence {
    typealias Element = Int
    let bis: Int

    struct Iterator: AsyncIteratorProtocol {
        var aktuell: Int
        let bis: Int

        mutating func next() async -> Int? {
            guard aktuell <= bis else { return nil }
            // Simuliertes async Delay
            try? await Task.sleep(for: .milliseconds(10))
            defer { aktuell += 1 }
            return aktuell
        }
    }

    func makeAsyncIterator() -> Iterator {
        Iterator(aktuell: 1, bis: bis)
    }
}

    print("  \(artikel.padding(toLength: 12, withPad: " ", startingAt: 0)): \(menge)x")
}

// ── PROTOCOL MIT ASSOCIATED TYPE ─────────────────────────────────
protocol Repository {
    associatedtype Entität: Identifiable
    func lade(id: Entität.ID) async throws -> Entität?
    func speichere(_ entität: Entität) async throws
    func alle() async throws -> [Entität]
}

struct Nutzer: Identifiable {
    let id: UUID
    var name: String
}

actor NutzerRepository: Repository {
    private var store: [UUID: Nutzer] = [:]

    func lade(id: UUID) async -> Nutzer? { store[id] }

    func speichere(_ nutzer: Nutzer) async {
        store[nutzer.id] = nutzer
    }

    func alle() async -> [Nutzer] {
        Array(store.values).sorted { $0.name < $1.name }
    }
}
