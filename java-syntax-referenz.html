        case String  s            -> "String: " + s;
        case null                 -> "null";
        default                   -> "Sonstiges: " + obj.getClass().getName();
    };
}

Klassen & Objekte 
// Klassendefinition
public class Person {
    // Felder (Instanz-Variablen)
    private final String name;
    private int alter;
    private static int anzahl = 0;   // Klassenvariable

    // Konstruktor
    public Person(String name, int alter) {
        if (name == null || name.isBlank())
            throw new IllegalArgumentException("Name darf nicht leer sein");
        this.name  = name;
        this.alter = alter;
        anzahl++;
    }

    // Getter (kein Setter für final name – immutablel!)
    public String getName()  { return name; }
    public int    getAlter() { return alter; }

    // Setter mit Validierung
    public void setAlter(int alter) {
        if (alter < 0 || alter > 150)
            throw new IllegalArgumentException("Ungültiges Alter: " + alter);
        this.alter = alter;
    }

    // Klassenmethode
    public static int getAnzahl() { return anzahl; }

    // toString
    @Override
    public String toString() {
        return "Person{name='%s', alter=%d}".formatted(name, alter);
    }

    // equals und hashCode (immer beide implementieren!)
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (!(o instanceof Person p)) return false;
        return alter == p.alter && name.equals(p.name);
    }

    @Override
    public int hashCode() {
        return Objects.hash(name, alter);
    }
}

// Verwendung
var alice = new Person("Alice", 30);
var bob   = new Person("Bob", 25);
System.out.println(alice);             // Person{name='Alice', alter=30}
System.out.println(alice.equals(bob)); // false
System.out.println(Person.getAnzahl()); // 2

// Sealed Classes (Java 17+) – eingeschränkte Vererbungshierarchie
public sealed interface Form permits Kreis, Rechteck, Dreieck {}

public record Kreis(double radius)            implements Form {}
public record Rechteck(double b, double h)    implements Form {}
public record Dreieck(double a, double b, double c) implements Form {}

// Exhaustiver switch mit Sealed Classes
double flaeche(Form f) {
    return switch (f) {
        case Kreis k      -> Math.PI * k.radius() * k.radius();
        case Rechteck r   -> r.b() * r.h();
        case Dreieck t -> {
            double s = (t.a() + t.b() + t.c()) / 2;
            yield Math.sqrt(s*(s-t.a())*(s-t.b())*(s-t.c()));
        }
        // Kein default nötig – Compiler prüft Vollständigkeit!
    };
}
Generics 
// Generische Klasse
public class Stack<T> {
    private final List<T> items = new ArrayList<>();

    public void push(T item) { items.add(item); }

    public T pop() {
        if (items.isEmpty())
            throw new EmptyStackException();
        return items.remove(items.size() - 1);
    }

    public T peek() { return items.get(items.size() - 1); }
    public boolean isEmpty() { return items.isEmpty(); }
    public int size() { return items.size(); }
}

// Verwendung
var intStack = new Stack<Integer>();
intStack.push(1); intStack.push(2); intStack.push(3);
System.out.println(intStack.pop());   // 3

// Generische Methode
public static <T extends Comparable<T>> T maximum(T a, T b) {
    return a.compareTo(b) >= 0 ? a : b;
}

maximum(3, 7)          // 7
maximum("Alice", "Bob") // "Bob"

// Wildcards
// ? extends T – Obergrenze (nur lesen)
double summe(List<? extends Number> zahlen) {
    return zahlen.stream().mapToDouble(Number::doubleValue).sum();
}

// ? super T – Untergrenze (nur schreiben)
void addIntegers(List<? super Integer> liste) {
    liste.add(1); liste.add(2); liste.add(3);
}

// PECS: Producer Extends, Consumer Super
public static <T> void kopiere(List<? extends T> quelle,
                                List<? super T> ziel) {
    for (T item : quelle) ziel.add(item);
}

// Stream = Datenpipeline, kein Speicher, lazy
// Operationen: intermediate (lazy) + terminal (eager)

List<String> namen = List.of("Alice", "Bob", "Carol", "Dave", "Eve");

// ── Intermediate Operations (lazy) ────────────────────────────────
// filter – behalten wenn Predicate true
// map    – transformieren
// flatMap – flach machen
// distinct – Duplikate entfernen
// sorted  – sortieren
// limit / skip – Anzahl begrenzen / überspringen
// peek    – Debugging ohne Veränderung

// ── Terminal Operations (starten die Verarbeitung) ────────────────
// forEach, forEachOrdered
// collect(Collectors.xxx)
// count, sum, average, min, max
// findFirst, findAny
// anyMatch, allMatch, noneMatch
// reduce
// toArray

// Beispiel-Pipeline
long anzahl = namen.stream()
    .filter(n -> n.length() > 3)          // ["Alice","Carol","Dave"]
    .map(String::toLowerCase)             // ["alice","carol","dave"]
    .sorted()                             // ["alice","carol","dave"]
    .count();                             // 3

    .filter(n -> n.startsWith("A"))

// switch Pattern Matching (Java 21)
String beschreibe(Object obj) {
    return switch (obj) {
        case Integer i when i < 0   -> "Negative Zahl: " + i;
        case Integer i              -> "Positive Zahl: " + i;
        case Double  d              -> "Float: " + d;
        case String  s when s.isBlank() -> "Leerer String";
        case String  s              -> "String: '" + s + "'";
        case int[]   arr            -> "Int-Array mit " + arr.length + " Elementen";
        case null                   -> "null";
        default                     -> "Sonstiges: " + obj.getClass().getSimpleName();
    };
}

        // Kein default nötig – Compiler prüft Vollständigkeit!

// Records mit @Override und Interfaces
public record Range(int start, int end) implements Comparable<Range> {
    Range {
        if (start > end) throw new IllegalArgumentException();
    }

    public int length()          { return end - start; }
    public boolean contains(int n) { return n >= start && n < end; }

    @Override
    public int compareTo(Range other) {
        return Integer.compare(this.start, other.start);
    }
}

// Immutabler Builder mit Generics
public final class Query<T> {
    private final Class<T>    typ;
    private final List<String> felder;
    private final String       bedingung;
    private final int          limit;
    private final int          offset;

    private Query(Builder<T> b) {
        this.typ       = Objects.requireNonNull(b.typ);
        this.felder    = List.copyOf(b.felder);
        this.bedingung = b.bedingung;
        this.limit     = b.limit;
        this.offset    = b.offset;
    }

    public static <T> Builder<T> von(Class<T> typ) {
        return new Builder<>(typ);
    }

    public static final class Builder<T> {
        private final Class<T> typ;
        private final List<String> felder = new ArrayList<>();
        private String bedingung;
        private int    limit  = Integer.MAX_VALUE;
        private int    offset = 0;

        private Builder(Class<T> typ) { this.typ = typ; }

        public Builder<T> feld(String name) {
            felder.add(name); return this;
        }
        public Builder<T> wo(String bedingung) {
            this.bedingung = bedingung; return this;
        }
        public Builder<T> limit(int limit) {
            this.limit = limit; return this;
        }
        public Builder<T> offset(int offset) {
            this.offset = offset; return this;
        }
        public Query<T> build() { return new Query<>(this); }
    }

    @Override
    public String toString() {
        var sb = new StringBuilder("SELECT ");
        sb.append(felder.isEmpty() ? "*" : String.join(", ", felder));
        sb.append(" FROM ").append(typ.getSimpleName());
        Optional.ofNullable(bedingung)
            .ifPresent(b -> sb.append(" WHERE ").append(b));
        if (limit != Integer.MAX_VALUE) sb.append(" LIMIT ").append(limit);
        if (offset > 0) sb.append(" OFFSET ").append(offset);
        return sb.toString();
    }
}

public class BestellungsAnalyse {
    public static void main(String[] args) {
        var bestellungen = List.of(
            new Bestellung(1, "Alice", "Laptop",   999.99, true),
            new Bestellung(2, "Bob",   "Maus",      29.99, true),
            new Bestellung(3, "Alice", "Tastatur",  59.99, false),
            new Bestellung(4, "Carol", "Monitor",  399.00, true),
            new Bestellung(5, "Bob",   "Kabel",      9.99, true),
            new Bestellung(6, "Alice", "Laptop",   999.99, true)
        );

        // Gesamtumsatz bezahlter Bestellungen
        double umsatz = bestellungen.stream()
            .filter(Bestellung::bezahlt)
            .mapToDouble(Bestellung::betrag)
            .sum();
        System.out.printf("Umsatz: %.2f EUR%n", umsatz);

        // Pro Kunde: Gesamtumsatz
        var proKunde = bestellungen.stream()
            .filter(Bestellung::bezahlt)
            .collect(Collectors.groupingBy(
                Bestellung::kunde,
                Collectors.summingDouble(Bestellung::betrag)
            ));
        proKunde.entrySet().stream()
            .sorted(Map.Entry.<String, Double>comparingByValue().reversed())
            .forEach(e -> System.out.printf("  %-10s %.2f EUR%n",
                e.getKey(), e.getValue()));

        // Top-3-Produkte nach Umsatz
        System.out.println("Top-Produkte:");
        bestellungen.stream()
            .collect(Collectors.groupingBy(
                Bestellung::produkt,
                Collectors.summingDouble(Bestellung::betrag)
            ))
            .entrySet().stream()
            .sorted(Map.Entry.<String, Double>comparingByValue().reversed())
            .limit(3)
            .forEach(e -> System.out.printf("  %-12s %.2f EUR%n",
                e.getKey(), e.getValue()));

        // Unbezahlte Bestellungen pro Kunde
        var unbezahlt = bestellungen.stream()
            .filter(b -> !b.bezahlt())
            .collect(Collectors.groupingBy(Bestellung::kunde,
                Collectors.mapping(Bestellung::produkt, Collectors.toList())));
        System.out.println("Unbezahlt: " + unbezahlt);
    }

// Algebraische Datentypen Java-Stil (Java 21)

sealed interface Ausdruck permits Zahl, Addition, Multiplikation, Negation {}

record Zahl(double wert)                       implements Ausdruck {}
record Addition(Ausdruck links, Ausdruck rechts) implements Ausdruck {}
record Multiplikation(Ausdruck links, Ausdruck rechts) implements Ausdruck {}
record Negation(Ausdruck ausdruck)             implements Ausdruck {}

public class Rechner {
    // Ausdrucks-Baum auswerten – exhaustiver switch!
    static double auswerten(Ausdruck expr) {
        return switch (expr) {
            case Zahl(double w)           -> w;
            case Addition(var l, var r)   -> auswerten(l) + auswerten(r);
            case Multiplikation(var l, var r) -> auswerten(l) * auswerten(r);
            case Negation(var e)          -> -auswerten(e);
        };
    }

    // Ausdrucks-Baum als String
    static String toString(Ausdruck expr) {
        return switch (expr) {
            case Zahl(double w)              -> String.valueOf(w);
            case Addition(var l, var r)      ->
                "(" + toString(l) + " + " + toString(r) + ")";
            case Multiplikation(var l, var r) ->
                "(" + toString(l) + " * " + toString(r) + ")";
            case Negation(var e)             ->
                "(-" + toString(e) + ")";
        };
    }

    public static void main(String[] args) {
        // (3 + 4) * (-2)
        Ausdruck ausdruck = new Multiplikation(
            new Addition(new Zahl(3), new Zahl(4)),
            new Negation(new Zahl(2))
        );

        System.out.println(toString(ausdruck));  // ((3.0 + 4.0) * (-2.0))
        System.out.println(auswerten(ausdruck));  // -14.0

        // 1 + 2 + 3 + 4 + 5
        Ausdruck summe = new Zahl(0);
        for (int i = 1; i <= 5; i++) {
            summe = new Addition(summe, new Zahl(i));
        }
        System.out.println(auswerten(summe));   // 15.0
    }
}

// Einfacher HTTP-Server nur mit JDK (kein Spring, kein Micronaut)
public class SimpleServer {

    record User(int id, String name, String email) {}

    // Simulierte Datenbank
    static final Map<Integer, User> users = Map.of(
        1, new User(1, "Alice", "alice@example.com"),
        2, new User(2, "Bob",   "bob@example.com")
    );

    static void handleUsers(HttpExchange exchange) throws IOException {
        String method = exchange.getRequestMethod();
        String path   = exchange.getRequestURI().getPath();

        switch (method) {
            case "GET" -> {
                // Alle User zurückgeben
                var json = users.values().stream()
                    .map(u -> """
                        {"id":%d,"name":"%s","email":"%s"}
                        """.formatted(u.id(), u.name(), u.email()).strip())
                    .collect(java.util.stream.Collectors.joining(",", "[", "]"));

                byte[] response = json.getBytes();
                exchange.getResponseHeaders().add("Content-Type", "application/json");
                exchange.sendResponseHeaders(200, response.length);
                try (var os = exchange.getResponseBody()) {
                    os.write(response);
                }
            }
            default -> {
                exchange.sendResponseHeaders(405, 0);
                exchange.close();
            }
        }
    }

    public static void main(String[] args) throws IOException {
        var server = HttpServer.create(new InetSocketAddress(8080), 0);

        server.createContext("/api/users", SimpleServer::handleUsers);
        server.createContext("/health",    exchange -> {
            byte[] ok = "OK".getBytes();
            exchange.sendResponseHeaders(200, ok.length);
            exchange.getResponseBody().write(ok);
            exchange.close();
        });

        // Java 21: Virtual Threads für massiv skalierbare Server!
        server.setExecutor(Executors.newVirtualThreadPerTaskExecutor());
        server.start();

        System.out.println("Server läuft auf http://localhost:8080");
        System.out.println("GET http://localhost:8080/api/users");
    }
}
