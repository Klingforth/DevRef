Das Prolog-Paradigma – komplett anders denken 
% Prolog = LOGISCHE Programmierung
% Du beschreibst FAKTEN und REGELN.
% Prolog findet durch BACKTRACKING die Antworten.
%
% In anderen Sprachen: "Berechne X aus A und B"
% In Prolog:           "Was ist X, wenn A und B wahr sind?"
%
% Das Programm in SWI-Prolog starten:
%   swipl
%   swipl datei.pl
%   ?- [datei].        % Datei laden im REPL

% ── FAKTEN ────────────────────────────────────────────────────────
% Ein Fakt ist eine bedingungslose Aussage.
% Syntax: praedikat(argumente).   <-- Punkt am Ende ist PFLICHT!

mensch(sokrates).           % sokrates ist ein Mensch
mensch(platon).
mensch(aristoteles).

mag(maria, pizza).          % maria mag pizza
mag(maria, sushi).
mag(peter, pizza).

tier(hund).
tier(katze).
tier(vogel).

% ── ANFRAGEN (Queries) ────────────────────────────────────────────
% Im REPL mit ?- stellen:

% ?- mensch(sokrates).
%    true.

% ?- mensch(zeus).
%    false.

% ?- mensch(X).           % X ist eine Variable (Großbuchstabe!)
%    X = sokrates ;        % Enter für erste Lösung, ; für weitere
%    X = platon ;
%    X = aristoteles.

% ── REGELN ────────────────────────────────────────────────────────
% Syntax: Kopf :- Rumpf.
% Lies: "Kopf ist wahr, WENN Rumpf wahr ist"

sterblich(X) :- mensch(X).   % X ist sterblich, wenn X ein Mensch ist

gross(X) :- mensch(X), sterblich(X).  % , bedeutet UND

% ?- sterblich(sokrates).
%    true.

% ?- sterblich(X).
%    X = sokrates ;
%    X = platon ;
%    X = aristoteles.
Terme – alles in Prolog ist ein Term 
% In Prolog gibt es nur EINE Datenstruktur: den Term.
% Alles – Fakten, Regeln, Daten, Code – ist ein Term.

% ── TYPEN VON TERMEN ──────────────────────────────────────────────

% 1. ATOME – Konstanten (Kleinbuchstabe oder in Quotes)
hallo               % Atom
sokrates            % Atom
'Hallo Welt'        % Atom mit Leerzeichen (Quotes nötig)
'it''s'             % Atom mit Apostroph ('' escaped)
[]                  % Leere Liste – auch ein Atom!

% 2. ZAHLEN
42                  % Integer
-7                  % Negativ
3.14                % Float
0'A                 % Char-Code von 'A' = 65
0b1010              % Binär: 10
0xff                % Hex: 255

% 3. VARIABLEN – beginnen mit GROSSBUCHSTABE oder _
X                   % Variable
Ergebnis            % Variable
_                   % Anonyme Variable (interessiert uns nicht)
_Hilfsvariable      % Benannte anonyme Variable (kein Warning)

% WICHTIG: Variablen sind NICHT veränderlich!
% Wenn X einmal gebunden ist, bleibt es gebunden.
% X = 5, X = 6 → FEHLER (kein Zuweisen, nur Unifikation!)

% 4. ZUSAMMENGESETZTE TERME (Funktoren)
punkt(3, 4)         % Funktor 'punkt', Arität 2
datum(2024, 1, 15)  % Funktor 'datum', Arität 3
baum(links, wurzel, rechts)  % Rekursiver Baum

% 5. LISTEN – syntaktischer Zucker für Terme
[1, 2, 3]           % Liste
[H|T]               % H = Kopf, T = Schwanz (Rest)
[1, 2|Rest]         % Erste zwei + Rest
[]                  % Leere Liste

% [1, 2, 3] ist intern: '.'(1, '.'(2, '.'(3, [])))
% (Cons-Zellen wie in Lisp/Scheme!)

% 6. OPERATOREN (nur Notation für Terme!)
1 + 2               % ist: +(1, 2)
X = Y               % ist: =(X, Y)  – Unifikation!
X :- Y              % ist: :-(X, Y) – Regel!
% Prolog behandelt alle Ausdrücke gleich.
Unifikation – das Herzstück 
% UNIFIKATION ist der fundamentalste Vorgang in Prolog.
% = bedeutet NICHT Zuweisung, sondern "kann gleichgemacht werden"?

% Im REPL:
% ?- X = 42.
%    X = 42.            % X wird an 42 gebunden

% ?- f(X, 2) = f(1, Y).
%    X = 1, Y = 2.      % Strukturell angeglichen!

% ?- [H|T] = [1, 2, 3].
%    H = 1, T = [2, 3]. % Listen-Dekonstruktion!

% ?- X = X.
%    true.              % Tautologie

% ?- 1 = 2.
%    false.             % Schlägt fehl

% ── VERSCHIEDENE GLEICHHEITSOPERATOREN ───────────────────────────
%
%  =     Unifikation (bindet Variablen)
%  \=    Unifikation schlägt fehl (nicht vereinbar)
%  ==    Strukturelle Gleichheit (ohne Bindung)
%  \==   Strukturelle Ungleichheit
%  is    Arithmetische Auswertung
%  =:=   Arithmetische Gleichheit
%  =\=   Arithmetische Ungleichheit

% Beispiele:
% ?- X = 1+2.        X = 1+2.      % X wird an den TERM 1+2 gebunden!
% ?- X is 1+2.       X = 3.        % X wird an den WERT 3 gebunden!
% ?- 1+2 == 1+2.     true.         % Strukturell gleich
% ?- 1+2 =:= 3.      true.         % Arithmetisch gleich
% ?- X == Y.         false.        % Zwei ungebundene Vars sind NICHT gleich!

% ── PATTERN MATCHING mit Unifikation ─────────────────────────────
% Prolog nutzt Unifikation für mächtiges Pattern Matching:

beschreibe([]).
beschreibe([_]).
beschreibe([_, _]).
beschreibe([H|_]) :- number(H), H > 0.

% ?- beschreibe([1, 2, 3]).
% Prolog probiert alle Regeln durch!
Backtracking & Kontrolle 
% BACKTRACKING: Prolog probiert automatisch alle Möglichkeiten!
% Wenn eine Lösung fehlschlägt, geht Prolog zurück und versucht die nächste.

% Beispiel:
farbe(rot).
farbe(gruen).
farbe(blau).

% ?- farbe(X), farbe(Y), X \== Y.
%    X = rot,   Y = gruen ;   % Prolog findet alle Paare!
%    X = rot,   Y = blau ;
%    X = gruen, Y = rot ;
%    ... usw.

% ── CUT (!) – Backtracking stoppen ───────────────────────────────
% ! sagt: "Hier nicht mehr zurückgehen!"
% Nützlich für Effizienz und um "erstes Ergebnis" zu erzwingen.

max(X, Y, X) :- X >= Y, !.   % Wenn X >= Y: X ist Maximum, fertig!
max(_, Y, Y).                  % Sonst: Y ist Maximum

% Ohne Cut würde Prolog auch die zweite Regel probieren.

% ── NEGATION ALS FAILURE (\+) ────────────────────────────────────
% \+ Goal = wahr, wenn Goal FEHLSCHLÄGT (Closed World Assumption!)
% NICHT logisches NOT – sondern "beweisbar falsch in dieser Datenbank"

vogel(tweety).
vogel(opus).
pinguin(opus).

kann_fliegen(V) :-
    vogel(V),
    \+ pinguin(V).   % \+ = "es ist nicht beweisbar, dass V ein Pinguin ist"

% ?- kann_fliegen(tweety).   true.
% ?- kann_fliegen(opus).     false.  (opus ist Pinguin!)

% ── IF-THEN-ELSE mit -> ; ─────────────────────────────────────────
% (Bedingung -> Dann ; Sonst)

vorzeichen(X, pos) :- X > 0, !.
vorzeichen(X, neg) :- X < 0, !.
vorzeichen(0, null).

% Alternativ mit ->:
vorzeichen2(X, V) :-
    (X > 0 -> V = pos
    ; X < 0 -> V = neg
    ;          V = null
    ).

% ── ONCE/1 – genau eine Lösung ───────────────────────────────────
% once(Goal) = wie call(Goal) aber ohne Backtracking zurück ins Goal
erste_farbe(X) :- once(farbe(X)).

% ?- erste_farbe(X).   X = rot.  (Nur eine Antwort!)

% ── BETWEEN für Schleifen ─────────────────────────────────────────
% between(Low, High, X) – erzeugt X von Low bis High

quadrate :-
    between(1, 5, X),
    Q is X * X,
    format("~w^2 = ~w~n", [X, Q]),
    fail.      % fail zwingt Backtracking = iteriert alle X!
quadrate.      % Zweite Klausel: Erfolg nach Erschöpfung
Listen & Rekursion 
% Listen sind das wichtigste Werkzeug in Prolog!
% [Kopf|Schwanz]  =  cons(Kopf, Schwanz)

% ── GRUNDLEGENDE LISTEN-PRÄDIKATE ────────────────────────────────

% Länge (rekursiv definiert)
mein_length([], 0).
mein_length([_|T], N) :-
    mein_length(T, N1),
    N is N1 + 1.

% ?- mein_length([a, b, c], N).   N = 3.

% Mitgliedschaft
mein_member(X, [X|_]).         % X ist Kopf: gefunden!
mein_member(X, [_|T]) :-       % Sonst: suche in Schwanz
    mein_member(X, T).

% ?- mein_member(b, [a, b, c]).  true.
% ?- mein_member(X, [1, 2, 3]). X = 1 ; X = 2 ; X = 3.

% Anhängen – auch UMGEKEHRT benutzbar!
mein_append([], L, L).
mein_append([H|T], L, [H|R]) :-
    mein_append(T, L, R).

% ?- mein_append([1,2], [3,4], X).  X = [1,2,3,4].
% ?- mein_append(X, Y, [1,2,3]).    % Alle Splits! Magic!
%    X = [],       Y = [1,2,3] ;
%    X = [1],      Y = [2,3] ;
%    X = [1,2],    Y = [3] ;
%    X = [1,2,3],  Y = [].

% Umkehren (Akkumulator-Stil)
mein_reverse(L, R) :- mein_reverse(L, [], R).
mein_reverse([], Acc, Acc).
mein_reverse([H|T], Acc, R) :-
    mein_reverse(T, [H|Acc], R).

% Letztes Element
mein_last([X], X).
mein_last([_|T], X) :- mein_last(T, X).

% ── LISTEN ERZEUGEN ───────────────────────────────────────────────
% numlist(Low, High, List) – erzeugt Zahlenliste
% ?- numlist(1, 5, L).   L = [1,2,3,4,5].

% Eingebaute Prädikate (SWI-Prolog):
% append(L1, L2, L3)
% length(L, N)
% member(X, L)
% last(L, X)
% nth0(I, L, X)   – 0-basiert
% nth1(I, L, X)   – 1-basiert
% flatten(L, F)
% sort(L, Sorted)  – auch Duplikate entfernen!
% msort(L, Sorted) – sortieren ohne Duplikate zu entfernen
Arithmetik 
% WICHTIG: is/2 wertet Arithmetik aus!
% X = 1+2  →  X wird an den TERM 1+2 gebunden (nicht 3!)
% X is 1+2 →  X wird an den WERT 3 gebunden

% ── GRUNDLEGENDE ARITHMETIK ──────────────────────────────────────
?- X is 2 + 3.          % X = 5
?- X is 10 - 4.         % X = 6
?- X is 3 * 7.          % X = 21
?- X is 17 / 5.         % X = 3.4    (Float!)
?- X is 17 // 5.        % X = 3      (Integer-Division)
?- X is 17 mod 5.       % X = 2      (Rest)
?- X is 2 ** 10.        % X = 1024.0 (Potenz, gibt Float)
?- X is 2 ^ 10.         % X = 1024   (Potenz, Integer)
?- X is abs(-5).        % X = 5
?- X is max(3, 7).      % X = 7
?- X is min(3, 7).      % X = 3
?- X is sqrt(9.0).      % X = 3.0
?- X is floor(3.7).     % X = 3
?- X is ceiling(3.2).   % X = 4
?- X is round(3.5).     % X = 4
?- X is truncate(3.9).  % X = 3

% Bitweise:
?- X is 5 /\ 3.         % X = 1     (AND)
?- X is 5 \/ 3.         % X = 7     (OR)
?- X is 5 xor 3.        % X = 6
?- X is .             % Bitweise NOT
?- X is 5 >> 1.         % X = 2     (Rechtsshift)
?- X is 5 << 1.         % X = 10    (Linksshift)

% ── VERGLEICHE ────────────────────────────────────────────────────
?- 5 > 3.          % true
?- 5 < 3.          % false
?- 5 >= 5.         % true
?- 5 =< 5.         % true  (ACHTUNG: =< nicht <=!)
?- 5 =:= 5.        % true  (arithmetische Gleichheit)
?- 5 =\= 3.        % true  (arithmetische Ungleichheit)

% ── AKKUMULATOR-MUSTER ───────────────────────────────────────────
% Für effiziente rekursive Berechnungen

summe(L, S) :- summe(L, 0, S).
summe([], Acc, Acc).
summe([H|T], Acc, S) :-
    Neu is Acc + H,
    summe(T, Neu, S).

% ?- summe([1,2,3,4,5], S).   S = 15.

% Fakultät (tail-recursive mit Akkumulator)
fak(N, F) :- fak(N, 1, F).
fak(0, Acc, Acc) :- !.
fak(N, Acc, F) :-
    N > 0,
    N1 is N - 1,
    Acc1 is Acc * N,
    fak(N1, Acc1, F).

% ?- fak(10, F).   F = 3628800.
Meta-Prädikate: findall, bagof, setof 
% Meta-Prädikate sammeln ALLE Lösungen auf einmal!

% Daten für Beispiele:
alter(alice, 30).
alter(bob,   25).
alter(carol, 35).
alter(dave,  25).

stadt(alice, berlin).
stadt(bob,   hamburg).
stadt(carol, berlin).
stadt(dave,  muenchen).

% ── FINDALL ───────────────────────────────────────────────────────
% findall(Vorlage, Ziel, Liste)
% Sammelt ALLE Vorlage-Werte wo Ziel erfüllt ist.
% Wenn keine Lösung: leere Liste (kein Fehler!)

% Alle Namen:
% ?- findall(X, alter(X, _), Namen).
%    Namen = [alice, bob, carol, dave].

% Alle über 28:
% ?- findall(X, (alter(X, A), A > 28), Ältere).
%    Ältere = [alice, carol].

% Als Paare:
% ?- findall(X-A, alter(X, A), Paare).
%    Paare = [alice-30, bob-25, carol-35, dave-25].

% ── BAGOF ─────────────────────────────────────────────────────────
% bagof(Vorlage, Ziel, Liste)
% Wie findall, aber gruppiert nach freien Variablen.
% Schlägt FEHL wenn keine Lösung (anders als findall!)

% ?- bagof(X, alter(X, 25), Liste).
%    Liste = [bob, dave].         % Alle 25-Jährigen

% ?- bagof(X, A^alter(X, A), Liste).  % ^ = "existiert A sodass"
%    Liste = [alice, bob, carol, dave].

% ── SETOF ─────────────────────────────────────────────────────────
% setof(Vorlage, Ziel, Liste)
% Wie bagof, aber sortiert und ohne Duplikate!

% ?- setof(A, X^alter(X, A), Alter).
%    Alter = [25, 30, 35].        % Eindeutige, sortierte Alter

% ── FORALL ────────────────────────────────────────────────────────
% forall(Bedingung, Aktion)
% Wahr wenn Aktion für ALLE Lösungen von Bedingung wahr ist

% ?- forall(alter(X, A), A > 20).  true. (Alle sind über 20)
% ?- forall(alter(X, A), A > 30).  false. (Nicht alle über 30)

% ── AGGREGATE ─────────────────────────────────────────────────────
% aggregate_all(count, Goal, Count)  – Anzahl
% aggregate_all(max(X), alter(_, X), Max)  – Maximum
% aggregate_all(sum(X), alter(_, X), Sum)  – Summe

% ?- aggregate_all(count, alter(_, _), N).
%    N = 4.

Assert & Retract – dynamische Fakten 
% Prolog-Programme können sich SELBST verändern!
% assert: Fakt hinzufügen
% retract: Fakt entfernen

% Dynamische Prädikate müssen deklariert werden:
:- dynamic counter/1.
:- dynamic besucht/1.
:- dynamic fact/2.

% Initialisieren
:- assertz(counter(0)).

% Zähler erhöhen
erhoehe_counter :-
    retract(counter(N)),     % Alten Wert entfernen
    N1 is N + 1,
    assertz(counter(N1)).    % Neuen Wert hinzufügen

% ?- erhoehe_counter, erhoehe_counter, counter(N).
%    N = 2.

% ── ASSERTA vs ASSERTZ ────────────────────────────────────────────
% asserta: Am ANFANG der Datenbank einfügen (wird zuerst gefunden)
% assertz: Am ENDE der Datenbank einfügen  (wird zuletzt gefunden)

:- assertz(fact(b, 2)).
:- asserta(fact(a, 1)).   % Wird VOR fact(b,2) eingefügt
:- assertz(fact(c, 3)).

% ?- fact(X, _).
%    X = a ;   (asserta: zuerst)
%    X = b ;
%    X = c.

% ── MEMOIZATION mit assert ────────────────────────────────────────
% Berechnete Werte speichern für Effizienz!
:- dynamic fib_memo/2.

fib(0, 0) :- !.
fib(1, 1) :- !.
fib(N, F) :-
    (fib_memo(N, F) -> true    % Schon berechnet? Direkt zurück!
    ;
        N1 is N - 1,
        N2 is N - 2,
        fib(N1, F1),
        fib(N2, F2),
        F is F1 + F2,
        assertz(fib_memo(N, F))  % Ergebnis merken
    ).

% ?- fib(30, F).   F = 832040.  (Viel schneller als ohne Memo!)

% ── RETRACTALL ────────────────────────────────────────────────────
% retractall(Head) – alle passenden Fakten löschen
% (Kein Fehler wenn nichts zum Löschen vorhanden)
:- retractall(fact(_, _)).   % Alle fact/2 löschen
DCG – Definite Clause Grammars 
% DCGs sind eine elegante Erweiterung für Grammatiken!
% --> statt :- für Grammatikregeln
% Prolog kann damit Sätze PARSEN und GENERIEREN

% ── EINFACHE GRAMMATIK ────────────────────────────────────────────
% Definiere eine Sprache: Satz = Subjekt + Verb + Objekt

satz --> subjekt, verb, objekt.

subjekt --> [der, mann].
subjekt --> [die, frau].
subjekt --> [das, kind].

verb --> [sieht].
verb --> [liebt].
verb --> [kennt].

objekt --> [den, hund].
objekt --> [die, katze].
objekt --> [ein, buch].

% Parsen:
% ?- phrase(satz, [der, mann, sieht, die, katze]).
%    true.

% ?- phrase(satz, [die, frau, mag, den, hund]).
%    false.  (mag ist kein Verb in unserer Grammatik)

% Generieren (alle möglichen Sätze!):
% ?- phrase(satz, S).
%    S = [der, mann, sieht, den, hund] ;
%    S = [der, mann, sieht, die, katze] ;
%    ... (18 Kombinationen)

% ── DCG MIT AKTIONEN ─────────────────────────────────────────────
% {} für Prolog-Code innerhalb einer Grammatikregel

% Arithmetischer Ausdruck Parser
ausdruck(E) --> term(T), ausdruck_rest(T, E).

ausdruck_rest(T, E) --> [+], term(T2), { T3 is T + T2 }, ausdruck_rest(T3, E).
ausdruck_rest(T, E) --> [-], term(T2), { T3 is T - T2 }, ausdruck_rest(T3, E).
ausdruck_rest(T, T) --> [].

term(T) --> [T], { number(T) }.

% ?- phrase(ausdruck(E), [3, +, 4, -, 1]).
%    E = 6.

% ── DCG NUTZEN ────────────────────────────────────────────────────
% phrase(Regel, Liste)           – parsen
% phrase(Regel, Liste, Rest)     – parsen mit Rest
% phrase(Regel, Liste)           – auch generieren wenn Liste var!

% ── ABGELEITETE REGELN ────────────────────────────────────────────
mutter(M, K) :- elternteil(M, K), weiblich(M).
vater(V, K)  :- elternteil(V, K), maennlich(V).

grosselternteil(G, E) :-
    elternteil(G, P),
    elternteil(P, E).

geschwister(X, Y) :-
    elternteil(P, X),
    elternteil(P, Y),
    X \== Y.          % Nicht die eigene Person!

vorfahre(X, Y) :- elternteil(X, Y).           % Direkt
vorfahre(X, Y) :-                              % Transitiv
    elternteil(X, Z),
    vorfahre(Z, Y).

        format("  ~w~n", [N])
    ).

% Das klassische Kryptarithmetik-Problem!
%   SEND
% + MORE
% ------
%  MONEY
%
% Jeder Buchstabe = eine andere Ziffer 0-9
% S und M dürfen nicht 0 sein (führende Stelle)

:- use_module(library(clpfd)).  % Constraint Logic Programming!

send_more_money(Vars) :-
    Vars = [S, E, N, D, M, O, R, Y],

    % Alle Variablen: Ziffern 0-9
    Vars ins 0..9,

    % Alle verschieden!
    all_different(Vars),

    % Keine führenden Nullen
    S #\= 0,
    M #\= 0,

    % Die eigentliche Gleichung:
    S*1000 + E*100 + N*10 + D
  + M*1000 + O*100 + R*10 + E
  #= M*10000 + O*1000 + N*100 + E*10 + Y,

    % Lösung suchen
    label(Vars).

% ── N-DAMEN PROBLEM ───────────────────────────────────────────────
% Platziere N Damen auf einem NxN-Schachbrett
% ohne dass sich zwei gegenseitig bedrohen

n_damen(N, Damen) :-
    numlist(1, N, Reihen),
    permutation(Reihen, Damen),      % Spalten als Permutation
    sicher(Damen).

sicher([]).
sicher([D|Ds]) :-
    kein_angriff(D, Ds, 1),
    sicher(Ds).

kein_angriff(_, [], _).
kein_angriff(D, [D1|Ds], Dist) :-
    D1 - D =\= Dist,
    D  - D1 =\= Dist,               % Keine Diagonale
    Dist1 is Dist + 1,
    kein_angriff(D, Ds, Dist1).

% ── QUICKSORT ─────────────────────────────────────────────────────
quicksort([], []).
quicksort([H|T], Sorted) :-
    partition(H, T, Kleiner, Groesser),
    quicksort(Kleiner, SortKleiner),
    quicksort(Groesser, SortGroesser),
    append(SortKleiner, [H|SortGroesser], Sorted).

partition(_, [], [], []).
partition(Pivot, [H|T], [H|Kleiner], Groesser) :-
    H =< Pivot, !,
    partition(Pivot, T, Kleiner, Groesser).
partition(Pivot, [H|T], Kleiner, [H|Groesser]) :-
    partition(Pivot, T, Kleiner, Groesser).

% ?- quicksort([3,1,4,1,5,9,2,6], S).
%    S = [1,1,2,3,4,5,6,9].

% ── MERGESORT ────────────────────────────────────────────────────
mergesort([], []) :- !.
mergesort([X], [X]) :- !.
mergesort(Liste, Sorted) :-
    halve(Liste, L1, L2),
    mergesort(L1, Sorted1),
    mergesort(L2, Sorted2),
    merge_sorted(Sorted1, Sorted2, Sorted).

halve(Liste, L1, L2) :-
    length(Liste, N),
    Half is N // 2,
    length(L1, Half),
    append(L1, L2, Liste).

merge_sorted([], L, L).
merge_sorted(L, [], L).
merge_sorted([H1|T1], [H2|T2], [H1|Rest]) :-
    H1 =< H2, !,
    merge_sorted(T1, [H2|T2], Rest).
merge_sorted([H1|T1], [H2|T2], [H2|Rest]) :-
    merge_sorted([H1|T1], T2, Rest).

% ── FIBONACCI (tail-recursive) ────────────────────────────────────
fib(N, F) :- fib(N, 0, 1, F).
fib(0, F, _, F) :- !.
fib(N, A, B, F) :-
    N > 0,
    N1 is N - 1,
    C is A + B,
    fib(N1, B, C, F).

    hat_symptom(schnupfen),

    hat_symptom(fieber),

diagnostiziere :-
    retractall(hat_symptom(_)),   % Vorherige Symptome löschen

    (Antwort = j -> assertz(hat_symptom(S)) ; true),
