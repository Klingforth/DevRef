20  REM
30  REM Der Apple II hat KEINEN echten Sound-Chip!
40  REM Töne werden durch schnelles POKE zum Lautsprecher erzeugt

70  REM EINFACHER TON (Frequenz durch Verzögerungsschleife)
80  FOR I = 1 TO 100
90    POKE 49200, 0       : REM Lautsprecher togglen
100   FOR J = 1 TO 5 : NEXT J   : REM Verzögerung = Frequenz
110 NEXT I
120 REM
130 REM TON-UNTERPROGRAMM (VARIABLER TON)
140 GOSUB 1000
150 REM
160 REM MELODIE SPIELEN
170 DATA 20, 30, 20, 30, 20, 30, 50  : REM Frequenz-Daten (Verzögerung)
180 DATA 10, 10, 10, 10, 10, 10, 15  : REM Dauer-Daten
190 FOR I = 1 TO 7
200   READ FR : READ DU
210   F = FR : D = DU
220   GOSUB 1000
230   FOR P = 1 TO 500 : NEXT P : REM Pause zwischen Tönen
240 NEXT I
250 END
260 REM
1000 REM *** TON-UNTERPROGRAMM ***
1010 REM F = Frequenz (Verzögerung: kleiner = höher)
1020 REM D = Dauer (Anzahl Wiederholungen)
1030 FOR I = 1 TO D * 20
1040   POKE 49200, 0
1050   FOR J = 1 TO F : NEXT J
1060 NEXT I
1070 RETURN
