        // String-Interpolation (nur mit einfachen Anführungszeichen!)

        trace(begruessung);

        var text = "Zeile 1
Zeile 2";

        // Arithmetik
        var summe = 10 + 3;      // 13
        var diff  = 10 - 3;      // 7
        var prod  = 10 * 3;      // 30
        var quot  = 10 / 3;      // 3.333... (IMMER Float!)
        var div   = Std.int(10 / 3); // 3 (Integer-Division)

// ── KLASSE ───────────────────────────────────────────────────────
class Person {
    // Felder
    public var name:String;
    public var alter:Int;
    private var _email:String;
    public static var anzahl:Int = 0;

    // Computed Property (get/set)
    public var istErwachsen(get, never):Bool;
    function get_istErwachsen():Bool return alter >= 18;

    // Read-write property
    public var email(get, set):String;
    function get_email():String return _email;
    function set_email(v:String):String {
        if (v.indexOf("@") < 0) throw "Ungültige E-Mail!";
        return _email = v;
    }

    // Konstruktor
    public function new(name:String, alter:Int, email:String = "") {
        this.name = name;
        this.alter = alter;
        this._email = email;
        anzahl++;
    }

    // Methode
    public function vorstellen():String {
        return 'Ich bin $name, $alter Jahre alt.';
    }

    // Static Methode
    public static function erstelle(name:String):Person {
        return new Person(name, 0);
    }

    // toString – automatisch bei trace()
    public function toString():String {
        return 'Person($name, $alter)';
    }
}

// ── ENUM MIT PARAMETERN ───────────────────────────────────────────
enum Form {
    Kreis(radius:Float);
    Rechteck(breite:Float, hoehe:Float);
    Dreieck(a:Float, b:Float, c:Float);
    Punkt;
}

function berechneFlaeche(f:Form):Float {
    return switch f {
        case Kreis(r): Math.PI * r * r;
        case Rechteck(b, h): b * h;
        case Dreieck(a, b, c):
            var s = (a + b + c) / 2;
            Math.sqrt(s * (s-a) * (s-b) * (s-c));
        case Punkt: 0.0;
    }
}

// Verwendung
var k = Form.Kreis(5.0);
trace(berechneFlaeche(k));  // 78.539...

Abstract Types – Haxes Superpower 
// Abstract Types sind Haxes einzigartigstes Feature!
// Compile-Zeit-Wrapper um bestehende Typen
// Kein Runtime-Overhead!

// ── EINFACHER ABSTRACT ────────────────────────────────────────────
abstract Meter(Float) {
    public inline function new(v:Float) this = v;
    public inline function toFloat():Float return this;

    // Operator-Überladung!
    @:op(A + B)
    public function plus(other:Meter):Meter return new Meter(this + other.toFloat());

    @:op(A * B)
    public function mal(skalar:Float):Meter return new Meter(this * skalar);

    public function toString():String return '$this m';
}

abstract Kilogramm(Float) {
    public inline function new(v:Float) this = v;
    public function toString():String return '$this kg';
}

// Typensicherheit zur Compile-Zeit!
var distanz = new Meter(5.0);
var gewicht = new Kilogramm(70.0);
// var falsch = distanz + gewicht;  // COMPILE-FEHLER! Typen inkompatibel

// ── FROM / TO CONVERSIONS ─────────────────────────────────────────
abstract NichtLeererString(String) {
    public inline function new(s:String) {
        if (s.length == 0) throw "String darf nicht leer sein!";
        this = s;
    }

    // Implizite Konvertierung FROM String
    @:from static function von(s:String):NichtLeererString {
        return new NichtLeererString(s);
    }

    // Implizite Konvertierung TO String
    @:to function zuString():String return this;
}

var s:NichtLeererString = "Hallo";  // Automatisch konvertiert!
var normal:String = s;              // Automatisch zurück

// ── ABSTRACT ENUM (kompakter als enum) ────────────────────────────
@:enum abstract Richtung(String) {
    var Nord = "N";
    var Sued = "S";
    var Ost  = "O";
    var West = "W";
}

var r:Richtung = Nord;
trace(r);   // "N"

// ── GENERICS ─────────────────────────────────────────────────────
class Stack<T> {
    var items:Array<T> = [];

    public function new() {}

    public function push(item:T):Void items.push(item);
    public function pop():Null<T> return items.pop();
    public function peek():Null<T> return items[items.length - 1];
    public function isEmpty():Bool return items.length == 0;
    public function size():Int return items.length;
}

var intStack = new Stack<Int>();
intStack.push(1); intStack.push(2);
trace(intStack.pop());   // 2

// Generische Funktion mit Constraint
function maximum<T:Comparable<T>>(a:T, b:T):T {
    return a.compareTo(b) >= 0 ? a : b;
}

class StringExt {
    public static function capitalize(s:String):String {
        if (s.length == 0) return s;
        return s.charAt(0).toUpperCase() + s.substr(1);
    }

    public static function istEmail(s:String):Bool {
        return s.indexOf("@") >= 0;
    }
}

// ── REKURSION ────────────────────────────────────────────────────
function fak(n:Int):Int {
    return if (n <= 1) 1 else n * fak(n - 1);
}

// Tail-recursive (manuell, Haxe hat keine auto-TCO)
function fakTail(n:Int, acc:Int = 1):Int {
    return if (n <= 1) acc else fakTail(n - 1, n * acc);
}

function verarbeite(liste:Array<Int>):Float {
    if (liste.length == 0) return 0.0;
    var max = liste[0];
    for (x in liste) if (x > max) max = x;
    return max;
}

// ── STRING MATCHING ───────────────────────────────────────────────
function parseToken(token:String):Dynamic {
    return switch token {
        case "true": true;
        case "false": false;
        case "null": null;
        case _ if (Std.parseInt(token) != null): Std.parseInt(token);
        case _: token;
    }
}

Makros (Compile-Zeit) 
// Haxe Makros laufen ZUR COMPILE-ZEIT in Haxe!
// Unterschied zu C-Makros: Typsicher, mächtig, kein Text-Ersetzen

// ── AUSDRUCK-MAKRO ────────────────────────────────────────────────
import haxe.macro.Context;
import haxe.macro.Expr;

class Makros {
    // Makro-Funktion: gibt Ausdruck zurück
    public static macro function log(e:Expr):Expr {
        // Position für Fehlermeldungen
        var pos = Context.currentPos();

        // Quelldatei und Zeile zur Compile-Zeit!
        var file = Context.getPosInfos(pos).file;
        var line = Context.getPosInfos(pos).min;

        // Generiere: trace("[file:line] " + e)
        return macro {
            trace("LOG: " + $e);
        };
    }

    // Typ-Inspektion zur Compile-Zeit
    public static macro function feldNamen(t:Expr):ExprOf<Array<String>> {
        var typ = Context.getType(ExprTools.toString(t));
        var felder = switch typ {
            case TInst(ref, _):
                ref.get().fields.get().map(f -> f.name);
            case _: [];
        };
        return macro $v{felder};
    }
}

class Spieler {
    public var x:Float;
    public var y:Float;
    public var stats:SpielStats;
    var geschwindigkeit:Float = 200.0;

    public function new(startX:Float, startY:Float) {
        x = startX;
        y = startY;
        stats = { punkte: 0, leben: 3, level: 1 };
    }

    public function bewege(richtung:Richtung, dt:Float):Void {
        var pixel = geschwindigkeit * dt;
        switch richtung {
            case Links:  x -= pixel;
            case Rechts: x += pixel;
            case Hoch:   y -= pixel;
            case Runter: y += pixel;
        }
    }

    public function sammelPunkte(punkte:Int):Void {
        stats = { ...stats, punkte: stats.punkte + punkte };
    }

    public function verliereLeben():Bool {
        stats = { ...stats, leben: stats.leben - 1 };
        return stats.leben > 0;
    }

    public function toString():String {
        return 'Spieler(${x},${y}) ${stats.punkte}pts ${stats.leben}♥';
    }
}

// ── OPTION TYPE ───────────────────────────────────────────────────
abstract Option<T>(Null<T>) {
    public inline function new(v:Null<T>) this = v;

    public static inline function some<T>(v:T):Option<T> return new Option(v);
    public static inline function none<T>():Option<T> return new Option(null);

    public inline function isSome():Bool return this != null;
    public inline function isNone():Bool return this == null;

    public inline function get():T {
        if (this == null) throw "Option.get() auf None!";
        return this;
    }

    public inline function getOrElse(def:T):T
        return if (this != null) this else def;

    public function map<U>(fn:T->U):Option<U>
        return if (this != null) some(fn(this)) else none();

    public function flatMap<U>(fn:T->Option<U>):Option<U>
        return if (this != null) fn(this) else none();

    public function filter(pred:T->Bool):Option<T>
        return if (this != null && pred(this)) this else none();
}

// ── PIPELINE MIT OPTION ───────────────────────────────────────────
function findeNutzer(id:Int):Option<String> {
    return if (id == 1) Option.some("Alice") else Option.none();
}

function verarbeite(name:String):String {
    return name.toUpperCase();
}

// ── GENERISCHER CACHE ─────────────────────────────────────────────
class Cache<K, V> {
    var store:Map<String, V>;
    var keyFn:K -> String;

    public function new(keyFn:K->String) {
        store = new Map();
        this.keyFn = keyFn;
    }

    public function get(key:K):Null<V>
        return store.get(keyFn(key));

    public function set(key:K, value:V):Void
        store.set(keyFn(key), value);

    public function getOrCompute(key:K, fn:K->V):V {
        var k = keyFn(key);
        if (!store.exists(k))
            store.set(k, fn(key));
        return store.get(k);
    }
}

var userCache = new Cache<Int, String>(id -> 'user_$id');
userCache.set(1, "Alice");
trace(userCache.get(1));   // "Alice"
trace(userCache.get(2));   // null

class QueryBuilder {
    var tabelle:String = "";
    var bedingungen:Array<String> = [];
    var limit:Null<Int> = null;
    var orderBy:Null<String> = null;

    public function new() {}

    public function von(tabelle:String):QueryBuilder {
        this.tabelle = tabelle; return this;
    }

    public function wo(bedingung:String):QueryBuilder {
        bedingungen.push(bedingung); return this;
    }

    public function begrenzeAuf(n:Int):QueryBuilder {
        limit = n; return this;
    }

    public function sortierteNach(feld:String):QueryBuilder {
        orderBy = feld; return this;
    }

    public function baue():String {
        var sql = 'SELECT * FROM $tabelle';
        if (bedingungen.length > 0)
            sql += ' WHERE ' + bedingungen.join(" AND ");
        if (orderBy != null) sql += ' ORDER BY $orderBy';
        if (limit != null)   sql += ' LIMIT $limit';
        return sql;
    }
}

var query = new QueryBuilder()
    .von("nutzer")
    .wo("aktiv = true")
    .wo("alter > 18")
    .sortierteNach("name")
    .begrenzeAuf(10)
    .baue();

trace(query);

// Kein Runtime-Overhead! Kompiliert wie einfache Floats

abstract Meter(Float) from Float {
    public inline function new(v:Float) this = v;

    @:op(A + B) public function plus(b:Meter):Meter return this + (b:Float);
    @:op(A - B) public function minus(b:Meter):Meter return this - (b:Float);
    @:op(A * B) public function mal(b:Float):Meter return this * b;
    @:op(A / B) public function durch(b:Float):Meter return this / b;
    @:op(A < B) public function kleiner(b:Meter):Bool return this < (b:Float);
    @:op(A > B) public function groesser(b:Meter):Bool return this > (b:Float);

    public function toKilometer():Kilometer return this / 1000;
    public function toString():String return '$this m';
}

abstract Kilometer(Float) from Float {
    public inline function new(v:Float) this = v;

    @:op(A + B) public function plus(b:Kilometer):Kilometer return this + (b:Float);
    public function toMeter():Meter return this * 1000;
    public function toString():String return '$this km';
}

abstract Sekunde(Float) from Float {
    public inline function new(v:Float) this = v;
    public function toMinuten():Minute return this / 60;
    public function toString():String return '$this s';
}

abstract Minute(Float) from Float {
    public inline function new(v:Float) this = v;
    public function toSekunden():Sekunde return this * 60;
    public function toString():String return '$this min';
}

// Geschwindigkeit als abstrakt typisierter Wert
abstract MeterProSekunde(Float) from Float {
    public function toString():String return '$this m/s';
    public function toKmH():KilometerProStunde return this * 3.6;
}

abstract KilometerProStunde(Float) from Float {
    public function toString():String return '$this km/h';
    public function toMs():MeterProSekunde return this / 3.6;
}

// Compile-Zeit-Makro: automatisch toString() für alle Klassen!
// Das ist die Stärke von Haxe Makros

import haxe.macro.Context;
import haxe.macro.Expr;
import haxe.macro.Type;

class AutoToString {
    // Build-Makro – transformiert die ganze Klasse
    public static macro function baue():Array<Field> {
        var fields = Context.getBuildFields();
        var pos = Context.currentPos();

        // Felder der aktuellen Klasse sammeln
        var klassenName = Context.getLocalClass().get().name;

        // toString-Implementierung generieren
        var feldAusdruecke:Array<Expr> = [];
        for (feld in fields) {
            switch feld.kind {
                case FVar(_, _):
                    var name = feld.name;
                    // Füge "feldName="+feldName zum String hinzu
                    feldAusdruecke.push(macro $v{name} + "=" + $i{name});
                case _:
            }
        }

        // toString-Funktion erstellen
        var koerper:Expr = if (feldAusdruecke.length == 0) {
            macro $v{klassenName} + "()";
        } else {
            var teile = feldAusdruecke;
            macro $v{klassenName} + "(" + ${
                teile.reduceRight(
                    (a, b) -> macro $b + ", " + $a,
                    teile[0]
                )
            } + ")";
        };

        // Neues Feld zur Klasse hinzufügen
        var toStringFeld:Field = {
            name: "toString",
            access: [APublic, AOverride],
            kind: FFun({
                args: [],
                ret: (macro :String),
                expr: macro return $koerper,
            }),
            pos: pos,
        };

        fields.push(toStringFeld);
        return fields;
    }
}

// Verwendung mit @:build
@:build(AutoToString.baue())
class Person {
    public var name:String;
    public var alter:Int;
    public var email:String;

    public function new(name:String, alter:Int, email:String) {
        this.name = name;
        this.alter = alter;
        this.email = email;
    }
    // toString() wird AUTOMATISCH generiert!
}

class Main {
    static function main() {
        var p = new Person("Alice", 30, "alice@example.com");
        trace(p);
        // Person(name=Alice, alter=30, email=alice@example.com)
        // Kein manuelles toString() geschrieben!
    }
}
