using System;
using System.Collections.Generic;
using System.Text;

public sealed class QueryBuilder
{
    private string?      _tabelle;
    private List<string> _felder    = [];
    private string?      _bedingung;
    private string?      _orderBy;
    private int?         _limit;
    private int?         _offset;

    public QueryBuilder Von(string tabelle)
        => Mutiere(b => b._tabelle = tabelle);

    public QueryBuilder Feld(params string[] felder)
        => Mutiere(b => b._felder.AddRange(felder));

    public QueryBuilder Wo(string bedingung)
        => Mutiere(b => b._bedingung = bedingung);

    public QueryBuilder NachOben(string spalte, bool asc = true)
        => Mutiere(b => b._orderBy = $"{spalte} {(asc ? "ASC" : "DESC")}");

    public QueryBuilder Begrenzen(int limit, int offset = 0)
        => Mutiere(b => { b._limit = limit; b._offset = offset; });

    private QueryBuilder Mutiere(Action<QueryBuilder> aktion)
        { aktion(this); return this; }

    public string Baue()
    {
        if (_tabelle is null)
            throw new InvalidOperationException("Tabelle fehlt");
        var sb = new StringBuilder("SELECT ");
        sb.Append(_felder.Count > 0 ? string.Join(", ", _felder) : "*");
        sb.Append($" FROM {_tabelle}");
        if (_bedingung is not null) sb.Append($" WHERE {_bedingung}");
        if (_orderBy   is not null) sb.Append($" ORDER BY {_orderBy}");
        if (_limit     is not null) sb.Append($" LIMIT {_limit}");
        if (_offset is > 0)         sb.Append($" OFFSET {_offset}");
        return sb.ToString();
    }
}

// Generisches Repository-Pattern
public interface IRepository<T, TId>
{
    Task<T?> FindByIdAsync(TId id);
    Task<IReadOnlyList<T>> FindAllAsync();
    Task<T> AddAsync(T entity);
    Task DeleteAsync(TId id);
}

public abstract class RepositoryBase<T, TId> : IRepository<T, TId>
    where T : class
{
    protected readonly Dictionary<TId, T> _store = new();

    public Task<T?> FindByIdAsync(TId id)
        => Task.FromResult(_store.GetValueOrDefault(id));

    public Task<IReadOnlyList<T>> FindAllAsync()
        => Task.FromResult<IReadOnlyList<T>>([.. _store.Values]);

    public Task<T> AddAsync(T entity)
    {
        _store[GetId(entity)] = entity;
        return Task.FromResult(entity);
    }

    public Task DeleteAsync(TId id)
    {
        _store.Remove(id);
        return Task.CompletedTask;
    }

    protected abstract TId GetId(T entity);
}

public record User(int Id, string Name, string Email);

public class UserRepository : RepositoryBase<User, int>
{
    protected override int GetId(User u) => u.Id;
    public async Task<User?> FindByEmailAsync(string email)
        => (await FindAllAsync()).FirstOrDefault(u => u.Email == email);
}

        => (IsSuccess, Value, Error) = (ok, val, err);
    public static Result<T> Ok(T val)       => new(true,  val, null);
    public static Result<T> Fail(string err) => new(false, default, err);
    public Result<U> Map<U>(Func<T, U> f)
        => IsSuccess ? Result<U>.Ok(f(Value!)) : Result<U>.Fail(Error!);
}

// ASP.NET Core 8 Minimal API
var builder = WebApplication.CreateBuilder(args);
builder.Services.AddSingleton<IUserService, UserService>();
var app = builder.Build();

app.MapGet("/api/users", async (IUserService svc) =>
    await svc.GetAllAsync());

app.MapGet("/api/users/{id:int}", async (int id, IUserService svc) =>
{
    var user = await svc.GetByIdAsync(id);
    return user is not null
        ? Results.Ok(user)
        : Results.NotFound($"User {id} nicht gefunden");
});

app.MapPost("/api/users", async (CreateUserRequest req, IUserService svc) =>
{
    if (string.IsNullOrWhiteSpace(req.Name))
        return Results.BadRequest("Name ist erforderlich");
    var user = await svc.CreateAsync(req.Name, req.Email);
    return Results.Created($"/api/users/{user.Id}", user);
});

app.MapDelete("/api/users/{id:int}", async (int id, IUserService svc) =>
{
    var ok = await svc.DeleteAsync(id);
    return ok ? Results.NoContent() : Results.NotFound();
});

app.Run();

    Task<List<User>> GetAllAsync();
    Task<User?> GetByIdAsync(int id);
    Task<User> CreateAsync(string name, string email);
    Task<bool> DeleteAsync(int id);
}
