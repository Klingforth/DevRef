// HolyC – von Terry A. Davis, für TempleOS (2013)
// "The 3rd Temple of Jerusalem" – ein 64-Bit x86 Betriebssystem,
// das Terry Davis ALLEINE über 15 Jahre entwickelte.

// WAS HOLYC BESONDERS MACHT:
// 1. JIT-Compiler: Code wird SOFORT in RAM kompiliert und ausgeführt
//    Kein Linker, kein ELF-Format, kein Laden von Disk
// 2. Ring 0 überall: KEIN User-Mode / Kernel-Mode Trennung
//    Jedes Programm läuft im Kernel-Modus (wie ein Bootloader!)
// 3. Direkter Hardware-Zugriff: Port I/O, Speicher, Interrupts
// 4. Eingebauter Assembler: Inline-ASM ohne Klimmzüge
// 5. Kein virtuelles Dateisystem: RedSea FS direkt auf Disk
// 6. 640x480, 16 Farben, kein Netzwerk – absichtlich simpel

U0 MinMax(I64 *arr, I64 laenge, I64 *min_out, I64 *max_out)
{
    I64 i;
    *min_out = arr[0];
    *max_out = arr[0];
    for (i = 1; i < laenge; i++) {
        if (arr[i] < *min_out) *min_out = arr[i];
        if (arr[i] > *max_out) *max_out = arr[i];
    }
}

I64 daten[6] = {3, 1, 4, 1, 5, 9};
I64 min_val, max_val;
MinMax(daten, 6, &min_val, &max_val);
Print("Min: %d, Max: %d
", min_val, max_val);   // Min: 1, Max: 9

// ── REKURSION ─────────────────────────────────────────────────────
I64 Fak(I64 n)
{
    if (n <= 1) return 1;
    return n * Fak(n - 1);
}

I64 Fib(I64 n)
{
    if (n <= 1) return n;
    return Fib(n - 1) + Fib(n - 2);
}

Print("5! = %d
", Fak(5));    // 120
Print("Fib(10) = %d
", Fib(10));  // 55

class Punkt
{
    I64 x, y;
};

class Rechteck
{
    I64 x, y;       // Position
    I64 breite;
    I64 hoehe;

    // Methoden – können auf die Felder zugreifen!
    I64 Flaeche()
    {
        return breite * hoehe;
    }

    I64 Umfang()
    {
        return 2 * (breite + hoehe);
    }

    Bool EnthaeltPunkt(I64 px, I64 py)
    {
        return px >= x && px < x + breite &&
               py >= y && py < y + hoehe;
    }
};

// Verwendung
Rechteck r;
r.x = 10;
r.y = 20;
r.breite = 100;
r.hoehe  = 50;

Print("Fläche: %d
", r.Flaeche());       // 5000
Print("Umfang: %d
", r.Umfang());         // 300
Print("Enthält (50,40): %d
",
    r.EnthaeltPunkt(50, 40));               // 1 (TRUE)

// Zeiger auf Klassen-Instanzen
Rechteck *rp = MAlloc(sizeof(Rechteck));
rp->x = 0;
rp->y = 0;

rp->hoehe  = 480;

U0 BeepTon(I64 frequenz)
{
    I64 divisor = 1193180 / frequenz;
    OutU8(0x43, 0xB6);            // PIT Kanal 2, Square Wave
    OutU8(0x42, divisor & 0xFF);  // Low Byte
    OutU8(0x42, (divisor >> 8) & 0xFF); // High Byte
    OutU8(0x61, InU8(0x61) | 3); // Speaker einschalten
    Sleep(200);
    OutU8(0x61, InU8(0x61) & ~3); // Speaker ausschalten
}

BeepTon(440);   // A4 – concert pitch!

// ── VOLL-BILDSCHIRM DEMO ─────────────────────────────────────────
U0 PlasmaDemo()
{
    I64 x, y, t = 0;
    while (!ScanKey()) {    // Bis Taste gedrückt
        CDC *dc = DCAllocGet();
        for (y = 0; y < 480; y++) {
            for (x = 0; x < 640; x++) {
                // Einfache Plasma-Funktion
                I64 v = (I64)(
                    Sin(x * 0.02 + t * 0.1) * 7 +
                    Sin(y * 0.03 + t * 0.07) * 7
                ) + 8;
                DCPixel(dc, x, y, v & 15);
            }
        }
        DCDel(dc);
        t++;
    }
}

// Doppelt verkettete Liste in HolyC
// Zeigt: Klassen, Zeiger, MAlloc/Free, Schleifen

class ListNode
{
    I64 wert;
    ListNode *prev;
    ListNode *next;
};

class LinkedList
{
    ListNode *head;
    ListNode *tail;
    I64 groesse;

    U0 Init()
    {
        head = NULL;
        tail = NULL;
        groesse = 0;
    }

    U0 PushBack(I64 val)
    {
        ListNode *n = CAlloc(sizeof(ListNode));
        n->wert = val;
        n->prev = tail;
        n->next = NULL;

        if (tail != NULL)
            tail->next = n;
        else
            head = n;

        tail = n;
        groesse++;
    }

    I64 PopFront()
    {
        if (head == NULL) return 0;

        ListNode *n = head;
        I64 val = n->wert;
        head = n->next;

        if (head != NULL)
            head->prev = NULL;
        else
            tail = NULL;

        Free(n);
        groesse--;
        return val;
    }

    U0 Print()
    {
        ListNode *n = head;
        while (n != NULL) {
            ::Print("%d -> ", n->wert);  // :: = global Print
            n = n->next;
        }
        ::Print("NULL
");
    }

    U0 Free()
    {
        ListNode *n = head;
        while (n != NULL) {
            ListNode *next = n->next;
            ::Free(n);
            n = next;
        }
        head = NULL;
        tail = NULL;
        groesse = 0;
    }
};

// Hauptprogramm
LinkedList liste;
liste.Init();

liste.PushBack(1);
liste.PushBack(2);
liste.PushBack(3);
liste.Print();              // 1 -> 2 -> 3 -> NULL

Print("Pop: %d
", liste.PopFront());  // 1
liste.Print();              // 2 -> 3 -> NULL
Print("Größe: %d
", liste.groesse);  // 2

// Terry Davis liebte es, Melodien in TempleOS zu spielen.

// Ton-Frequenzen (Hz)
#define C4  262
#define D4  294
#define E4  330
#define F4  349
#define G4  392
#define A4  440
#define B4  494
#define C5  523
#define PAUSE 0

U0 SpieleTon(I64 freq, I64 dauer_ms)
{
    if (freq == 0) {
        Sleep(dauer_ms);   // Pause
        return;
    }

    // PIT (Programmable Interval Timer) konfigurieren
    I64 divisor = 1193180 / freq;
    OutU8(0x43, 0xB6);                   // PIT Kanal 2, Square Wave Mode
    OutU8(0x42, divisor & 0xFF);          // Low Byte
    OutU8(0x42, (divisor >> 8) & 0xFF);  // High Byte

    // Speaker einschalten
    OutU8(0x61, InU8(0x61) | 0x03);

    Sleep(dauer_ms);

    // Speaker ausschalten
    OutU8(0x61, InU8(0x61) & ~0x03);
    Sleep(20);    // Kurze Pause zwischen Noten
}

// "Ode an die Freude" (vereinfacht)
U0 OdeAnDieFreude()
{
    // Noten: {Frequenz, Dauer_ms}
    I64 melodie[][2] = {
        {E4, 300}, {E4, 300}, {F4, 300}, {G4, 300},
        {G4, 300}, {F4, 300}, {E4, 300}, {D4, 300},
        {C4, 300}, {C4, 300}, {D4, 300}, {E4, 300},
        {E4, 450}, {D4, 150}, {D4, 600},
        {PAUSE, 300},
        {E4, 300}, {E4, 300}, {F4, 300}, {G4, 300},
        {G4, 300}, {F4, 300}, {E4, 300}, {D4, 300},
        {C4, 300}, {C4, 300}, {D4, 300}, {E4, 300},
        {D4, 450}, {C4, 150}, {C4, 600},
    };

    I64 i;
    I64 anzahl = sizeof(melodie) / sizeof(melodie[0]);

    Print("Spiele Ode an die Freude...
");
    for (i = 0; i < anzahl; i++)
        SpieleTon(melodie[i][0], melodie[i][1]);

    Print("Fertig!
");
}

OdeAnDieFreude();
// Mandelbrot-Menge in TempleOS Textmodus
// Terry Davis schrieb ähnliche Demos für TempleOS!

U0 Mandelbrot()
{
    // Textmodus: 80x25 Zeichen
    I64 breite  = 78;
    I64 hoehe   = 22;
    F64 x_min   = -2.5;
    F64 x_max   =  1.0;
    F64 y_min   = -1.2;
    F64 y_max   =  1.2;
    I64 max_iter = 64;

    U8 zeichen[] = " .:-=+*#%@";
    I64 anz_zeichen = StrLen(zeichen);

    I64 px, py;
    for (py = 0; py < hoehe; py++) {
        F64 cy = y_min + (y_max - y_min) * py / hoehe;

        for (px = 0; px < breite; px++) {
            F64 cx = x_min + (x_max - x_min) * px / breite;

            // Mandelbrot-Iteration: z = z² + c
            F64 zr = 0.0, zi = 0.0;
            I64 iter = 0;

            while (zr*zr + zi*zi < 4.0 && iter < max_iter) {
                F64 nr = zr*zr - zi*zi + cx;
                F64 ni = 2.0 * zr * zi + cy;
                zr = nr;
                zi = ni;
                iter++;
            }

            // Zeichen je nach Iterationstiefe
            I64 idx = (iter * anz_zeichen) / max_iter;
            if (idx >= anz_zeichen) idx = anz_zeichen - 1;
            Print("%c", zeichen[idx]);
        }
        Print("
");
    }
}

Print("=== MANDELBROT SET ===
");
Mandelbrot();
Print("=====================

// Sieb des Eratosthenes in HolyC
// Zeigt: Arrays, Schleifen, Port-Zugriff für Timing

U0 SiebEratosthenes(I64 limit)
{
    // Boolean-Array: ist_prim[i] = TRUE wenn i prim
    Bool *ist_prim = CAlloc((limit + 1) * sizeof(Bool));
    I64 i, j;

    // Alle als prim markieren
    for (i = 2; i <= limit; i++)
        ist_prim[i] = TRUE;

    // Sieben
    for (i = 2; i * i <= limit; i++) {
        if (ist_prim[i]) {
            // Alle Vielfachen von i als nicht-prim markieren
            for (j = i * i; j <= limit; j += i)
                ist_prim[j] = FALSE;
        }
    }

    // Primzahlen ausgeben
    I64 anzahl = 0;
    I64 pro_zeile = 0;

    for (i = 2; i <= limit; i++) {
        if (ist_prim[i]) {
            Print("%4d", i);
            anzahl++;
            pro_zeile++;
            if (pro_zeile >= 15) {
                Print("
");
                pro_zeile = 0;
            }
        }
    }
    if (pro_zeile > 0) Print("
");

    Print("
Gefunden: %d Primzahlen bis %d
", anzahl, limit);
    Free(ist_prim);
}
