fun main() {
    // val = unveränderlich (wie final in Java)
    val name = "Alice"          // Typ inferiert: String
    val alter: Int = 30         // Expliziter Typ

    // var = veränderlich
    var zaehler = 0
    zaehler++

    // Null-Safety ist eingebaut! Kein NullPointerException by default!
    val sicher: String  = "nie null"  // Nie null
    val nullable: String? = null       // Explizit nullable

    // String-Templates – viel mächtiger als Java
    println("Hallo, $name! Du bist $alter Jahre alt.")
    println("Doppelt: ${alter * 2}")

    // Alles ist ein Ausdruck in Kotlin!
    val label = if (alter >= 18) "Erwachsen" else "Minderjährig"

    // Kein Semikolon nötig (Optional)
    println(label)
}

Variablen, Null-Safety & Smart Casts 
// val – unveränderlich (nach Zuweisung nicht änderbar)
val x = 42
val pi = 3.14159
val name = "Alice"

// var – veränderlich
var zaehler = 0
zaehler = zaehler + 1
zaehler++

// Explizite Typen (wo nötig oder zur Klarheit)
val n: Long = 100_000_000L
val f: Float = 3.14f

// ── NULL-SAFETY ──────────────────────────────────────────────────
val s: String = "Hallo"     // Nie null – Compiler-Garantie!
// s = null                 // Compile-Fehler!

val ns: String? = null      // Explizit nullable

// Safe Call ?.
val laenge = ns?.length     // null wenn ns null (kein NPE!)
val gross  = ns?.uppercase()?.trim()  // Chaining

// Elvis Operator ?: – Default wenn null
val l = ns?.length ?: 0     // 0 wenn ns null
val wert = holeWert() ?: throw IllegalStateException("Null!")

// Not-null assertion !! (Vorsicht! Kann NPE werfen)
val sicher = ns!!.length    // KotlinNullPointerException wenn null!

// ── SMART CASTS ──────────────────────────────────────────────────
// Nach is-Check automatisch gecastet – kein explizites (String)x!
val obj: Any = "Hallo"

if (obj is String) {
    println(obj.length)     // obj ist AUTOMATISCH String hier!
    println(obj.uppercase())
}

// when mit Smart Cast
when (obj) {
    is String  -> println("String: ${obj.length} Zeichen")  // Auto-cast
    is Int     -> println("Int: ${obj * 2}")
    is List<*> -> println("Liste: ${obj.size} Elemente")
    else       -> println("Unbekannt: $obj")
}

class MeinService {

// ── SEALED CLASS – typsichere Discriminated Unions ──────────────
sealed class Ergebnis<out T> {
    data class Erfolg<T>(val wert: T) : Ergebnis<T>()
    data class Fehler(val meldung: String, val ex: Exception? = null) : Ergebnis<Nothing>()
    object Laden : Ergebnis<Nothing>()
}

fun verarbeite(result: Ergebnis<String>): String = when (result) {
    is Ergebnis.Erfolg  -> "Wert: ${result.wert}"   // Smart Cast!
    is Ergebnis.Fehler  -> "Fehler: ${result.meldung}"
    Ergebnis.Laden      -> "Lädt..."
    // kein else nötig – Compiler prüft Vollständigkeit!
}

Extension Functions & Properties 
// Extension Functions – Methoden zu bestehenden Klassen hinzufügen
// OHNE die Klasse zu verändern oder zu erben!

// String erweitern
fun String.capitalize2(): String =
    if (isEmpty()) this else this[0].uppercaseChar() + substring(1)

fun String.istEmail(): Boolean = contains("@") && contains(".")

fun String.wiederhole(n: Int): String = repeat(n)

// Verwendung wie normale Methoden!
println("hallo".capitalize2())    // "Hallo"
println("alice@example.com".istEmail())  // true
println("ab".wiederhole(3))       // "ababab"

// Extension Properties
val String.letztesBuchstabe: Char
    get() = this[length - 1]

val Int.istGerade: Boolean
    get() = this % 2 == 0

println("Hallo".letztesBuchstabe)  // 'o'
println(4.istGerade)               // true

// Extension auf generischem Typ
fun <T> List<T>.zweitesOderNull(): T? = if (size >= 2) this[1] else null

// Extension als Scope-Funktion
fun <T> T.auch(block: (T) -> Unit): T {
    block(this)
    return this
}

// Stdlib-Scope-Funktionen (immer verfügbar!)
val text = "  Hallo, Welt!  "
    .also { println("Vorher: '$it'") }
    .trim()
    .also { println("Nachher: '$it'") }

// let, run, apply, also, with
val ergebnis = "hallo"
    .let { it.uppercase() }    // Gibt Resultat zurück, it = receiver
    .let { "$it!" }            // Verkettung

val person = Person("Alice", 30).apply {
    // apply: this = Person, gibt Person zurück
    // Perfekt für Builder-Pattern!
}

with(person) {
    // with: this = person, kein Rückgabewert erwartet
    println(name)
    println(alter)
}
Lambdas & Higher-Order Functions 
// Funktionen sind First-Class in Kotlin!

// Funktionstyp: (ParameterTypen) -> RückgabeTyp
val verdopple: (Int) -> Int = { x -> x * 2 }
val addiere: (Int, Int) -> Int = { a, b -> a + b }

// Einzeiler-Lambda
val quadrat = { x: Int -> x * x }

// it = impliziter Parametername wenn nur 1 Parameter
val istGerade: (Int) -> Boolean = { it % 2 == 0 }

// Higher-Order Function
fun anwende(wert: Int, f: (Int) -> Int): Int = f(wert)

println(anwende(5, verdopple))    // 10
println(anwende(5) { x -> x * x }) // 25 – Lambda außerhalb der ()
println(anwende(5) { it * 3 })    // 15 – it-Syntax

// Trailing Lambda (häufig in Kotlin!)
// Wenn letzter Parameter ein Lambda: außerhalb der Klammern
listOf(1, 2, 3).forEach { println(it) }
listOf(1, 2, 3).filter { it > 1 }.map { it * 2 }

// Function References (::)
fun verdopple2(x: Int) = x * 2
val ref: (Int) -> Int = ::verdopple2
val auch_ref = String::uppercase     // Method reference

listOf("a", "b", "c").map(String::uppercase)  // ["A", "B", "C"]
listOf(1, 2, 3).map(::verdopple2)

// Closures – Variablen werden captured
var counter = 0
val zähle = { counter++ }   // counter wird captured!
zähle(); zähle(); zähle()
println(counter)             // 3

// Inline Functions (für Performance bei HO-Functions)
inline fun <T> messen(block: () -> T): T {
    val start = System.nanoTime()
    val result = block()
    println("${(System.nanoTime() - start) / 1_000_000} ms")
    return result
}

val ergebnis = messen {
    (1..1_000_000).sum()    // Lambda wird INLINED – kein Overhead!
}

// Einfaches when (wie switch)
val tag = 3
val name = when (tag) {
    1    -> "Montag"
    2    -> "Dienstag"
    3, 4 -> "Mittwoch oder Donnerstag"  // Mehrere Werte!
    in 5..7 -> "Wochentag (5-7)"         // Range!
    else -> "Unbekannt"
}

// when als Ausdruck (IMMER einen Wert)
val beschreibung = when (val alter = getAlter()) {
    in 0..12   -> "Kind"
    in 13..17  -> "Teenager"
    in 18..64  -> "Erwachsener ($alter)"
    else       -> "Senior"
}

// when ohne Argument (wie if-else if)
fun kategorisiere(n: Int): String = when {
    n < 0      -> "negativ"
    n == 0     -> "null"
    n % 2 == 0 -> "gerade positiv"
    else       -> "ungerade positiv"
}

// when mit is (Smart Cast!)
fun verarbeite(obj: Any): String = when (obj) {
    is Int     -> "Int: ${obj * 2}"        // obj ist Auto-Cast zu Int
    is String  -> "String: ${obj.length} Zeichen"
    is List<*> -> "Liste: ${obj.size} Elemente"
    null       -> "null"
    else       -> obj.toString()

Coroutines – async/await in Kotlin 
import kotlinx.coroutines.*

// Coroutines sind Kotlins Antwort auf async/await!
// Keine Callbacks, kein Blocking, leichtgewichtiger als Threads!

// ── GRUNDLAGEN ───────────────────────────────────────────────────
fun main() = runBlocking {  // runBlocking = Brücke zwischen blocking und suspending Welt
    // launch: Fire-and-forget (gibt Job zurück, kein Wert)
    val job = launch {
        delay(1000)          // delay ist NON-BLOCKING (suspending!)
        println("Welt!")
    }
    println("Hallo,")        // Wird zuerst gedruckt
    job.join()               // Warte auf Job
    // Ausgabe: "Hallo," dann "Welt!" (nach 1 Sekunde)
}

// suspend fun – Funktion die pausieren kann (ohne Thread zu blockieren!)
suspend fun holeDaten(url: String): String {
    delay(100)               // Simuliert Netzwerkaufruf
    return "Daten von $url"
}

// ── PARALLEL AUSFÜHREN ───────────────────────────────────────────
suspend fun ladeAlles() = coroutineScope {
    // async gibt Deferred<T> zurück (= Future/Promise)
    val daten1 = async { holeDaten("api1.example.com") }
    val daten2 = async { holeDaten("api2.example.com") }
    val daten3 = async { holeDaten("api3.example.com") }

    // Alle drei laufen PARALLEL!
    val alle = listOf(daten1.await(), daten2.await(), daten3.await())
    alle.joinToString()
}

// ── DISPATCHERS – Auf welchem Thread laufen? ─────────────────────
launch(Dispatchers.IO)     { /* Datei, Netzwerk, Datenbank */ }
launch(Dispatchers.Default){ /* CPU-intensive Berechnungen */ }
launch(Dispatchers.Main)   { /* UI-Updates (Android!)      */ }

// withContext – Kontext wechseln
suspend fun verarbeiteDatei(): String = withContext(Dispatchers.IO) {
    // Läuft im IO-Thread
    val daten = readFile("gross.dat")
    withContext(Dispatchers.Default) {
        // Läuft im Default-Thread für CPU-Arbeit
        berechne(daten)
    }
}

// ── STRUCTURED CONCURRENCY ───────────────────────────────────────
// coroutineScope: Warte auf alle Kind-Coroutines
// supervisorScope: Kind-Fehler canceln nicht die anderen
suspend fun robuste() = supervisorScope {
    val j1 = launch { riskant1() }
    val j2 = launch { riskant2() }
    // j1 kann fehlschlagen ohne j2 zu beenden!
}

// ── CANCELLATION ─────────────────────────────────────────────────
val job = launch {
    repeat(100) { i ->
        println("Arbeite... $i")
        delay(500)            // Cancellation-Punkt!
    }
}
delay(2000)
job.cancel()               // Bricht die Coroutine ab

Flow – Reaktive Streams 
import kotlinx.coroutines.*
import kotlinx.coroutines.flow.*

// Flow = asynchrone Sequence – Werte werden über Zeit emittiert
// Kalt (cold): läuft nur wenn collected

// ── FLOW ERSTELLEN ───────────────────────────────────────────────
fun zahlenFlow(): Flow<Int> = flow {
    for (i in 1..5) {
        delay(100)
        emit(i)              // Wert emittieren
    }
}

// ── FLOW VERARBEITEN ─────────────────────────────────────────────
fun main() = runBlocking {
    zahlenFlow()
        .filter { it % 2 == 0 }
        .map { it * it }
        .collect { println(it) }   // Auslöser!
    // 4, 16
}

// ── FLOW-OPERATOREN ──────────────────────────────────────────────
zahlenFlow()
    .map { it * 2 }           // Transformieren
    .filter { it > 4 }        // Filtern
    .take(3)                  // Erste 3
    .onEach { println(it) }   // Side Effect
    .catch { e -> emit(-1) }  // Fehler abfangen
    .collect { }

// Terminal Operators
zahlenFlow().toList()         // [1, 2, 3, 4, 5]
zahlenFlow().first()          // 1
zahlenFlow().single()         // Exception wenn != 1 Element
zahlenFlow().count()          // 5
zahlenFlow().fold(0) { acc, v -> acc + v }  // 15

// ── STATEFLOW & SHAREDFLOW ───────────────────────────────────────
// StateFlow = Observable State (wie LiveData in Android)
val _zustand = MutableStateFlow(0)
val zustand: StateFlow<Int> = _zustand.asStateFlow()

// Wert ändern
_zustand.value = 42

// Beobachten
zustand.collect { println("Neuer Zustand: $it") }

// SharedFlow = Hot Flow (für Events)
val _events = MutableSharedFlow<String>()
val events: SharedFlow<String> = _events.asSharedFlow()

// ── COMBINE & MERGE ──────────────────────────────────────────────
val flow1 = flowOf(1, 2, 3)
val flow2 = flowOf("a", "b", "c")

combine(flow1, flow2) { n, s -> "$n$s" }
    .collect { println(it) }   // "1a", "2b", "3c"

merge(zahlenFlow(), andererFlow())   // Zusammenführen

// Typsichere State Machine mit sealed classes

sealed class LadeZustand<out T> {
    object Warten  : LadeZustand<Nothing>()
    object Laden   : LadeZustand<Nothing>()
    data class Erfolg<T>(val daten: T)  : LadeZustand<T>()
    data class Fehler(val ex: Throwable): LadeZustand<Nothing>()
}

data class User(val id: Int, val name: String, val email: String)

// Repository
class UserRepository {
    private val users = mapOf(
        1 to User(1, "Alice", "alice@example.com"),
        2 to User(2, "Bob",   "bob@example.com"),
    )

    suspend fun ladeUser(id: Int): User =
        users[id] ?: throw NoSuchElementException("User $id nicht gefunden")
}

// ViewModel (vereinfacht, kein Android nötig)
class UserViewModel(private val repo: UserRepository) {
    var zustand: LadeZustand<User> = LadeZustand.Warten
        private set

    suspend fun ladeUser(id: Int) {
        zustand = LadeZustand.Laden
        zustand = try {
            LadeZustand.Erfolg(repo.ladeUser(id))
        } catch (e: Exception) {
            LadeZustand.Fehler(e)
        }
    }
}

// View – exhaustiver when auf sealed class
fun render(zustand: LadeZustand<User>): String = when (zustand) {
    LadeZustand.Warten  -> "Bereit"
    LadeZustand.Laden   -> "Lädt..."
    is LadeZustand.Erfolg -> with(zustand.daten) {
        "User: $name ($email)"
    }
    is LadeZustand.Fehler -> "Fehler: ${zustand.ex.message}"
    // Kein else nötig! Vollständigkeit garantiert.
}

fun main() = runBlocking {
    val vm = UserViewModel(UserRepository())

    println(render(vm.zustand))    // Bereit

    vm.ladeUser(1)
    println(render(vm.zustand))    // User: Alice (alice@example.com)

    vm.ladeUser(99)
    println(render(vm.zustand))    // Fehler: User 99 nicht gefunden
}

// HTML-DSL Beispiel
class HtmlTag(val name: String) {
    private val attribute = mutableMapOf<String, String>()
    private val kinder = mutableListOf<Any>()

    operator fun String.unaryPlus() { kinder.add(this) }

    fun attr(key: String, value: String) { attribute[key] = value }

    fun tag(name: String, block: HtmlTag.() -> Unit): HtmlTag {
        val child = HtmlTag(name)
        child.block()
        kinder.add(child)
        return child
    }

    fun render(einrückung: Int = 0): String {
        val prefix = "  ".repeat(einrückung)
        val attrs = if (attribute.isEmpty()) ""
            else " " + attribute.map { """${it.key}="${it.value}"""" }.joinToString(" ")
        return buildString {
            appendLine("$prefix<$name$attrs>")
            kinder.forEach { kind ->
                when (kind) {
                    is HtmlTag -> appendLine(kind.render(einrückung + 1).trimEnd())
                    is String  -> appendLine("$prefix  $kind")
                }
            }
            append("$prefix</$name>")
        }
    }
}

fun html(block: HtmlTag.() -> Unit) = HtmlTag("html").apply(block)

// Der DSL-Aufruf sieht aus wie echtes HTML!
val dokument = html {
    tag("head") {
        tag("title") { +"Meine Seite" }
    }
    tag("body") {
        attr("class", "container")
        tag("h1") { +"Willkommen" }
        tag("p")  { +"Das ist Kotlin-DSL!" }
        tag("ul") {
            listOf("Punkt 1", "Punkt 2", "Punkt 3").forEach { p ->
                tag("li") { +p }
            }
        }
    }
}

data class Produkt(val id: Int, val name: String, val preis: Double)

// "Datenbank" simulieren
suspend fun holeProdukte(): List<Produkt> {
    delay(50)
    return listOf(
        Produkt(1, "Laptop",   999.99),
        Produkt(2, "Maus",      29.99),
        Produkt(3, "Tastatur",  79.99),
    )
}

suspend fun holeLager(produktId: Int): Lager {
    delay(30)
    return Lager(produktId, (10..100).random())
}

suspend fun holeBewertungen(produktId: Int): Bewertung {
    delay(40)
    return Bewertung(produktId, (30..50).random() / 10.0)
}

data class ProduktDetail(
    val produkt: Produkt,
    val lager: Lager,
    val bewertung: Bewertung,
)

suspend fun ladeDetails(produkt: Produkt): ProduktDetail = coroutineScope {
    // Lager und Bewertung PARALLEL laden!
    val lager     = async { holeLager(produkt.id) }
    val bewertung = async { holeBewertungen(produkt.id) }
    ProduktDetail(produkt, lager.await(), bewertung.await())
}

fun main() = runBlocking {
    val dauer = measureTime {
        val produkte = holeProdukte()

        // ALLE Details parallel laden!
        val details = produkte.map { p ->
            async { ladeDetails(p) }
        }.awaitAll()

        println("=== Produkt-Katalog ===")
        details.forEach { d ->
            println("${d.produkt.name.padEnd(12)} " +
                    "${d.produkt.preis.let { "%.2f€".format(it) }.padStart(10)}  " +
                    "Lager: ${d.lager.menge.toString().padStart(3)}  " +
                    "⭐ ${"%.1f".format(d.bewertung.sterne)}")
        }

        // Flow-basierte Verarbeitung
        println("
=== Günstige Verfügbare ===")
        produkte.asFlow()
            .map { async { ladeDetails(it) } }
            .map { it.await() }
            .filter { it.produkt.preis < 100 && it.lager.menge > 20 }
            .sortedBy { it.produkt.preis }
            .collect { d ->
                println("${d.produkt.name}: ${d.produkt.preis}€")
            }
    }
    println("
Gesamt: $dauer (parallel statt sequenziell!)")
}

// Validator DSL (wie in production Kotlin-Projekten!)

class ValidationError(val feld: String, val nachricht: String)

class Validator<T>(private val wert: T, private val feld: String) {
    private val fehler = mutableListOf<ValidationError>()

    fun istNichtNull(msg: String = "Darf nicht null sein"): Validator<T> {
        if (wert == null) fehler.add(ValidationError(feld, msg))
        return this
    }

    fun erfüllt(bedingung: (T) -> Boolean, msg: String): Validator<T> {
        if (!bedingung(wert)) fehler.add(ValidationError(feld, msg))
        return this
    }

    fun fehler() = fehler.toList()
    fun istGültig() = fehler.isEmpty()
}

class Formularvalidator {
    private val alle = mutableListOf<Validator<*>>()

    fun <T> validiere(wert: T, feld: String, block: Validator<T>.() -> Unit): Validator<T> {
        val v = Validator(wert, feld).apply(block)
        alle.add(v)
        return v
    }

    fun alleGültig() = alle.all { it.istGültig() }
    fun alleFehler() = alle.flatMap { it.fehler() }
}

data class Benutzer(val name: String?, val email: String?, val alter: Int?)

fun validiereBenutzer(b: Benutzer): List<ValidationError> {
    val form = Formularvalidator()

    form.validiere(b.name, "name") {
        istNichtNull()
        erfüllt({ it != null && it.length >= 2 }, "Mindestens 2 Zeichen")
        erfüllt({ it != null && it.length <= 50 }, "Maximal 50 Zeichen")
    }

    form.validiere(b.email, "email") {
        istNichtNull()
        erfüllt({ it != null && "@" in it }, "Muss @ enthalten")
        erfüllt({ it != null && "." in it.substringAfter("@") }, "Ungültige Domain")
    }

    form.validiere(b.alter, "alter") {
        istNichtNull("Alter ist erforderlich")
        erfüllt({ it != null && it in 0..150 }, "Muss zwischen 0 und 150 sein")
    }

    return form.alleFehler()
}

fun main() {
    val gültig = Benutzer("Alice", "alice@example.com", 30)
    val ungültig = Benutzer(null, "keine-email", 200)

    println("=== Gültig ===")
    validiereBenutzer(gültig).forEach { println("${it.feld}: ${it.nachricht}") }
    println("Keine Fehler: ${validiereBenutzer(gültig).isEmpty()}")

    println("
=== Ungültig ===")
    validiereBenutzer(ungültig).forEach { println("${it.feld}: ${it.nachricht}") }
}
