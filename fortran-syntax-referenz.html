! ── SUBROUTINE ───────────────────────────────────────────────────
subroutine tausche(a, b)
    real(real64), intent(inout) :: a, b   ! in-out: beides!
    real(real64) :: temp
    temp = a
    a    = b
    b    = temp
end subroutine

! Intent-Attribute:
! intent(in)    -- nur lesen (Wert nicht aenderbar)
! intent(out)   -- nur schreiben (Eingabewert unbekannt)
! intent(inout) -- lesen und schreiben

subroutine statistik(daten, n, mittel, stdabw)
    integer, intent(in)  :: n
    real(real64), intent(in)  :: daten(n)
    real(real64), intent(out) :: mittel, stdabw

    mittel = sum(daten) / n
    stdabw = sqrt(sum((daten - mittel)**2) / (n-1))
end subroutine

module mathe_konstanten
    use iso_fortran_env, only: real64
    implicit none

    ! Alles in diesem Modul ist oeffentlich (default)
    real(real64), parameter :: PI    = acos(-1.0_real64)
    real(real64), parameter :: E     = exp(1.0_real64)
    real(real64), parameter :: SQRT2 = sqrt(2.0_real64)

    private :: intern_hilfsfunktion   ! Nur intern sichtbar

contains

    pure function grad_zu_rad(grad) result(rad)
        real(real64), intent(in) :: grad
        real(real64) :: rad
        rad = grad * PI / 180.0_real64
    end function

    ! Private Hilfsfunktion (nicht nach aussen sichtbar)
    pure function intern_hilfsfunktion(x) result(y)
        real(real64), intent(in) :: x
        real(real64) :: y
        y = x * x
    end function

end module mathe_konstanten

module physik_modul
    use iso_fortran_env, only: real64
    implicit none

    ! Basistyp
    type :: Teilchen
        real(real64) :: masse
        real(real64) :: position(3)
        real(real64) :: geschwindigkeit(3)
    contains
        procedure :: kinetische_energie
        procedure :: bewege
        procedure :: drucke => drucke_teilchen
    end type Teilchen

    ! Vererbung (Fortran 2003)
    type, extends(Teilchen) :: GeladenesTeilchen
        real(real64) :: ladung
    contains
        procedure :: drucke => drucke_geladen   ! Override!
    end type GeladenesTeilchen

contains

    ! Gebundene Methode (type-bound procedure)
    function kinetische_energie(self) result(ek)
        class(Teilchen), intent(in) :: self
        real(real64) :: ek
        ek = 0.5_real64 * self%masse * sum(self%geschwindigkeit**2)
    end function

    subroutine bewege(self, dt)
        class(Teilchen), intent(inout) :: self
        real(real64), intent(in) :: dt
        self%position = self%position + self%geschwindigkeit * dt
    end subroutine

    subroutine drucke_teilchen(self)
        class(Teilchen), intent(in) :: self
        write(*,'(a,3f8.3)') "Position:", self%position
        write(*,'(a,f10.4)') "Ek:      ", self%kinetische_energie()
    end subroutine

    subroutine drucke_geladen(self)
        class(GeladenesTeilchen), intent(in) :: self
        call self%Teilchen%drucke()            ! Eltern-Methode aufrufen
        write(*,'(a,f8.3)') "Ladung:  ", self%ladung
    end subroutine

end module physik_modul

module numerik
    use iso_fortran_env, only: real32, real64
    implicit none

    ! Generische Interface (Ueberladung)
    interface norm
        module procedure norm_real32
        module procedure norm_real64
    end interface norm

contains

    ! Zwei Implementierungen, eine generische Schnittstelle
    pure function norm_real32(v) result(n)
        real(real32), intent(in) :: v(:)
        real(real32) :: n
        n = sqrt(sum(v**2))
    end function

    pure function norm_real64(v) result(n)
        real(real64), intent(in) :: v(:)
        real(real64) :: n
        n = sqrt(sum(v**2))
    end function

end module numerik

Coarrays – Parallelprogrammierung (Fortran 2008+) 
! Coarrays: eingebaute Parallelisierung ohne MPI-Bibliothek
! Jeder "Image" (Prozess) hat seine eigene Kopie

program coarray_demo
    use iso_fortran_env
    implicit none

    integer :: n
    real(real64) :: lokale_summe, gesamt_summe
    real(real64), allocatable :: daten(:)

    ! Coarray-Variable -- jeder Image hat eine Kopie
    real(real64) :: teilsumme[*]   ! [*] = Coarray!

    n = 1000000 / num_images()     ! Arbeit aufteilen

    allocate(daten(n))
    daten = [(real(i + (this_image()-1)*n, real64), i=1,n)]

    ! Lokale Berechnung
    teilsumme = sum(daten)

    ! Alle Images synchronisieren
    sync all

    ! Image 1 sammelt alles ein
    if (this_image() == 1) then
        gesamt_summe = 0.0_real64
        do i = 1, num_images()
            gesamt_summe = gesamt_summe + teilsumme[i]   ! [i] = anderes Image!
        end do
        write(*,'(a,f20.0)') "Summe: ", gesamt_summe
    end if

end program coarray_demo

! Fortran und C arbeiten hervorragend zusammen
! iso_c_binding (F2003) macht die Schnittstelle typsicher

module c_schnittstelle
    use iso_c_binding
    implicit none

    ! Typen die C-kompatibel sind
    ! c_int, c_long, c_double, c_float, c_char, c_ptr, c_funptr

    interface
        ! C-Funktion importieren
        function c_sqrt(x) bind(C, name="sqrt") result(y)
            use iso_c_binding
            real(c_double), value, intent(in) :: x
            real(c_double) :: y
        end function c_sqrt

        ! C-Funktion mit String
        subroutine c_puts(str) bind(C, name="puts")
            use iso_c_binding
            character(kind=c_char) :: str(*)
        end subroutine c_puts

    end interface

contains

    ! Fortran-Funktion fuer C exportieren
    function fortran_norm(v, n) result(laenge) bind(C, name="fortran_norm")
        use iso_c_binding
        integer(c_int), value, intent(in) :: n
        real(c_double), intent(in) :: v(n)
        real(c_double) :: laenge
        laenge = sqrt(sum(v**2))
    end function

end module c_schnittstelle

! OpenMP: pragmabasierte Parallelisierung
! Kompilieren: gfortran -fopenmp programm.f90 -o prog

program openmp_demo
    use omp_lib           ! OpenMP-Bibliothek (optional)
    implicit none

    integer :: n = 1000000
    real(8), allocatable :: a(:), b(:), c(:)
    integer :: i, nthreads, thread_id

    allocate(a(n), b(n), c(n))
    a = 1.0d0; b = 2.0d0

    ! Parallele Region
    !$omp parallel
        nthreads  = omp_get_num_threads()
        thread_id = omp_get_thread_num()
        !$omp single
            write(*,'(a,i0,a)') "Starte mit ", nthreads, " Threads"
        !$omp end single
    !$omp end parallel

    ! Parallele Schleife (automatisch aufgeteilt)
    !$omp parallel do private(i) shared(a, b, c)
    do i = 1, n
        c(i) = a(i) + b(i)
    end do
    !$omp end parallel do

    ! Reduktion (kein Race Condition!)
    real(8) :: summe = 0.0d0
    !$omp parallel do reduction(+:summe)
    do i = 1, n
        summe = summe + c(i)
    end do
    !$omp end parallel do

    write(*,'(a,f20.2)') "Summe:", summe

    ! Simd (Vektorisierung)
    !$omp simd
    do i = 1, n
        c(i) = sqrt(a(i)**2 + b(i)**2)
    end do

end program openmp_demo

! Fortrans Staerke: numerische Berechnungen mit LAPACK/BLAS
! gfortran programm.f90 -o prog -llapack -lblas

program lineare_algebra
    use iso_fortran_env, only: real64
    implicit none

    integer, parameter :: n = 3
    real(real64) :: A(n,n), b(n), x(n)
    real(real64) :: eigenvalues(n)
    integer :: ipiv(n), info

    ! Matrix A und Rechte Seite b
    A = reshape([2.0, 1.0, 0.0,  &
                 1.0, 3.0, 1.0,  &
                 0.0, 1.0, 2.0], [n, n])
    b = [1.0_real64, 2.0_real64, 3.0_real64]

    ! LU-Zerlegung und Loesung: Ax = b (LAPACK dgesv)
    x = b
    call dgesv(n, 1, A, n, ipiv, x, n, info)
    if (info == 0) then
        write(*,'(a,3f10.4)') "Loesung x =", x
    else
        write(*,*) "LAPACK-Fehler, info =", info
    end if

    ! Matrix-Vektor-Produkt (BLAS dgemv)
    real(real64) :: y(n)
    call dgemv('N', n, n, 1.0_real64, A, n, x, 1, 0.0_real64, y, 1)
    write(*,'(a,3f10.4)') "Ax =", y     ! Sollte = b sein

    ! Reine Fortran-Alternative (keine LAPACK-Abhaengigkeit)
    y = matmul(A, x)    ! Direkt, aber langsamer fuer grosse Matrizen

    ! Frobenius-Norm
    real(real64) :: fnorm
    fnorm = sqrt(sum(A**2))
    write(*,'(a,f10.4)') "Frobenius-Norm:", fnorm

    ! Spur (Summe der Diagonalen)
    real(real64) :: spur
    spur = sum([(A(i,i), i=1,n)])
    write(*,'(a,f10.4)') "Spur:", spur

end program lineare_algebra
! Monte Carlo: Pi-Berechnung durch Zufallspunkte
! Zeigt Fortrans Staerke bei Array-Operationen

program monte_carlo_pi
    use iso_fortran_env, only: real64, int64
    implicit none

    integer(int64), parameter :: N = 10000000    ! 10 Millionen Punkte
    real(real64), allocatable :: x(:), y(:)
    real(real64) :: pi_approx
    integer(int64) :: treffer
    integer :: seed_groesse
    integer, allocatable :: seed(:)

    ! Zufallszahlen-Generator initialisieren
    call random_seed(size=seed_groesse)
    allocate(seed(seed_groesse))
    seed = 42   ! Reproduzierbar
    call random_seed(put=seed)

    ! Punkte generieren (als Arrays -- sehr schnell!)
    allocate(x(N), y(N))
    call random_number(x)    ! Alle N Zahlen auf einmal!
    call random_number(y)

    ! Bereich [-1, 1] x [-1, 1]
    x = 2.0_real64 * x - 1.0_real64
    y = 2.0_real64 * y - 1.0_real64

    ! Zählen wie viele im Einheitskreis liegen
    ! count() auf logisches Array -- kein Schleife!
    treffer = count(x**2 + y**2 <= 1.0_real64)

    ! Pi-Naherung
    pi_approx = 4.0_real64 * real(treffer, real64) / real(N, real64)

    write(*,'(a,f12.8)') "Pi (approximiert): ", pi_approx
    write(*,'(a,f12.8)') "Pi (exakt):        ", acos(-1.0_real64)
    write(*,'(a,es10.3)') "Fehler:            ", abs(pi_approx - acos(-1.0_real64))

    deallocate(x, y, seed)

end program monte_carlo_pi

! Typische Laufzeit: ~0.1s fuer 10M Punkte

! Vergleich reines Python: ~50x langsamer
! Numerische Integration einer ODE mit Runge-Kutta 4. Ordnung
! dy/dt = f(t, y), y(0) = y0

module ode_modul
    use iso_fortran_env, only: real64
    implicit none

    ! Abstrakte Interface fuer die Ableitung
    abstract interface
        pure function ableitungsfunktion(t, y) result(dydt)
            import real64
            real(real64), intent(in) :: t
            real(real64), intent(in) :: y(:)
            real(real64) :: dydt(size(y))
        end function
    end interface

contains

    ! Runge-Kutta 4. Ordnung Schritt
    pure subroutine rk4_schritt(f, t, y, h, y_neu)
        procedure(ableitungsfunktion) :: f
        real(real64), intent(in)    :: t, h
        real(real64), intent(in)    :: y(:)
        real(real64), intent(out)   :: y_neu(size(y))

        real(real64) :: k1(size(y)), k2(size(y))
        real(real64) :: k3(size(y)), k4(size(y))

        k1 = h * f(t,         y)
        k2 = h * f(t + h/2,   y + k1/2)
        k3 = h * f(t + h/2,   y + k2/2)
        k4 = h * f(t + h,     y + k3)

        y_neu = y + (k1 + 2*k2 + 2*k3 + k4) / 6.0_real64
    end subroutine

    ! Vollstaendige Integration
    subroutine integriere(f, t0, t_end, y0, n_schritte, t_arr, y_arr)
        procedure(ableitungsfunktion) :: f
        real(real64), intent(in)  :: t0, t_end, y0(:)
        integer, intent(in)       :: n_schritte
        real(real64), intent(out) :: t_arr(0:n_schritte)
        real(real64), intent(out) :: y_arr(size(y0), 0:n_schritte)

        real(real64) :: h, t
        integer :: i

        h = (t_end - t0) / n_schritte
        t_arr(0) = t0
        y_arr(:, 0) = y0

        do i = 1, n_schritte
            call rk4_schritt(f, t_arr(i-1), y_arr(:,i-1), h, y_arr(:,i))
            t_arr(i) = t0 + i * h
        end do
    end subroutine

end module ode_modul

! Beispiel: Gedaempfter harmonischer Oszillator
! y'' + 2*gamma*omega*y' + omega^2*y = 0
! als System: [y, v]' = [v, -2*gamma*omega*v - omega^2*y]

program oszillator
    use ode_modul
    use iso_fortran_env, only: real64
    implicit none

    integer, parameter :: n_schritte = 1000
    real(real64) :: t_arr(0:n_schritte), y_arr(2, 0:n_schritte)
    real(real64) :: y0(2)
    integer :: i

    y0 = [1.0_real64, 0.0_real64]   ! Anfangsbedingungen: y=1, v=0

    call integriere(ableitung, 0.0_real64, 10.0_real64, y0, n_schritte, t_arr, y_arr)

    ! Ausgabe
    open(10, file="oszillator.dat", status="replace")
    do i = 0, n_schritte
        write(10,'(3f12.6)') t_arr(i), y_arr(1,i), y_arr(2,i)
    end do
    close(10)
    write(*,*) "Ausgabe in oszillator.dat"

contains

    pure function ableitung(t, y) result(dydt)
        real(real64), intent(in) :: t, y(:)
        real(real64) :: dydt(size(y))
        real(real64), parameter :: omega = 2.0_real64, gamma = 0.1_real64
        dydt(1) = y(2)
        dydt(2) = -2*gamma*omega*y(2) - omega**2*y(1)
    end function

end program oszillator
! N-Koerper-Simulation: Gravitationssystem
! Klassisches HPC-Problem -- Fortran glaenzt hier!

module nbody
    use iso_fortran_env, only: real64
    implicit none

    integer, parameter :: NDIM = 3
    real(real64), parameter :: G = 6.674e-11_real64   ! Gravitationskonstante

contains

    subroutine berechne_kraefte(pos, masse, kraft, n)
        integer, intent(in)  :: n
        real(real64), intent(in)  :: pos(NDIM, n), masse(n)
        real(real64), intent(out) :: kraft(NDIM, n)

        real(real64) :: r(NDIM), abstand, f_mag
        integer :: i, j

        kraft = 0.0_real64

        do j = 1, n
            do i = 1, j-1
                ! Verbindungsvektor
                r = pos(:, j) - pos(:, i)
                abstand = sqrt(sum(r**2)) + 1.0e-10_real64   ! Softening

                ! Gravitationskraft
                f_mag = G * masse(i) * masse(j) / abstand**2

                ! Kraft auf i (in Richtung j)
                kraft(:, i) = kraft(:, i) + f_mag * r / abstand

                ! Reaktionskraft auf j (Newton: actio = reactio)
                kraft(:, j) = kraft(:, j) - f_mag * r / abstand
            end do
        end do
    end subroutine

    subroutine zeitschritt(pos, vel, masse, n, dt)
        integer, intent(in)    :: n
        real(real64), intent(inout) :: pos(NDIM, n), vel(NDIM, n)
        real(real64), intent(in)    :: masse(n), dt

        real(real64) :: kraft(NDIM, n)
        integer :: i

        call berechne_kraefte(pos, masse, kraft, n)

        ! Leapfrog-Integration (energieerhaltend!)
        do i = 1, n
            vel(:, i) = vel(:, i) + kraft(:, i) / masse(i) * dt
            pos(:, i) = pos(:, i) + vel(:, i) * dt
        end do
    end subroutine

end module nbody

program simulation
    use nbody
    use iso_fortran_env, only: real64
    implicit none

    integer, parameter :: N = 4         ! Sonne + 3 Planeten
    integer, parameter :: STEPS = 10000
    real(real64) :: pos(3, N), vel(3, N), masse(N)
    real(real64) :: dt = 86400.0_real64  ! 1 Tag in Sekunden
    integer :: i

    ! Initiale Positionen (vereinfacht, in AU * AU_TO_M)
    ! Sonne, Merkur, Venus, Erde
    real(real64), parameter :: AU = 1.496e11_real64
    masse = [1.989e30_real64, 3.3e23_real64, 4.87e24_real64, 5.97e24_real64]
    pos   = 0.0_real64
    pos(1, 2:4) = [0.39_real64, 0.72_real64, 1.0_real64] * AU   ! x-Positionen

    ! Kreisbahn-Geschwindigkeiten (v = sqrt(GM/r))
    vel = 0.0_real64
    do i = 2, N
        vel(2, i) = sqrt(G * masse(1) / pos(1, i))   ! y-Geschwindigkeit
    end do

    ! Simulation
    do i = 1, STEPS
        call zeitschritt(pos, vel, masse, N, dt)
    end do

    write(*,'(a)') "Endpositionen (in AU):"
    do i = 1, N
        write(*,'(i3,3f12.4)') i, pos(:,i) / AU
    end do
