Vergleichsoperatoren 
# ── Numerisch (in [[ ]] und (( )) ) ────────────────────────────
[[ $a -eq $b ]]   # equal
[[ $a -ne $b ]]   # not equal
[[ $a -lt $b ]]   # less than
[[ $a -le $b ]]   # less or equal
[[ $a -gt $b ]]   # greater than
[[ $a -ge $b ]]   # greater or equal

# In (( )) normale Operatoren möglich:
(( a == b ))
(( a != b ))
(( a < b ))
(( a > b ))
(( a <= b ))
(( a >= b ))

# ── Strings (in [[ ]] ) ─────────────────────────────────────────
[[ "$a" == "$b" ]]    # gleich
[[ "$a" != "$b" ]]    # ungleich
[[ "$a" < "$b" ]]     # lexikographisch kleiner
[[ "$a" > "$b" ]]     # lexikographisch größer
[[ -z "$a" ]]         # leer (zero length)
[[ -n "$a" ]]         # nicht leer (non-zero)
[[ "$a" =~ regex ]]   # Regex-Match (Bash 3.2+)

# ── Datei-Tests ─────────────────────────────────────────────────
[[ -e pfad ]]    # existiert (exists)
[[ -f pfad ]]    # reguläre Datei (file)
[[ -d pfad ]]    # Verzeichnis (directory)
[[ -L pfad ]]    # symbolischer Link
[[ -r pfad ]]    # lesbar (readable)
[[ -w pfad ]]    # schreibbar (writable)
[[ -x pfad ]]    # ausführbar (executable)
[[ -s pfad ]]    # nicht leer (size > 0)
[[ -p pfad ]]    # Named Pipe (FIFO)
[[ -S pfad ]]    # Socket
[[ -b pfad ]]    # Block-Gerät
[[ -c pfad ]]    # Character-Gerät
[[ f1 -nt f2 ]]  # f1 neuer als f2 (newer than)
[[ f1 -ot f2 ]]  # f1 älter als f2 (older than)
[[ f1 -ef f2 ]]  # f1 und f2 sind dasselbe Inode (equal file)

mit_local() {

# Funktionen können rekursiv sein
fakultät() {
    local n=$1
    if (( n <= 1 )); then
        echo 1
        return
    fi
    local sub
    sub=$(fakultät $((n - 1)))
    echo $(( n * sub ))
}

echo $(fakultät 5)    # 120

# Funktion mit Validierung
lade_datei() {
    local pfad="$1"

    [[ -z "$pfad" ]]   && fehler "Kein Pfad angegeben"
    [[ -f "$pfad" ]]   || fehler "Datei nicht gefunden: $pfad"
    [[ -r "$pfad" ]]   || fehler "Keine Leseberechtigung: $pfad"

    cat "$pfad"
}

LOG_FILE="${LOG_FILE:-}"
declare -A LEVEL_NUM=([DEBUG]=0 [INFO]=1 [WARN]=2 [ERROR]=3)

log() {
    local level="$1"; shift
    local msg="$*"
    local timestamp
    timestamp=$(date "+%Y-%m-%d %H:%M:%S")

    # Level-Filter
    [[ ${LEVEL_NUM[$level]:-0} -lt ${LEVEL_NUM[$LOG_LEVEL]:-1} ]] && return

    # Farbcodes
    local color=""
    case "$level" in
        DEBUG) color="\033[36m" ;;   # Cyan
        INFO)  color="\033[32m" ;;   # Grün
        WARN)  color="\033[33m" ;;   # Gelb
        ERROR) color="\033[31m" ;;   # Rot

    local reset="\033[0m"

    local formatted="[$timestamp] [$level] $msg"

    # Auf Terminal: mit Farbe, sonst ohne
    if [[ -t 2 ]]; then
        printf "${color}%s${reset}\n" "$formatted" >&2
    else
        echo "$formatted" >&2
    fi

    # Optional in Datei schreiben
    [[ -n "$LOG_FILE" ]] && echo "$formatted" >> "$LOG_FILE"
}

#!/usr/bin/env bash

a=10; b=3

(( a-- ))           # Dekrement

# Dateien parallel verarbeiten mit Job-Control

set -euo pipefail

MAX_JOBS="${MAX_JOBS:-4}"   # Maximale parallele Prozesse

# Einfachste Methode: xargs -P
# Liste von URLs parallel herunterladen
urls=(
    "https://example.com/datei1.txt"
    "https://example.com/datei2.txt"
    "https://example.com/datei3.txt"
)
printf "%s\n" "${urls[@]}" | xargs -P "$MAX_JOBS" -I{} curl -O {}

# ── Eigene parallele Verarbeitung mit Job-Control ────────────
aktive_jobs=0

verarbeite() {
    local datei="$1"
    echo "Starte: $datei"
    # Schwere Verarbeitung...
    sleep $((RANDOM % 3 + 1))
    echo "Fertig: $datei"
}

warte_auf_freien_slot() {
    while (( aktive_jobs >= MAX_JOBS )); do
        # Auf irgendeinen Job warten
        wait -n 2>/dev/null && ((aktive_jobs--)) || true
    done
}

# Alle .log Dateien parallel komprimieren
shopt -s nullglob
log_dateien=(*.log)

if [[ ${#log_dateien[@]} -eq 0 ]]; then
    echo "Keine .log Dateien gefunden"
    exit 0
fi

echo "Verarbeite ${#log_dateien[@]} Dateien mit max. $MAX_JOBS Parallel-Jobs"

pids=()
for datei in "${log_dateien[@]}"; do
    warte_auf_freien_slot
    gzip -k "$datei" &   # Hintergrundprozess
    pids+=($!)
    ((aktive_jobs++))
    echo "Job gestartet: $datei (PID: ${pids[-1]})"
done

# Auf alle ausstehenden Jobs warten
echo "Warte auf verbleibende ${#pids[@]} Jobs..."
fehler_count=0
for pid in "${pids[@]}"; do
    if ! wait "$pid"; then
        echo "Job $pid fehlgeschlagen!" >&2
        ((fehler_count++))
    fi
done

echo "Fertig. Fehler: $fehler_count"
exit $fehler_count
