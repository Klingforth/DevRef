{ Unterschiede zu C-ähnlichen Sprachen:
  - begin...end statt {}
  - := für Zuweisung (= nur für Vergleich)

  - Kommentare: { } oder (* *) oder //
  - Case-insensitiv: begin = BEGIN = Begin }

Operatoren 
{ Arithmetik }
+   -   *          { Addition, Subtraktion, Multiplikation }
/                  { Division (gibt immer Real zurück!) }
div                { Integer-Division (= ganzzahlig) }
mod                { Modulo-Rest }

{ Wichtig: / gibt IMMER Real zurück! }
{ 7 / 2 = 3.5 (Real), 7 div 2 = 3 (Integer) }

{ Vergleich }
=    <>    <    <=    >    >=
{ <> ist "nicht gleich" (kein != wie in C!) }

{ Logisch }
and    or    not    xor

{ Bit-Operationen (auf Integer) }
and    or    xor    not    { auch bitweise! }
shl                        { Shift left:  n shl 2 = n * 4 }
shr                        { Shift right: n shr 1 = n div 2 }

{ Zuweisung }
:=                         { Zuweisung (NICHT =!) }
x := 42;
Inc(x);      { x := x + 1; }
Dec(x);      { x := x - 1; }
Inc(x, 5);   { x := x + 5; }

{ Zeiger-Operatoren }
^   { Pointer-Dereferenzierung: p^ gibt den Wert zurück }
@   { Adresse-Von: @x gibt den Zeiger auf x zurück }
p := @x;      { p zeigt auf x }
p^ := 99;     { Ändert den Wert, auf den p zeigt }

{ String-Konkatenation }
'Hallo' + ' ' + 'Welt'  { => "Hallo Welt" }

{ Mengen-Operatoren (Set) }
+   { Vereinigung }
-   { Differenz }
*   { Schnittmenge }
in  { Enthält: 'a' in ['a','e','i','o','u'] }

{ Generischer Stack }
generic TStack<T> = class
private
  FDaten:  array of T;
  FAnzahl: Integer;
public
  constructor Create;
  procedure Push(const AItem: T);
  function  Pop: T;
  function  Peek: T;
  function  IsEmpty: Boolean;
  property  Count: Integer read FAnzahl;
end;

constructor TStack.Create;
begin
  inherited Create;
  FAnzahl := 0;
  SetLength(FDaten, 16);
end;

procedure TStack.Push(const AItem: T);
begin
  if FAnzahl >= Length(FDaten) then
    SetLength(FDaten, Length(FDaten) * 2);
  FDaten[FAnzahl] := AItem;
  Inc(FAnzahl);
end;

function TStack.Pop: T;
begin
  if FAnzahl = 0 then
    raise Exception.Create('Stack ist leer!');
  Dec(FAnzahl);
  Result := FDaten[FAnzahl];
end;

function TStack.Peek: T;
begin
  if FAnzahl = 0 then
    raise Exception.Create('Stack ist leer!');
  Result := FDaten[FAnzahl - 1];
end;

function TStack.IsEmpty: Boolean;
begin
  Result := FAnzahl = 0;
end;

{ Verwendung }
type
  TIntStack    = specialize TStack<Integer>;
  TStringStack = specialize TStack<String>;

var
  iStack: TIntStack;
  sStack: TStringStack;
begin
  iStack := TIntStack.Create;
  try
    iStack.Push(1);
    iStack.Push(2);
    iStack.Push(3);
    WriteLn(iStack.Pop);   { 3 }
    WriteLn(iStack.Count); { 2 }
  finally
    iStack.Free;
  end;
end;

unit MathUtils;

{$mode objfpc}

interface

{ INTERFACE: Öffentlicher Teil (Deklarationen) }

uses
  SysUtils;

const
  PI_APPROX = 3.14159265358979323846;

type
  TPoint = record
    X, Y: Double;
  end;

function Abstand(P1, P2: TPoint): Double;
function Kreisflaeche(Radius: Double): Double;
function IstPrimzahl(n: Integer): Boolean;
procedure QuickSort(var arr: array of Integer; Links, Rechts: Integer);

implementation

{ IMPLEMENTATION: Privater Teil (Definitionen) }

function Abstand(P1, P2: TPoint): Double;
begin
  Result := Sqrt(Sqr(P2.X - P1.X) + Sqr(P2.Y - P1.Y));
end;

function Kreisflaeche(Radius: Double): Double;
begin
  Result := PI_APPROX * Sqr(Radius);
end;

function IstPrimzahl(n: Integer): Boolean;
var i: Integer;
begin
  if n < 2 then begin Result := False; Exit; end;
  if n = 2 then begin Result := True; Exit; end;
  if n mod 2 = 0 then begin Result := False; Exit; end;
  Result := True;
  i := 3;
  while i * i <= n do begin
    if n mod i = 0 then begin Result := False; Exit; end;
    Inc(i, 2);
  end;
end;

procedure QuickSort(var arr: array of Integer; Links, Rechts: Integer);
var pivot, temp, i, j: Integer;
begin
  if Links >= Rechts then Exit;
  pivot := arr[(Links + Rechts) div 2];
  i := Links; j := Rechts;
  repeat
    while arr[i] < pivot do Inc(i);
    while arr[j] > pivot do Dec(j);
    if i <= j then begin
      temp := arr[i]; arr[i] := arr[j]; arr[j] := temp;
      Inc(i); Dec(j);
    end;
  until i > j;
  QuickSort(arr, Links, j);
  QuickSort(arr, i, Rechts);
end;

{ Optionale Sections }
initialization
  { Wird beim Laden der Unit ausgeführt }
  Randomize;

finalization
  { Wird beim Beenden ausgeführt }

end.

  Insert(' schöne', s, 6);    { 'Hallo, schöne Welt!' }

  { Groß-/Kleinschreibung }
  t := UpperCase(s);           { 'HALLO, WELT!' }
  t := LowerCase(s);           { 'hallo, welt!' }
  t := AnsiUpperCase(s);       { Unicode-bewusst }

  { Trimmen }
  s := '  Hallo  ';
  t := Trim(s);                { 'Hallo' }
  t := TrimLeft(s);            { 'Hallo  ' }
  t := TrimRight(s);           { '  Hallo' }

  { Konkatenation }
  t := 'Hallo' + ', ' + 'Welt!';
  t := Concat('A', 'B', 'C');  { 'ABC' }

  { Konvertierung }
  t := IntToStr(42);            { '42' }
  t := FloatToStr(3.14);        { '3.14' }
  t := Format('%d + %d = %d', [2, 3, 5]);  { '2 + 3 = 5' }

  n := StrToIntDef('abc', -1);  { -1 (sicher, kein Exception) }

  { Zeichen-Zugriff }
  c := s[1];                   { Erstes Zeichen (1-basiert!) }
  s[1] := 'h';                 { Zeichen ändern }

  if s = t then WriteLn('Gleich');

    WriteLn('Gleich (case insensitive)');

  t := StringReplace(s, 'Welt', 'Pascal', [rfReplaceAll]);

  if StartsText('Hallo', s) then WriteLn('Beginnt mit Hallo');

  if ContainsText(s, 'Welt') then WriteLn('Enthält Welt');

program LinkedListDemo;
uses SysUtils;

type
  PNode = ^TNode;
  TNode = record
    Wert:    Integer;
    Naechster: PNode;
  end;

  TLinkedList = class
  private
    FKopf:  PNode;
    FAnzahl: Integer;
  public
    constructor Create;
    destructor  Destroy; override;
    procedure   PushFront(AWert: Integer);
    procedure   PushBack(AWert: Integer);
    function    PopFront: Integer;
    function    Contains(AWert: Integer): Boolean;
    procedure   Drucke;
    property    Count: Integer read FAnzahl;
    property    IsEmpty: Boolean read (FAnzahl = 0);
  end;

constructor TLinkedList.Create;
begin
  inherited Create;
  FKopf   := nil;
  FAnzahl := 0;
end;

destructor TLinkedList.Destroy;
var aktuell, naechster: PNode;
begin
  aktuell := FKopf;
  while aktuell <> nil do begin
    naechster := aktuell^.Naechster;
    Dispose(aktuell);
    aktuell := naechster;
  end;
  inherited Destroy;
end;

procedure TLinkedList.PushFront(AWert: Integer);
var neu: PNode;
begin
  New(neu);
  neu^.Wert      := AWert;
  neu^.Naechster := FKopf;
  FKopf          := neu;
  Inc(FAnzahl);
end;

procedure TLinkedList.PushBack(AWert: Integer);
var neu, aktuell: PNode;
begin
  New(neu);
  neu^.Wert      := AWert;
  neu^.Naechster := nil;
  if FKopf = nil then begin
    FKopf := neu;
  end else begin
    aktuell := FKopf;
    while aktuell^.Naechster <> nil do
      aktuell := aktuell^.Naechster;
    aktuell^.Naechster := neu;
  end;
  Inc(FAnzahl);
end;

function TLinkedList.PopFront: Integer;
var alt: PNode;
begin
  if FKopf = nil then
    raise Exception.Create('Liste ist leer!');
  Result := FKopf^.Wert;
  alt    := FKopf;
  FKopf  := FKopf^.Naechster;
  Dispose(alt);
  Dec(FAnzahl);
end;

function TLinkedList.Contains(AWert: Integer): Boolean;
var aktuell: PNode;
begin
  aktuell := FKopf;
  while aktuell <> nil do begin
    if aktuell^.Wert = AWert then begin
      Result := True; Exit;
    end;
    aktuell := aktuell^.Naechster;
  end;
  Result := False;
end;

procedure TLinkedList.Drucke;
var aktuell: PNode;
begin
  Write('[');
  aktuell := FKopf;
  while aktuell <> nil do begin
    Write(aktuell^.Wert);
    if aktuell^.Naechster <> nil then Write(' -> ');
    aktuell := aktuell^.Naechster;
  end;
  WriteLn(']');
end;

var liste: TLinkedList;
begin
  liste := TLinkedList.Create;
  try
    liste.PushBack(1); liste.PushBack(2); liste.PushBack(3);
    liste.PushFront(0);
    liste.Drucke;                          { [0 -> 1 -> 2 -> 3] }
    WriteLn('Enthält 2: ', liste.Contains(2));  { True }
    WriteLn('Pop: ', liste.PopFront);      { 0 }
    liste.Drucke;                          { [1 -> 2 -> 3] }
  finally
    liste.Free;
  end;
end.

  liste: TStringList;
  i: Integer;

  try
    liste.StrictDelimiter := True;
    liste.Delimiter       := Trenner[1];
    liste.DelimitedText   := Zeile;
    SetLength(Result, liste.Count);
    for i := 0 to liste.Count - 1 do
      Result[i] := Trim(liste[i]);
  finally
    liste.Free;
  end;

type
  TRecord = record
    Name:   String;
    Alter:  Integer;
    Gehalt: Double;
  end;

function LeseCSVDatei(const Dateiname: String): array of TRecord;
var
  f:     TextFile;
  zeile: String;
  felder: TCSVZeile;
  anzahl: Integer;
  erster: Boolean;
begin
  SetLength(Result, 0);
  erster := True;

  AssignFile(f, Dateiname);
  Reset(f);
  try
    while not EOF(f) do begin
      ReadLn(f, zeile);
      zeile := Trim(zeile);
      if zeile = '' then Continue;

      { Erste Zeile = Header überspringen }
      if erster then begin
        erster := False;
        Continue;
      end;

      felder := SplitCSV(zeile, ',');
      if Length(felder) >= 3 then begin
        anzahl := Length(Result);
        SetLength(Result, anzahl + 1);
        with Result[anzahl] do begin
          Name   := felder[0];
          Alter  := StrToIntDef(felder[1], 0);
          Gehalt := StrToFloatDef(felder[2], 0.0);
        end;
      end;
    end;
  finally
    CloseFile(f);
  end;
end;

{ Vereinfachter BST (Binary Search Tree) mit Generics }
type
  generic TBST<TKey, TValue> = class
  private
    type
      PNode = ^TNode;
      TNode = record
        Key:   TKey;
        Value: TValue;
        Links, Rechts: PNode;
      end;

    FWurzel: PNode;
    FAnzahl: Integer;

    procedure FreeNode(Node: PNode);
    function  FindNode(Node: PNode; const Key: TKey): PNode;
    function  InsertNode(Node: PNode; const Key: TKey; const Value: TValue): PNode;
    procedure InOrder(Node: PNode; Liste: TStringList; Convert: TObject);

  public
    constructor Create;
    destructor  Destroy; override;
    procedure   Insert(const Key: TKey; const Value: TValue);
    function    TryGet(const Key: TKey; out Value: TValue): Boolean;
    function    Contains(const Key: TKey): Boolean;
    property    Count: Integer read FAnzahl;
  end;

{ Implementierungen }
constructor TBST.Create;
begin inherited; FWurzel := nil; FAnzahl := 0; end;

destructor TBST.Destroy;
begin FreeNode(FWurzel); inherited; end;

procedure TBST.FreeNode(Node: PNode);
begin
  if Node = nil then Exit;
  FreeNode(Node^.Links);
  FreeNode(Node^.Rechts);
  Dispose(Node);
end;

function TBST.FindNode(Node: PNode; const Key: TKey): PNode;
begin
  if Node = nil then begin Result := nil; Exit; end;
  if Key < Node^.Key then
    Result := FindNode(Node^.Links, Key)
  else if Key > Node^.Key then
    Result := FindNode(Node^.Rechts, Key)
  else
    Result := Node;
end;

function TBST.InsertNode(Node: PNode; const Key: TKey; const Value: TValue): PNode;
begin
  if Node = nil then begin
    New(Node);
    Node^.Key    := Key;
    Node^.Value  := Value;
    Node^.Links  := nil;
    Node^.Rechts := nil;
    Inc(FAnzahl);
    Result := Node;
    Exit;
  end;
  if Key < Node^.Key then
    Node^.Links  := InsertNode(Node^.Links, Key, Value)
  else if Key > Node^.Key then
    Node^.Rechts := InsertNode(Node^.Rechts, Key, Value)
  else
    Node^.Value := Value;   { Update }
  Result := Node;
end;

procedure TBST.Insert(const Key: TKey; const Value: TValue);
begin
  FWurzel := InsertNode(FWurzel, Key, Value);
end;

function TBST.TryGet(const Key: TKey; out Value: TValue): Boolean;
var Node: PNode;
begin
  Node := FindNode(FWurzel, Key);
  Result := Node <> nil;
  if Result then Value := Node^.Value;
end;

function TBST.Contains(const Key: TKey): Boolean;
begin Result := FindNode(FWurzel, Key) <> nil; end;

end.
