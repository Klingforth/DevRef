#include <string>
#include <stdexcept>
#include <utility>

class Person {
public:
    // Konstruktor mit Member-Initializer-Liste (bevorzugt!)
    Person(std::string name, int alter)
        : name_{std::move(name)}   // Move statt Copy!
        , alter_{alter}
    {
        if (name_.empty())
            throw std::invalid_argument{"Name darf nicht leer sein"};
        if (alter_ < 0 || alter_ > 150)
            throw std::out_of_range{"Ungültiges Alter"};
    }

    // Destruktor (virtual für Basisklassen!)
    ~Person() = default;

    // Copy + Move automatisch generiert wenn kein Destruktor custom

    // Getter (const-Methoden!)
    const std::string& name()  const { return name_; }
    int                alter() const { return alter_; }

    // Setter
    void setAlter(int alter) {
        if (alter < 0) throw std::invalid_argument{"Negatives Alter"};
        alter_ = alter;
    }

    // Statische Methode
    static int anzahl() { return anzahl_; }

    // Operator-Overloading
    bool operator==(const Person& o) const {
        return name_ == o.name_ && alter_ == o.alter_;
    }

    auto operator<=>(const Person& o) const = default;  // C++20

    // Stream-Operator (als friend)
    friend std::ostream& operator<<(std::ostream& os, const Person& p) {
        return os << "Person{" << p.name_ << ", " << p.alter_ << "}";
    }

private:
    std::string name_;
    int         alter_;
    inline static int anzahl_ = 0;   // C++17: inline static
};

class Form {
public:
    virtual ~Form() = default;   // virtual Destruktor IMMER bei Vererbung!

    virtual double flaeche()  const = 0;   // pure virtual (abstrakt)
    virtual double umfang()   const = 0;
    virtual std::string name() const = 0;

    // Nicht-virtuelle Methode
    void drucke() const {
        std::cout << name() << ": Fläche=" << flaeche()
                  << " Umfang=" << umfang() << "
";
    }
};

// Abgeleitete Klasse
class Kreis final : public Form {   // final = keine weitere Ableitung
    double radius_;
public:
    explicit Kreis(double r) : radius_{r} {}

    double      flaeche()  const override { return M_PI * radius_ * radius_; }
    double      umfang()   const override { return 2 * M_PI * radius_; }
    std::string name()     const override { return "Kreis"; }
};

class Rechteck : public Form {
    double b_, h_;
public:
    Rechteck(double b, double h) : b_{b}, h_{h} {}

    double      flaeche()  const override { return b_ * h_; }
    double      umfang()   const override { return 2 * (b_ + h_); }
    std::string name()     const override { return "Rechteck"; }
};

// Rule of ZERO: Wenn Smart Pointer und STL-Container genutzt werden,
// braucht man NICHTS manuell definieren.

// Rule of FIVE: Wenn Rohe Ressourcen verwaltet werden:
class RawRessource {
    int* data_;
    size_t size_;
public:
    // 1. Konstruktor
    explicit RawRessource(size_t n)
        : data_{new int[n]}, size_{n} {}

    // 2. Destruktor
    ~RawRessource() { delete[] data_; }

    // 3. Copy-Konstruktor (tiefe Kopie)
    RawRessource(const RawRessource& o)
        : data_{new int[o.size_]}, size_{o.size_} {
        std::copy(o.data_, o.data_ + o.size_, data_);
    }

    // 4. Copy-Zuweisung
    RawRessource& operator=(const RawRessource& o) {
        if (this != &o) {
            delete[] data_;
            data_ = new int[o.size_];
            size_ = o.size_;
            std::copy(o.data_, o.data_ + o.size_, data_);
        }
        return *this;
    }

    // 5. Move-Konstruktor (übernimmt, gibt nichts zurück)
    RawRessource(RawRessource&& o) noexcept
        : data_{std::exchange(o.data_, nullptr)}
        , size_{std::exchange(o.size_, 0)} {}

    // 6. Move-Zuweisung
    RawRessource& operator=(RawRessource&& o) noexcept {
        if (this != &o) {
            delete[] data_;
            data_ = std::exchange(o.data_, nullptr);
            size_ = std::exchange(o.size_, 0);
        }
        return *this;
    }
};

// Concept: T muss kopierbar und ausgabefähig sein
template<typename T>
concept Stackable = std::copyable<T>;

template<Stackable T>
class Stack {
    std::vector<T> data_;

public:
    // Wert pushen (Copy oder Move)
    void push(T value) {
        data_.push_back(std::move(value));
    }

    // Top-Element holen (Referenz)
    T& top() {
        if (empty()) throw std::underflow_error{"Stack ist leer"};
        return data_.back();
    }

    const T& top() const {
        if (empty()) throw std::underflow_error{"Stack ist leer"};
        return data_.back();
    }

    // Pop – gibt Wert zurück (move!)
    T pop() {
        if (empty()) throw std::underflow_error{"Stack ist leer"};
        T val = std::move(data_.back());
        data_.pop_back();
        return val;
    }

    bool   empty() const { return data_.empty(); }
    size_t size()  const { return data_.size(); }

    // Iteration
    auto begin() const { return data_.rbegin(); }
    auto end()   const { return data_.rend(); }
};

int main() {
    Stack<int> s;
    s.push(1); s.push(2); s.push(3);

    std::println("Größe: {}", s.size());   // 3
    std::println("Top: {}",   s.top());    // 3

    while (!s.empty()) {
        std::print("{} ", s.pop());        // 3 2 1
    }
    std::println();

    // String-Stack
    Stack<std::string> ss;
    ss.push("Hallo");
    ss.push("Welt");
    std::println("{}", ss.pop());   // Welt
}

template<typename Derived>
class Form {
public:
    // Nicht-virtuelle Methoden rufen Derived-Methoden auf
    double flaeche() const {
        return static_cast<const Derived*>(this)->flaeche_impl();
    }
    double umfang() const {
        return static_cast<const Derived*>(this)->umfang_impl();
    }

    void drucke() const {
        std::cout << "Fläche: " << flaeche()
                  << " Umfang: " << umfang() << "
";
    }

    // Gemeinsame Logik
    bool istGrösserAls(const Form& o) const {
        return flaeche() > o.flaeche();
    }
};

class Kreis : public Form<Kreis> {
    double r_;
public:
    explicit Kreis(double r) : r_{r} {}
    double flaeche_impl() const { return M_PI * r_ * r_; }
    double umfang_impl()  const { return 2 * M_PI * r_; }
};

class Rechteck : public Form<Rechteck> {
    double b_, h_;
public:
    Rechteck(double b, double h) : b_{b}, h_{h} {}
    double flaeche_impl() const { return b_ * h_; }
    double umfang_impl()  const { return 2 * (b_ + h_); }
};

    Rechteck  r{4.0, 6.0};

    k.drucke();   // Fläche: 78.5398 Umfang: 31.4159
    r.drucke();   // Fläche: 24 Umfang: 20

// Algebraische Datentypen in C++ via std::variant
// Kein Heap, kein virtual, vollständig typsicher!

struct Zahl     { double wert; };
struct Addition { struct Expr* links; struct Expr* rechts; };
struct Mul      { struct Expr* links; struct Expr* rechts; };
struct Negation { struct Expr* inner; };

using ExprVariant = std::variant<Zahl, Addition, Mul, Negation>;

struct Expr {
    ExprVariant data;
    Expr(auto v) : data{v} {}
};

// Visitor mit overloaded (C++17 Pattern)
template<typename... Ts>
struct overloaded : Ts... { using Ts::operator()...; };

double auswerten(const Expr& e) {
    return std::visit(overloaded{
        [](const Zahl& z)     { return z.wert; },
        [&](const Addition& a) {
            return auswerten(*a.links) + auswerten(*a.rechts);
        },
        [&](const Mul& m) {
            return auswerten(*m.links) * auswerten(*m.rechts);
        },
        [&](const Negation& n) {
            return -auswerten(*n.inner);
        },
    }, e.data);
}

std::string stringify(const Expr& e) {
    return std::visit(overloaded{
        [](const Zahl& z)     { return std::format("{}", z.wert); },
        [&](const Addition& a) {
            return std::format("({} + {})",
                stringify(*a.links), stringify(*a.rechts));
        },
        [&](const Mul& m) {
            return std::format("({} * {})",
                stringify(*m.links), stringify(*m.rechts));
        },
        [&](const Negation& n) {
            return std::format("(-{})", stringify(*n.inner));
        },
    }, e.data);
}

int main() {
    // (3 + 4) * (-2)
    auto drei  = Expr{Zahl{3}};
    auto vier  = Expr{Zahl{4}};
    auto minus2 = Expr{Negation{new Expr{Zahl{2}}}};
    auto summe = Expr{Addition{&drei, &vier}};
    auto prod  = Expr{Mul{&summe, &minus2}};

    std::println("{} = {}", stringify(prod), auswerten(prod));
    // ((3 + 4) * (-2)) = -14
}
